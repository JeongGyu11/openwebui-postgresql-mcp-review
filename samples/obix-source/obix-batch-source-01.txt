package kr.fasol.obix.batch.biz.common.service;

import java.text.SimpleDateFormat;
import java.util.Calendar;
import java.util.Date;
import java.util.HashMap;

import java.util.Map;

import jakarta.annotation.Resource;

import kr.fasol.obix.batch.dao.comm.BizDayDAO;
import kr.fasol.obix.batch.dto.comm.HoldyBasQryDTO;

/**
 * 시스템명 : 
 * 업무구분명 : 공통 > 영업일조회
 * 프로그램명 : 영업일조회
 * 설명 :

@Service("BizDayService")
public class BizDayService {

    @Resource(name = "BizDayDAO")
    private BizDayDAO bizDayDAO;

    /**
     * 기준일로 부터 영업일을 조회한다.
     * @param baseDt 기준일
     * @param day - 0:당영업일, -1:전영업일, -2:전전영업일, 1:익영업일, 2:익익영업일
     * @return 영업일
     */
	public Date getBizDay(Date baseDt, int day) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);

		// 휴일목록 조회
		HoldyBasQryDTO req = new HoldyBasQryDTO();
		if (day < 0) {
			req.setStartDt(sdf.format(DateUtils.addMonths(baseDt, 0 - ((day / 15) + 1))));
			req.setEndDt(sdf.format(baseDt));
		} else {
			req.setStartDt(sdf.format(baseDt));
			req.setEndDt(sdf.format(DateUtils.addMonths(baseDt, ((day / 15) + 1))));
		}

		List<HoldyBasQryDTO> list = bizDayDAO.listHoldyBas(req);

		// 휴일 목록을 일자별 Map으로 변환
		Map<String, Object> bizDayMap = new HashMap<>();
		if (list != null && !list.isEmpty()) {
			bizDayMap = list.stream().collect(Collectors.toMap(HoldyBasQryDTO::getHoldyYmd, HoldyBasQryDTO::clone));
		}

		/*
		 * baseDt 기준 영업일 조회
		 */
		Calendar start = Calendar.getInstance();
		start.setTime(baseDt);

		if (day == 0) {
			while (true) {
				String targetDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(targetDt)) {
					break;
				}
				start.add(Calendar.DATE, 1);
			}
		} else {
			int bizDtCnt = 0;
			while (bizDtCnt < (day < 1 ? 0 - day : day)) {
				start.add(Calendar.DATE, (day < 1 ? -1 : 1));
				String targetDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(targetDt)) {
					bizDtCnt++;
				}
			}
		}

		return start.getTime();
	}

    /**
     * 기준일로 부터 영업일을 조회한다.
     * @param baseDt 기준일(yyyyMMdd)
     * @param day - 0:당영업일, -1:전영업일, -2:전전영업일, 1:익영업일, 2:익익영업일
     * @return 영업일(yyyyMMdd)
     */
	@Cacheable(cacheNames = "BIZDAY_LIST", key = "#baseDt + ':' + #day")
	public String getBizDay(String baseDt, int day) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);
		return sdf.format( getBizDay(sdf.parse(baseDt), day) );
	}


	/**
	 * 기준일로부터 대상일자가 몇 영업일인지 계산
	 * @param baseDt 기준일
	 * @param targetDt 대상일자
	 * @return - 0:당영업일, 1:익영업일, 2:익익영업일
	 */
	public int getBizDayDiff(Date baseDt, Date targetDt) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);

		// 휴일목록 조회
		HoldyBasQryDTO req = new HoldyBasQryDTO();
		req.setStartDt(sdf.format(baseDt));
		req.setEndDt(sdf.format(targetDt));

		List<HoldyBasQryDTO> list = bizDayDAO.listHoldyBas(req);

		// 휴일 목록을 일자별 Map으로 변환
		Map<String, Object> bizDayMap = new HashMap<>();
		if (list != null && !list.isEmpty()) {
			bizDayMap = list.stream().collect(Collectors.toMap(HoldyBasQryDTO::getHoldyYmd, HoldyBasQryDTO::clone));
		}


		int day = 0;
		if (!baseDt.equals(targetDt)) {
			// baseDt 기준 영업일 조회
			Calendar start = Calendar.getInstance();
			start.setTime(baseDt);

			while (true) {
				start.add(Calendar.DATE, 1);
				String diffDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(diffDt)) {
					day++;
				}
				if (targetDt.equals(sdf.parse(diffDt))) {
					 break;
				}
			}
		}

		return day;
	}

	/**
	 * 기준일로부터 대상일자가 몇 영업일인지 계산
	 * @param baseDt 기준일(yyyyMMdd)
	 * @param targetDt 대상일자(yyyyMMdd)
	 * @return - 0:당영업일, 1:익영업일, 2:익익영업일
	 */
	public int getBizDayDiff(String baseDt, String targetDt) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);
		return getBizDayDiff(sdf.parse(baseDt), sdf.parse(targetDt));
	}

}

/**
 * 시스템명 : Batch
 * 업무구분명 : 공통 > Service > 배치실행내역 처리

@Slf4j
@Service("BtchExeTxnService")
public class BtchExeTxnService {

    @Resource(name = "BtchExeTxnDAO")
	private BtchExeTxnDAO btchExeTxnDAO;

	/**
	 * 배치실행내역 조회
	 */
	public List<BtchExeTxnDTO> listBtchExeTxn(BtchExeTxnDTO req) throws Exception {
		return btchExeTxnDAO.listBtchExeTxn(req);
	}

	/**
	 * 배치기본 상세 조회
	 */
	public BtchBasDTO selectBtchBas(BtchBasDTO req) throws Exception {
		return btchExeTxnDAO.selectBtchBas(req);
	}


	/**
     * 배치실행내역 - 시작
     */
	public BtchExeTxnQryDTO startBtchExeTxn(BtchExeTxnQryDTO input) throws Exception {
		BtchExeTxnQryDTO output = input.clone(BtchExeTxnQryDTO.class);

		if (StringUtils.isBlank(input.getUserId())) {
			input.setUserId("BATCH");
		}
		input.setBtchExeStusCd(CodeConstant.BTCH_EXE_STUS_CD.실행중.getCode());
		
		Integer result = btchExeTxnDAO.insertBtchExeTxn(input);
		if (Objects.isNull(result) || result < 1 || Objects.isNull(input.getBtchExeSeqno())) { 
			throw new SQLException("배치실행내역 입력 실패"); 
		}

		output.setBtchExeSeqno(input.getBtchExeSeqno());
		return output;
	}
	
	/**
     * 배치실행내역 - 종료
     */
	public BtchExeTxnQryDTO endBtchExeTxn(BtchExeTxnQryDTO input) throws Exception {
		String btchExeTxn = "";
		
		if(input.isSuccess()) {
			//배치실행내역-수정
			if(input.getTotalCnt() ==0) {
				btchExeTxn = "";//배치실행 대상이 없습니다.";
			}else if(input.getFailCnt() ==0) {
				btchExeTxn = "";//배치실행이 전체성공되었습니다.(전체갯수:" + input.getTotalCnt() + ",성공갯수:"+ input.getSuccCnt() + ")";
			}else if(input.getSuccCnt() ==0) {
				btchExeTxn = "배치실행이 실패되었습니다.(전체갯수:" + input.getTotalCnt() + ",실패갯수:"+ input.getFailCnt() + ")";
			}else {
				btchExeTxn = "배치실행이 부분성공되었습니다.(전체갯수:" + input.getTotalCnt() + ",성공갯수:"+ input.getSuccCnt() + ",실패갯수:"+ input.getFailCnt() + ")";
			}
			log.info("{} 결과요약 :: {} ", input.getBtchJobId(), btchExeTxn);
			
			if(StringUtils.isNotBlank(input.getResultMesg()))
				btchExeTxn = btchExeTxn.concat(input.getResultMesg());
			input.setBtchPrmtVal(input.getJobBaseDt());      //배치파라미터값//VARCHAR2(300) //String //
			input.setTagetTrtCscnt(new BigDecimal(input.getTotalCnt()));      //대상처리건수//NUMBER(20) //Long //
			input.setBtchExeStusCd(CodeConstant.BTCH_EXE_STUS_CD.완료.getCode());      //배치실행상태코드//VARCHAR2(32) //String //
			input.setSucssYn(input.getFailCnt()==0?"Y":"N");      //성공여부//VARCHAR2(1) //String //
			input.setBtchErrCtnt(btchExeTxn);      //배치오류내용//VARCHAR2(4000) //String //
			Integer result = btchExeTxnDAO.updateBtchExeTxn(input);
			if (result < 1) { 
				throw new SQLException("배치실행내역 수정 실패"); 
			}
		}else {
			if(Objects.nonNull(input.getBtchExeSeqno())) {
				input.setBtchPrmtVal(input.getJobBaseDt());      //배치파라미터값//VARCHAR2(300) //String //
				input.setBtchExeStusCd(CodeConstant.BTCH_EXE_STUS_CD.오류.getCode());      //배치실행상태코드//VARCHAR2(32) //String //
				if(input.getResultMesg() != null) {
					input.setBtchErrCtnt(input.getResultMesg());
				}
				Integer result = btchExeTxnDAO.updateBtchExeTxn(input);
				if (result < 1) { 
					throw new SQLException("배치실행내역 수정 실패"); 
				}
			}
		}
		BtchExeTxnQryDTO output = input.clone(BtchExeTxnQryDTO.class);
		return output;
	}

}

    private CmnCdDAO cmnCdDAO;


	/**
	 * 코드그룹 조회
	 */
    @Cacheable(cacheNames = "CMN_CD_GROUP")
	public List<CmnCdGroupDTO> getListCmnCdGroup() throws Exception {
		return cmnCdDAO.listCmnCdGroup();
	}

    /**
     * 코드그룹 조회
     * @param cdGroupVal 코드그룹값
     */
	@Cacheable(cacheNames = "CMN_CD_GROUP", key = "#cdGroupVal")
    public CmnCdGroupDTO getCmnCdGroup(String cdGroupVal) throws Exception {
    	List<CmnCdGroupDTO> list = getListCmnCdGroup();
    	if (list != null && !list.isEmpty()) {
    		return list.stream().filter(x -> x.getCdGroupVal().equals(cdGroupVal)).findFirst().orElse(null);
    	}
    	return null;
    }

	/**
	 * 코드목록 조회
	 * @param req 코드그룹값
	 */
    @Cacheable(cacheNames = "CMN_CD", key = "#req.cdGroupVal")
	public List<CmnCdQryDTO> getListCmnCd(CmnCdQryDTO req) throws Exception {
		return cmnCdDAO.listCmnCdByCdGroupVal(req);
	}

	/**

	 */
	@Cacheable(cacheNames = "CMN_CD", key = "#cdGroupVal")
	public List<CmnCdQryDTO> getListCmnCd(String cdGroupVal) throws Exception {
		return getListCmnCd(CmnCdQryDTO.builder().cdGroupVal(cdGroupVal).build());

package kr.fasol.obix.batch.biz;

import java.util.Map;

import org.springframework.web.bind.annotation.RequestMapping;

public class HealthCheckController {

	@RequestMapping(method = {RequestMethod.GET, RequestMethod.HEAD})

    /**
     * 배치 Job 목록
     */
    @RequestMapping("/jobList.json")
    public Response<?> jobList(HttpServletRequest request) throws Exception {
        List<Map<String, Object>> output = new ArrayList<>();

        List<BtchBasDTO> jobList = batchService.getListBtchBas();
        if (jobList != null) {
            for (BtchBasDTO dto : jobList) {
                Map<String, Object> map = new HashMap<>();
                map.put("btchJobId", dto.getBtchJobId());
                map.put("btchJobNm", dto.getBtchJobNm());
                map.put("btchExePgmNm", dto.getBtchExePgmNm());
                output.add(map);
            }
        }

        return Response.ok(output);
    }

    /**
     * 배치 Job 실행
     */
    @RequestMapping("/jobLauncher.json")
    public Response<?> jobLauncher(@ModelAttribute BtchBasDTO input, HttpServletRequest request) throws Exception {

        BtchBasDTO btchBasDTO = btchExeTxnDAO.selectBtchBas(input);
        if (btchBasDTO == null) {
            throw new UserException("BIZ.BATCH.01").setDebuggingMessage("[등록된 Job 없음] 배치작업ID = " + input.getBtchJobId());
        }

        /*
         * 입력 파라미터 설정
         */
        Map<String, JobParameter<?>> parameters = new HashMap<>();
        for (String key : request.getParameterMap().keySet()) {
            parameters.put(key, new JobParameter<>(request.getParameter(key), String.class));
        }

        // 재실행가능여부
        if (!parameters.containsKey("executeTime")) {
            if ("Y".equals(btchBasDTO.getReExePosblYn())) {
                parameters.put("executeTime", new JobParameter<>(DateUtil.getCurrentDateTime(), String.class));
            } else {
                parameters.put("executeTime", new JobParameter<>(DateUtil.getCurrentDate(), String.class));
            }
        }

        batchService.asyncExecuteJobLauncher(btchBasDTO, new JobParameters(parameters));
        return Response.SUCCESS;
    }

    /**
     * Local 실행용 배치 Job 실행
     */
    @RequestMapping("/localJobLauncher.json")
    public Response<?> localJobLauncher(@ModelAttribute BtchBasDTO input, HttpServletRequest request) throws Exception {

        // 로컬 환경에서만 실행
        if (!ProfileUtil.isLocal()) {
            return Response.builder().status("CB1004", "로컬 개발 환경에서만 실행 가능합니다.").build();
        }

        /*
         * 입력 파라미터 설정
         */
        Map<String,JobParameter<?>> parameters = new HashMap<>();
        for (String key : request.getParameterMap().keySet()) {
            parameters.put(key, new JobParameter<>(request.getParameter(key),String.class));
        }
        parameters.put("executeTime", new JobParameter<>(DateUtil.getCurrentDateTime(), String.class));

        return batchService.executeJobLauncher(input, new JobParameters(parameters));
    }

@Slf4j
public class CustomCacheEntryListener implements CacheEntryCreatedListener<Object, Object>, CacheEntryUpdatedListener<Object, Object>, CacheEntryRemovedListener<Object, Object>, CacheEntryExpiredListener<Object, Object> {

    @Override
    public void onCreated(Iterable<CacheEntryEvent<?, ?>> iterable) throws CacheEntryListenerException {
        iterable.forEach(cacheEntryEvent -> {
            log.info("Cache Entry Created. Cache Event: {} / Key: {} / Value:{}",
                    cacheEntryEvent.getEventType(),
                    cacheEntryEvent.getKey(),
                    cacheEntryEvent.getValue());
        });
    }

    @Override
    public void onUpdated(Iterable<CacheEntryEvent<?, ?>> iterable) throws CacheEntryListenerException {
        iterable.forEach(cacheEntryEvent -> {
            log.info("Cache Entry Updated. Cache Event: {} / Key: {} / Old Value: {} / New Value:{}",
                    cacheEntryEvent.getEventType(),
                    cacheEntryEvent.getKey(),
                    cacheEntryEvent.getOldValue(),
                    cacheEntryEvent.getValue());
        });
    }

    @Override
    public void onRemoved(Iterable<CacheEntryEvent<?, ?>> iterable) throws CacheEntryListenerException {
        iterable.forEach(cacheEntryEvent -> {
            log.info("Cache Entry Removed. Cache Event: {} / Key: {} / Value:{}",
                    cacheEntryEvent.getEventType(),
                    cacheEntryEvent.getKey(),
                    cacheEntryEvent.getValue());
        });
    }

    @Override
    public void onExpired(Iterable<CacheEntryEvent<?, ?>> iterable) throws CacheEntryListenerException {
        iterable.forEach(cacheEntryEvent -> {
            log.info("Cache Entry Expired. Cache Event: {} / Key: {} / Value:{}",
                    cacheEntryEvent.getEventType(),

    }

	private void processScheduledTask(Scheduled scheduled, Runnable runnable, Method method, Object bean) {
		try {
			boolean processedSchedule = false;
			String errorMessage = "Exactly one of the 'cron', 'fixedDelay' or 'fixedRate' attributes is required";
			Set<ScheduledTask> tasks = new LinkedHashSet<>(4);
			Duration initialDelay = toDuration(scheduled.initialDelay(), scheduled.timeUnit());
			String initialDelayString = scheduled.initialDelayString();
			if (StringUtils.hasText(initialDelayString)) {
				Assert.isTrue(initialDelay.isNegative(), "Specify 'initialDelay' or 'initialDelayString', not both");
				if (this.embeddedValueResolver != null) {
					initialDelayString = this.embeddedValueResolver.resolveStringValue(initialDelayString);
				}

				if (StringUtils.hasLength(initialDelayString)) {
					try {
						initialDelay = toDuration(initialDelayString, scheduled.timeUnit());
					} catch (RuntimeException e) {
						throw new IllegalArgumentException("Invalid initialDelayString value \"" + initialDelayString + "\"; " + e);
					}
				}
			}

			String cron = scheduled.cron();
			if (StringUtils.hasText(cron)) {
				String zone = scheduled.zone();
				if (this.embeddedValueResolver != null) {
					cron = this.embeddedValueResolver.resolveStringValue(cron);
					zone = this.embeddedValueResolver.resolveStringValue(zone);
				}

				if (StringUtils.hasLength(cron)) {
					Assert.isTrue(initialDelay.isNegative(), "'initialDelay' not supported for cron triggers");
					processedSchedule = true;
					if (!"-".equals(cron)) {
						CronTrigger trigger;
						if (StringUtils.hasText(zone)) {
							trigger = new CronTrigger(cron, StringUtils.parseTimeZoneString(zone));
						} else {
							trigger = new CronTrigger(cron);
						}

						tasks.add(this.registrar.scheduleCronTask(new CronTask(runnable, trigger)));
					}
				}
			}

			Duration delayToUse = initialDelay.isNegative() ? Duration.ZERO : initialDelay;
			Duration fixedDelay = toDuration(scheduled.fixedDelay(), scheduled.timeUnit());
			if (!fixedDelay.isNegative()) {
				Assert.isTrue(!processedSchedule, errorMessage);
				processedSchedule = true;
				tasks.add(this.registrar.scheduleFixedDelayTask(new FixedDelayTask(runnable, fixedDelay, delayToUse)));
			}

			String fixedDelayString = scheduled.fixedDelayString();
			if (StringUtils.hasText(fixedDelayString)) {
				if (this.embeddedValueResolver != null) {
					fixedDelayString = this.embeddedValueResolver.resolveStringValue(fixedDelayString);
				}

				if (StringUtils.hasLength(fixedDelayString)) {
					Assert.isTrue(!processedSchedule, errorMessage);
					processedSchedule = true;

					try {
						fixedDelay = toDuration(fixedDelayString, scheduled.timeUnit());
					} catch (RuntimeException e) {
						throw new IllegalArgumentException("Invalid fixedDelayString value \"" + fixedDelayString + "\"; " + e);
					}

					tasks.add(this.registrar.scheduleFixedDelayTask(new FixedDelayTask(runnable, fixedDelay, delayToUse)));
				}
			}

			Duration fixedRate = toDuration(scheduled.fixedRate(), scheduled.timeUnit());
			if (!fixedRate.isNegative()) {
				Assert.isTrue(!processedSchedule, errorMessage);
				processedSchedule = true;
				tasks.add(this.registrar.scheduleFixedRateTask(new FixedRateTask(runnable, fixedRate, delayToUse)));
			}

			String fixedRateString = scheduled.fixedRateString();
			if (StringUtils.hasText(fixedRateString)) {
				if (this.embeddedValueResolver != null) {

					}

					tasks.add(this.registrar.scheduleFixedRateTask(new FixedRateTask(runnable, fixedRate, delayToUse)));
				}
			}

			if (!processedSchedule) {
				if (initialDelay.isNegative()) {
					throw new IllegalArgumentException("One-time task only supported with specified initial delay");
				}

				tasks.add(this.registrar.scheduleOneTimeTask(new OneTimeTask(runnable, delayToUse)));
			}

			synchronized(this.scheduledTasks) {
				Set regTasks = (Set)this.scheduledTasks.computeIfAbsent(bean, (key) -> {

	public void createScheduledTask(final String beanName, final String methodName, final String cronExpression, Object... args) {
		Object beanClass;
		try {
			if (beanName != null && beanName.contains(".")) {
				beanClass = applicationContext.getBean(ClassUtils.getClass(beanName));
			} else {
				beanClass = applicationContext.getBean(beanName);
			}
		} catch (Exception e) {
    		log.error("[등록된 Job 없음] 배치작업코드 = {}.{}", beanName, methodName);
    		return;
		}

		Method jobMethod;
		if (args != null && args.length > 0) {
			jobMethod = MethodUtils.getAccessibleMethod(beanClass.getClass(), methodName, ClassUtils.toClass(args));
		} else {
			jobMethod = MethodUtils.getAccessibleMethod(beanClass.getClass(), methodName);
		}
		if (jobMethod == null) {
			log.error("### 배치 스케쥴러 등록 실패 = {}.{}", beanName, methodName);
			return;
		}

		Scheduled scheduledMethod = new Scheduled() {
			public Class<? extends Annotation> annotationType() { return null; }
			public String cron() { return cronExpression; }
			public String zone() { return null; }
			public long fixedDelay() { return -1; }
			public String fixedDelayString() { return null; }
			public long fixedRate() { return -1; }
			public String fixedRateString() { return null; }
			public long initialDelay() { return -1; }
			public String initialDelayString() { return null; }
			public TimeUnit timeUnit() { return TimeUnit.MILLISECONDS; }
			public String scheduler() { return ""; }
		};

		processScheduledSync(scheduledMethod, jobMethod, beanClass, args);
	}

	/**
	 * Scheduled 삭제
	 */
	public void removeScheduledTask(final String beanName, final String methodName) {
		Object beanClass;
		try {
			if (beanName != null && beanName.contains(".")) {
				beanClass = applicationContext.getBean(ClassUtils.getClass(beanName));
			} else {
				beanClass = applicationContext.getBean(beanName);
			}
		} catch (Exception e) {
    		log.error("[등록된 Job 없음] 배치작업코드 = {}.{}", beanName, methodName);
    		return;
		}

		postProcessBeforeDestruction(beanClass, null);
	}

	/**
	 * Scheduled 목록 조회
	 */
	public List<ScheduledTask> getListScheduledTask() {
		return new ArrayList<>(getScheduledTasks());
	}

	/**
	 * ScheduledTask 실행 중인지 여부
	 */
	public boolean isExecuteTask(final String className, final String methodName) {
		return (getScheduledTasks().stream().anyMatch(x -> x.getTask().toString().equals(className + "." + methodName)));
	}

	/**
	 * 등록된 Cron Task 목록 조회
	 */
	public List<CronTask> getListCronTask(String className, String methodName, String cronExpression) {
		List<CronTask> list = new ArrayList<>();
		for (ScheduledTask task : getListScheduledTask()) {
			CronTask cronTask = (CronTask) task.getTask();
			ScheduledMethodRunnable runnable = (ScheduledMethodRunnable) cronTask.getRunnable();
			log.debug("### ScheduledTask List = {}, {}, {}",
					runnable.getTarget().getClass().getSuperclass().getName(),
					runnable.getMethod().getName(),
					cronTask.getExpression());
		}
		return list;
	}

}

import java.lang.reflect.InvocationTargetException;
import java.lang.reflect.Method;
import java.lang.reflect.UndeclaredThrowableException;

import org.springframework.scheduling.support.ScheduledMethodRunnable;
import org.springframework.util.ReflectionUtils;

/**
 * 설명 : 배치 스케쥴러 등록을 위한 ScheduledMethodRunnable를 상속받는 Custom 클래스
 * - Cron 스케쥴러에 파라미터를 추가할 수 있도록 수정
 *
 * @Class  CustomScheduledMethodRunnable.java
 * @author  정영탁
 * @since   2023. 3. 16.
 * @version 1.0
 *
 * @Copyright (c) Finance All Solutions.
 *--------------------------------------------------------------------
 * Modification Information
 *--------------------------------------------------------------------
 * 수정일               수정자            수정내용
 *--------------------------------------------------------------------
 * 2023. 3. 16.        정영탁          최초생성
 */
public class CustomScheduledMethodRunnable extends ScheduledMethodRunnable {

	private Object[] args;

	public CustomScheduledMethodRunnable(Object target, Method method, Object... args) {
		super(target, method);
		this.args = args;
	}

	/**
	 * Create a {@code ScheduledMethodRunnable} for the given target instance,
	 * calling the specified method by name.
	 * @param target the target instance to call the method on
	 * @param methodName the name of the target method
	 * @throws NoSuchMethodException if the specified method does not exist
	 */
	public CustomScheduledMethodRunnable(Object target, String methodName, Object... args) throws NoSuchMethodException {
		super(target, methodName);
		this.args = args;
	}


	@Override
	public void run() {
		try {
			ReflectionUtils.makeAccessible(getMethod());
			//TODO [Custom] CronTask 호출 시 파라미터 추가
			if (this.args != null && this.args.length > 0) {
				getMethod().invoke(getTarget(), this.args);
			} else {
				getMethod().invoke(getTarget());
			}
		}
		catch (InvocationTargetException ex) {
			ReflectionUtils.rethrowRuntimeException(ex.getTargetException());
		}
		catch (IllegalAccessException ex) {
			throw new UndeclaredThrowableException(ex);
		}
	}

}

package kr.fasol.obix.batch.job;

import jakarta.annotation.Resource;

 * 프로그램명 : 배치 서비스 관리

@Slf4j
@Service("BatchService")
public class BatchService {

	private static final Logger criticalLog = LoggerFactory.getLogger("CRITICAL.ERROR.LOGGER");

    @Resource(name = "BtchExeTxnDAO")
	private BtchExeTxnDAO btchExeTxnDAO;

	//Batch Job의 실행을 위한 제어에 사용되는 JobLauncher
	@Autowired
	private JobLauncher jobLauncher;
	//Batch Job의 시작, 정지, 재시작에 사용되는 JobOperator
	@Autowired
	private JobOperator jobOperator;
	//JobExecution 조회 등에 사용되는 JobExplorer
	@Autowired
	private JobExplorer jobExplorer;
	//가장 최근에 실행된 JobExecution, StepExecution 조회 등에 사용되는 JobRepository
	@Autowired
	private JobRepository jobRepository;

	/**
	 * 배치 목록
	 */
	public List<BtchBasDTO> getListBtchBas() throws Exception {
		List<BtchBasDTO> jobList = new ArrayList<>();

		List<BtchBasDTO> btchBasList = btchExeTxnDAO.listBtchBas(BtchBasDTO.newBuilder().build());
        String jobName;
        for (BtchBasDTO dto : btchBasList) {
            if (StringUtils.isNotBlank(dto.getBtchExePgmNm())) {
                jobName = dto.getBtchExePgmNm();
            } else {
                jobName = dto.getBtchJobId() + "JOB";
            }
            dto.setBtchExePgmNm(jobName);
            jobList.add(dto);
        }

        return jobList;
	}

    /**
     * 배치 실행
     */
    public Response<?> executeJobLauncher(BtchBasDTO input, Map<String, String> paramMap) throws Exception {
		/*
		 * 입력 파라미터 설정
		 */
		Map<String,JobParameter<?>> parameters = new HashMap<>();
		for (Map.Entry<String, String> map : paramMap.entrySet()) {
			parameters.put(map.getKey(), new JobParameter<>(map.getValue(), String.class));
		}

		// 재실행가능여부
		if (!parameters.containsKey("executeTime")) {
			if ("Y".equals(input.getReExePosblYn())) {
				parameters.put("executeTime", new JobParameter<>(DateUtil.getCurrentDateTime(), String.class));
			} else {
				parameters.put("executeTime", new JobParameter<>(DateUtil.getCurrentDate(), String.class));
			}
		}

		return executeJobLauncher(input, new JobParameters(parameters));
    }

    /**
     * 배치 실행
     */
    public Response<?> executeJobLauncher(BtchBasDTO input, HttpServletRequest request) throws Exception {
		/*
		 * 입력 파라미터 설정
		 */
		Map<String,JobParameter<?>> parameters = new HashMap<>();
		for (String key : request.getParameterMap().keySet()) {
			parameters.put(key, new JobParameter<>(request.getParameter(key), String.class));
		}

		// 재실행가능여부
		if (!parameters.containsKey("executeTime")) {
			if ("Y".equals(input.getReExePosblYn())) {
				parameters.put("executeTime", new JobParameter<>(DateUtil.getCurrentDateTime(), String.class));
			} else {
				parameters.put("executeTime", new JobParameter<>(DateUtil.getCurrentDate(), String.class));
			}
		}

		return executeJobLauncher(input, new JobParameters(parameters));
    }

    /**
     * Async로 배치 실행
     */
    @Async
    public void asyncExecuteJobLauncher(BtchBasDTO input, JobParameters jobParameters) throws Exception {
		Response<?> res = executeJobLauncher(input, jobParameters);

		if (log.isInfoEnabled()) {
			log.info("### 배치 실행 결과 = {}", res);
		}
    }

    /**
     * 배치 실행
     */
    public Response<?> executeJobLauncher(BtchBasDTO input, JobParameters jobParameters) throws Exception {
		Map<String, Object> output = new LinkedHashMap<>();

		String jobName;
		if (StringUtils.isNotBlank(input.getBtchExePgmNm())) {
			jobName = input.getBtchExePgmNm();
		} else {
			jobName = input.getBtchJobId() + "JOB";
		}

		try {
			Job job = SpringUtil.getBean(jobName, Job.class);

			if (log.isDebugEnabled()) {
				log.debug("### BATCH JOB_NAME = {}", job.getName());
				log.debug("### BATCH JOB_PARAMETERS = {}", jobParameters);
			}

			JobExecution jobExecution = jobLauncher.run(job, new JobParametersBuilder(jobParameters).addString("btchJobId", input.getBtchJobId()).toJobParameters());

			output.put("btchJobId", input.getBtchJobId());
			output.put("btchJobNm", input.getBtchJobNm());
			output.put("jobName", jobExecution.getJobInstance().getJobName());
			output.put("createTime", jobExecution.getCreateTime());
			output.put("startTime", jobExecution.getStartTime());
			output.put("endTime", jobExecution.getEndTime());
			output.put("exitStatus", jobExecution.getExitStatus().getExitCode());


			/*
			 * Batch Job Service에서의 에러 처리
			 */
			if (StringUtils.equals(jobExecution.getExitStatus().getExitCode(), ExitStatus.FAILED.getExitCode())) {
				UserException userException = (UserException) jobExecution.getAllFailureExceptions().stream().filter(x -> x.getClass().equals(UserException.class)).findFirst().orElse(null);
				if (userException != null) {
					log.error("{}", userException.toString(), userException);
				}
				CriticalException criticalException = (CriticalException) jobExecution.getAllFailureExceptions().stream().filter(x -> x.getClass().equals(CriticalException.class)).findFirst().orElse(null);
				if (criticalException != null) {
					log.error("{}", criticalException.toString(), criticalException);
					// EMS 연동을 위한 Critical Error 출력
					criticalLog.error("{} [{}]{}", criticalException.getOrgErrorType(), criticalException.getErrorCode(), AppMessageUtil.getMessage(criticalException.getErrorCode()));
				}
			}

		}
		catch (NoSuchBeanDefinitionException e) {
			log.error(e.getMessage(), e);
			throw new UserException("BIZ.BATCH.01", e).setDebuggingMessage("[등록된 Job 없음] 배치작업ID = " + input.getBtchJobId());
		}
		catch (JobExecutionAlreadyRunningException | JobInstanceAlreadyCompleteException e) {
			log.error(e.getMessage(), e);
			throw new UserException("BIZ.BATCH.02", e).setDebuggingMessage("[Job 실행 오류] 배치작업ID = " + input.getBtchJobId());
		}
		catch (JobRestartException e) {
			log.error(e.getMessage(), e);
			throw new UserException("BIZ.BATCH.03", e).setDebuggingMessage("[Job 실행 오류] 배치작업ID = " + input.getBtchJobId());
		}
		catch (JobParametersInvalidException e) {
			log.error(e.getMessage(), e);
			throw new UserException("BIZ.BATCH.04", e).setDebuggingMessage("[Job 실행 오류] 배치작업ID = " + input.getBtchJobId());
		}
		catch (Exception e) {
			log.error(e.getMessage(), e);
			throw new UserException("BIZ.BATCH.05", e).setDebuggingMessage("[Job 실행 오류] 배치작업ID = " + input.getBtchJobId());
		}

		return Response.ok(output);
    }

}

@Slf4j
@RequiredArgsConstructor
@Configuration
public class BATCHTESTJob extends AbstractBaseJobBatch {

	// JOB 명
	private final static String BATCH_NAME = "BATCHTESTJOB";

    // 배치실행내역 공통 Service
    @Autowired
    private BtchExeTxnService btchExeTxnService;

    /**
     * Job 등록
     */
    @Bean(BATCH_NAME)
    public Job job(JobRepository jobRepository, @Qualifier(BATCH_NAME + "Step") Step step) {
		return new JobBuilder(getBatchName(), jobRepository)
				.start(step)
        		.listener(baseJobListener())
                .build();
    }

    /**
     * Step 등록
     */
    @Bean(BATCH_NAME + "Step")
    @JobScope
    public Step step(JobRepository jobRepository, PlatformTransactionManager transactionManager) {
		return new StepBuilder(getBatchName() + "_Step", jobRepository)
				.tasklet(this::execute, transactionManager)
                .build();
    }

	/**
	 * JOB 명
	 */
	@Override
	public String getBatchName() {
		return BATCH_NAME;
	}

    /**
     * 배치 처리 업무
     */
	@Override
	public RepeatStatus execute(StepContribution stepContribution, ChunkContext chunkContext) throws Exception {

        log.info("[job][{}]", getBatchName());
    	BtchExeTxnQryDTO btchExeTxnQryDTO  = BtchExeTxnQryDTO.builder().build();
    	String baseDt = null;

    	String jobParamJobId = stepContribution.getStepExecution().getJobParameters().getString("btchJobId", StringUtils.EMPTY);
    	String jobParamUsrNo = stepContribution.getStepExecution().getJobParameters().getString("usrNo", StringUtils.EMPTY);
    	String requestUserId = stepContribution.getStepExecution().getJobParameters().getString("requestUserId", StringUtils.EMPTY);

    	try {
        	// 배치로그(시작)
        	btchExeTxnQryDTO  = btchExeTxnService.startBtchExeTxn(BtchExeTxnQryDTO.builder().btchJobId(jobParamJobId).userId(requestUserId).build());

        	/*
        	 * 배치 로직 작성
        	 * - Service 호출 로직
        	 */
        	BatchResultDTO batchRequestDTO = BatchResultDTO.builder().jobBaseDt(baseDt).usrNo(jobParamUsrNo).pgmName(jobParamJobId).build();
        	BatchResultDTO batchResultDTO = BatchResultDTO.builder().build();

//        	batchResultDTO = bCMOPService.BCMOP100(batchRequestDTO);

        	if (log.isDebugEnabled() ) {
        		log.debug("#############################");
        		log.debug("### 배치 실행 결과 = {}", batchResultDTO);
        	}


        	//배치로그(종료)
    		btchExeTxnQryDTO.setTotalCnt(batchResultDTO.getTotalCnt());
    		btchExeTxnQryDTO.setSuccCnt(batchResultDTO.getSuccCnt());
    		btchExeTxnQryDTO.setFailCnt(batchResultDTO.getFailCnt());
    		btchExeTxnQryDTO.setResultMesg(batchResultDTO.getResultMesg());
    		btchExeTxnQryDTO.setJobBaseDt(baseDt);
    		btchExeTxnQryDTO.setSuccess(true);
        	btchExeTxnService.endBtchExeTxn(btchExeTxnQryDTO);

		} catch (CriticalException e) {
			btchExeTxnQryDTO.setSuccess(false);
			btchExeTxnQryDTO.setResultMesg(e.toString());
			btchExeTxnService.endBtchExeTxn(btchExeTxnQryDTO);

			log.error("CriticalException[job] {}, errorCode : {}", BATCH_NAME, e.getErrorCode(), e);
			throw e;
		} catch (Exception e) {
			btchExeTxnQryDTO.setSuccess(false);
			btchExeTxnQryDTO.setResultMesg(e.toString());
			btchExeTxnService.endBtchExeTxn(btchExeTxnQryDTO);

        	log.error("Exception[job] {}", BATCH_NAME, e);
        	throw new CriticalException("BIZ.BATCH.05").setOrgErrorType("BATCH").setDebuggingMessage("JOB:"+ jobParamJobId+"|기준일:" + baseDt + " 실행오류 발생했습니다");
        }

		return RepeatStatus.FINISHED;
    }

}

/**

@Slf4j
@Component
public class JobScheduler {

	@Autowired
	@Resource(name = "BizDayService")
	private BizDayService bizDayService;

	@Autowired
	@Resource(name = "BatchService")
	private BatchService batchService;

	@Autowired
	@Resource(name = "BtchExeTxnService")
	private BtchExeTxnService btchExeTxnService;

    private Gson gson;

    @PostConstruct
    public void init() {
        //init gson
        gson = new GsonBuilder()
                .setPrettyPrinting()
                .setDateFormat("yyyy'-'MM'-'dd'T'HH':'mm':'ss'.'SSS'Z'")
                .setFieldNamingStrategy(f -> CaseFormat.LOWER_CAMEL.to(CaseFormat.UPPER_UNDERSCORE, f.getName()))
                .create();
    }

	/**
	 * 배치 스케쥴러 설정
	 * - 매 1분마다 0초에 실행
	 */
	@SuppressWarnings("unchecked")
	@Scheduled(cron = "0 * * * * *")
	public void runJob() {

		/*
		 * 로컬 환경일 경우
		 */
        if (ProfileUtil.isLocal()) {
			return;
        }

		try {
			// AsyncThread 호출을 위한 결과대기 List
			List<CompletableFuture<Response<?>>> threadResList = new ArrayList<>();

			// 영업일 조회
			String currentDate = DateUtil.getCurrentDate(); // 현재 일
			boolean isBizDay = bizDayService.getBizDay(currentDate, 0).equals(currentDate);

			// 매주 첫영업일 일자 조회
			String weekFirstBizday = "";
			int dayOfWeek = DateUtil.getCalendar(currentDate).get(Calendar.DAY_OF_WEEK);
			if (dayOfWeek < Calendar.MONDAY) {
				weekFirstBizday = bizDayService.getBizDay(currentDate, 0);
			} else {
				weekFirstBizday = bizDayService.getBizDay(DateUtil.minusDays(currentDate, dayOfWeek - Calendar.MONDAY), 0);
			}

			// 매월 쳣영업일 일자 조회
			String monthFirstBizday = bizDayService.getBizDay(currentDate.substring(0,6).concat("01"), 0);

			// 매년 쳣영업일 일자 조회
			String yearFirstBizday = bizDayService.getBizDay(currentDate.substring(0,4).concat("0101"), 0);


			// 배치 목록 조회
			List<BtchBasDTO> jobList = batchService.getListBtchBas();

			for (BtchBasDTO dto : jobList) {

				if (!"Y".equals(dto.getAutoExeYn())) {
					// 자동실행여부 체크
					continue;
				}

				// 시간 체크
				if (StringUtils.isNotBlank(dto.getBtchJobTtm()) && StringUtils.isNumeric(dto.getBtchJobTtm())) {
					int currentTime = Integer.parseInt(DateUtil.getTime("HHmm"));
					String btchJobTtm = StringUtils.rightPad(dto.getBtchJobTtm(), 4, '0');
					if (currentTime < Integer.parseInt(btchJobTtm)) {
						continue;
					}
				} else {
					continue;
				}

	            if (CodeConstant.BTCH_CYCL_CD.매영업일.getCode().equals(dto.getBtchCyclCd())) {
					log.debug("### [배치작업ID={}] [btchJobNm={}] [배치실행프로그램명={}] [배치주기코드=매영업일] [배치작업시간={}]", dto.getBtchJobId(), dto.getBtchJobNm(), dto.getBtchExePgmNm(), dto.getBtchJobTtm());
					// 영업일 체크
					if (!isBizDay) {
						continue;
					}

				} else if (CodeConstant.BTCH_CYCL_CD.매일.getCode().equals(dto.getBtchCyclCd())) {
					log.debug("### [배치작업ID={}] [btchJobNm={}] [배치실행프로그램명={}] [배치주기코드=매일] [배치작업시간={}]", dto.getBtchJobId(), dto.getBtchJobNm(), dto.getBtchExePgmNm(), dto.getBtchJobTtm());

				} else if (CodeConstant.BTCH_CYCL_CD.매주.getCode().equals(dto.getBtchCyclCd())) {
					log.debug("### [배치작업ID={}] [btchJobNm={}] [배치실행프로그램명={}] [배치주기코드=매주] [배치작업시간={}]", dto.getBtchJobId(), dto.getBtchJobNm(), dto.getBtchExePgmNm(), dto.getBtchJobTtm());
					// 매주 첫 영업일인지 체크
					if (!weekFirstBizday.equals(currentDate)) {
						continue;
					}

				} else if (CodeConstant.BTCH_CYCL_CD.월초.getCode().equals(dto.getBtchCyclCd())) {
					log.debug("### [배치작업ID={}] [btchJobNm={}] [배치실행프로그램명={}] [배치주기코드=월초] [배치작업시간={}]", dto.getBtchJobId(), dto.getBtchJobNm(), dto.getBtchExePgmNm(), dto.getBtchJobTtm());
					// 월초인지 체크
					if (!monthFirstBizday.equals(currentDate)) {
						continue;
					}

				} else if (CodeConstant.BTCH_CYCL_CD.월말.getCode().equals(dto.getBtchCyclCd())) {
					log.debug("### [배치작업ID={}] [btchJobNm={}] [배치실행프로그램명={}] [배치주기코드=월말] [배치작업시간={}]", dto.getBtchJobId(), dto.getBtchJobNm(), dto.getBtchExePgmNm(), dto.getBtchJobTtm());
					// 월말인지 체크
					if (DateUtil.getCurrentLastDayByMonth() != DateUtil.getCurrentDay()) {
						continue;
					}

				} else if (CodeConstant.BTCH_CYCL_CD.월중지정일.getCode().equals(dto.getBtchCyclCd())) {
					log.debug("### [배치작업ID={}] [btchJobNm={}] [배치실행프로그램명={}] [배치주기코드=월중지정일] [배치실행지정일={}] [배치작업시간={}]", dto.getBtchJobId(), dto.getBtchJobNm(), dto.getBtchExePgmNm(), dto.getBtchExeDesigDd(), dto.getBtchJobTtm());
					// 월중(지정일)인지 체크
					if (StringUtils.isBlank(dto.getBtchExeDesigDd()) || !StringUtils.isNumeric(dto.getBtchExeDesigDd()) || DateUtil.getCurrentDay() != NumberUtils.toInt(dto.getBtchExeDesigDd())) {
						continue;
					}

				} else if (CodeConstant.BTCH_CYCL_CD.연초.getCode().equals(dto.getBtchCyclCd())) {
					log.debug("### [배치작업ID={}] [btchJobNm={}] [배치실행프로그램명={}] [배치주기코드=연초] [배치작업시간={}]", dto.getBtchJobId(), dto.getBtchJobNm(), dto.getBtchExePgmNm(), dto.getBtchJobTtm());
					// 연초인지 체크
					if (!yearFirstBizday.equals(currentDate)) {
						continue;
					}

				} else {
					log.debug("### [배치작업ID={}] [btchJobNm={}] [배치실행프로그램명={}] [배치주기코드=Unknown] [배치작업시간={}]", dto.getBtchJobId(), dto.getBtchJobNm(), dto.getBtchExePgmNm(), dto.getBtchJobTtm());
					continue;
				}

	            // 배치가 이미 실행되었는지 여부 판별
	            if (isExecutedJob(dto.getBtchJobId(), StringUtils.rightPad(currentDate + dto.getBtchJobTtm(), 14, '0'))) {
	            	continue;
	            }

	            /*
				 * 배치 실행
				 */
	            JobParametersBuilder jobParametersBuilder = new JobParametersBuilder();
	            //jobParametersBuilder.addLong("timestamp", new Date().getTime());
	            jobParametersBuilder.addString("executeTime", ("Y".equals(dto.getReExePosblYn()) ? DateUtil.getCurrentDateTime() : DateUtil.getCurrentDate()));

	            try {
		            Map<String, Object> btchExePrmtMap = gson.fromJson(dto.getBtchExePrmt(), Map.class);
		            if (btchExePrmtMap != null) {
		            	for (Map.Entry<String, Object> entry : btchExePrmtMap.entrySet()) {
		            		jobParametersBuilder.addString(entry.getKey(), String.valueOf(entry.getValue()));
		            	}
		            }
	            } catch (JsonSyntaxException e) {
	            	log.error("### 배치실행파라미터 파싱 오류 (btchExePrmt = {})", dto.getBtchExePrmt());
	            }

	            threadResList.add( asyncExecuteJobLauncher(dto, jobParametersBuilder.toJobParameters()) );
			}
			
			if (!threadResList.isEmpty()) {
				// 전문 호출결과 수신
		    	CompletableFuture.allOf(threadResList.toArray(new CompletableFuture<?>[0])).join();

		    	for (CompletableFuture<Response<?>> res : threadResList) {
		    		log.debug("### 배치실행 결과 = {}", res.get());
		    	}
			}

		} catch (Exception e) {
			log.error("[JobScheduler]][runJob][Unknown Error]", e);
		}
	}

	/**
	 * 배치 실행 여부
	 * @param btchJobId 배치작업ID
	 * @param btchStartDtm 배치시작일시(yyyyMMddHHmmss)
	 */
	private boolean isExecutedJob(String btchJobId, String btchStartDtm) throws Exception {
		BtchExeTxnDTO req = BtchExeTxnDTO.newBuilder()
				.btchJobId(btchJobId)
				.btchStartDtm(btchStartDtm)
				.build();
		List<BtchExeTxnDTO> btchExeTxnList = btchExeTxnService.listBtchExeTxn(req);

		return (btchExeTxnList != null && !btchExeTxnList.isEmpty());
	}

	/**
	 * Async Thread로 배치 실행
	 */
	@Async
    public CompletableFuture<Response<?>> asyncExecuteJobLauncher(BtchBasDTO input, JobParameters jobParameters) throws Exception {
		Response<?> res;
		try {
			res = batchService.executeJobLauncher(input, jobParameters);
		} catch (UserException e) {
			res = Response.builder().statusAndMessage(e.getErrorCode(), AppMessageUtil.getMessage(e.getErrorCode())).result(e.getDebuggingMessage()).build();
		}
		return CompletableFuture.completedFuture(res);
    }

}

        FROM (

                AA.*
            FROM (
    </sql>
