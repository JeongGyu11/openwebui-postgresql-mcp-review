		/*AdmScrenTranBasDAO.listAdmScrenTranBasByHierarchy 어드민화면거래기본 계층별 목록조회 */
		SELECT SCREN_TRAN_ID	/* 화면거래아이디 */
			 , SCREN_TRAN_DV_CD	/* 화면거래구분코드 */
			 , ADM_BZWK_LCLS_CD	/* 어드민업무대분류코드 */
			 , ADM_BZWK_MCLS_CD	/* 어드민업무중분류코드 */
			 , UP_SCREN_TRAN_ID	/* 상위화면거래아이디 */
			 , SCREN_TRAN_NM	/* 화면거래명 */
			 , URL_ADDR	/* URL주소 */
			 , SCREN_TRAN_LVEL	/* 화면거래레벨 */
			 , MENU_REPST_YN	/* 메뉴표시여부 */
			 , WHOL_ALLOW_YN	/* 전체허용여부 */
			 , SSTV_INFO_YN	/* 민감정보여부 */
			 , REPST_ORDR	/* 표시순서 */
			 , DESC_CTNT	/* 설명내용 */
			 , USE_YN	/* 사용여부 */
		FROM (
				 WITH RECURSIVE CTR AS
									(
										SELECT A.*,
											   CONVERT( LPAD(CONVERT(A.REPST_ORDR,CHAR(10)), 10, '0'), CHAR(100)) AS HIERARCHY,
											   1 AS LEVEL
										FROM ADM_SCREN_TRAN_BAS A
										WHERE A.USE_YN = 'Y'
										  AND A.SCREN_TRAN_LVEL = 0
										  AND A.SCREN_TRAN_DV_CD IN ('00','10') /* 화면거래구분코드(00=메뉴그룹, 10=화면일반) */
										UNION ALL
										SELECT B.*,
											   CONVERT( CONCAT(C.HIERARCHY, '.', LPAD(CONVERT(B.REPST_ORDR,CHAR(10)), 10, '0')), CHAR(100)) AS HIERARCHY,
											   C.LEVEL + 1 AS LEVEL
										FROM ADM_SCREN_TRAN_BAS B
												 INNER JOIN CTR C ON B.UP_SCREN_TRAN_ID = C.SCREN_TRAN_ID
										WHERE B.USE_YN = 'Y'
										  AND B.SCREN_TRAN_DV_CD IN ('00','10') /* 화면거래구분코드(00=메뉴그룹, 10=화면일반) */
									)
				 SELECT *
				 FROM CTR
			 ) T
		WHERE 1=1
		  AND SCREN_TRAN_DV_CD IN ('00','10') /* 화면거래구분코드(00=메뉴그룹, 10=화면일반) */
		ORDER BY HIERARCHY ASC
	</select>

				WHERE B.USE_YN = 'Y'

* Description : 페이징을 위한 쿼리
* - MariaDB에서는 ROW_NUMBER() OVER()를 사용할 수 없어서 변수를 사용하여 페이징 처리
* - MySQL 8.0.2 이상부터 ROW_NUMBER() OVER()를 사용할 수 있음
* - PostgreSQL에서는 ROW_NUMBER() OVER()를 사용할 수 있음
-->
<mapper namespace="COMMON">

    <sql id="pagingPreSQL">
        SELECT *
        FROM (
            SELECT
                @rownum := @rownum + 1 AS ROW_NUM,
                #{pageNo} AS PAGE_NO,
                #{pageSize} AS PAGE_SIZE,
                AA.*
            FROM (
    </sql>

    <sql id="pagingPostSQL">
                LIMIT 18446744073709551615
            ) AA
            CROSS JOIN (SELECT @rownum := 0) BB
        ) SUB
        WHERE ROW_NUM BETWEEN ((#{pageNo} - 1) * #{pageSize} + 1) AND (#{pageNo} * #{pageSize})
    </sql>
