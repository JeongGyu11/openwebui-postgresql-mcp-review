        if (this.methodType == HttpClientConstant.MethodType.GET) {
            return getRequest(apiUrl).asObjectAsync(genericType);

    /**
     * 지정된 URL과 구성으로 GET 요청을 만듭니다.
     *
     * @param apiUrl GET 요청의 URL
     * @return 구성된 GetRequest
     */
    private GetRequest getRequest(String apiUrl) {
        GetRequest httpRequest = primaryInstance.get(apiUrl);
        if (StringUtils.isNotBlank(this.proxyHost)) {
            httpRequest.proxy(this.proxyHost, this.proxyPort);
        }
        httpRequest.socketTimeout(this.readTimeout);
        httpRequest.connectTimeout(this.connTimeout);
        if (this.cookie != null) {
            httpRequest.cookie(this.cookie);
        }

        HttpClientDefaultConfig.setDefaultHeader(this.businessTierHeaderInfoProvider, httpRequest);
        httpRequest.headers(this.headers);
        httpRequest.queryString(this.fields);

        return httpRequest;
    }

    /**
     * 지정된 URL 및 구성에 대한 필드가 있는 요청을 만듭니다.
     *
     * @param apiUrl 요청의 URL
     * @return 구성된 MultipartBody
     */
    private MultipartBody requestFields(String apiUrl) {
        HttpRequestWithBody httpRequestWithBody = primaryInstance.request(this.methodType.name(), apiUrl);
        if (StringUtils.isNotBlank(this.proxyHost)) {
            httpRequestWithBody.proxy(this.proxyHost, this.proxyPort);
        }
        httpRequestWithBody.socketTimeout(this.readTimeout);
        httpRequestWithBody.connectTimeout(this.connTimeout);
        if (this.contentType != null) {
            httpRequestWithBody.contentType(this.contentType);
        }
        if (this.cookie != null) {
            httpRequestWithBody.cookie(this.cookie);
        }

        HttpClientDefaultConfig.setDefaultHeader(this.businessTierHeaderInfoProvider, httpRequestWithBody);
        httpRequestWithBody.headers(this.headers);

        return httpRequestWithBody.fields(this.fields);
    }

    /**
     * 지정된 URL과 구성에 대한 본문이 있는 요청을 만듭니다.
     *
     * @param apiUrl 요청의 URL
     * @return 구성된 RequestBodyEntity
     */
    private RequestBodyEntity requestBody(String apiUrl) {
        HttpRequestWithBody httpRequestWithBody = primaryInstance.request(this.methodType.name(), apiUrl);
        if (StringUtils.isNotBlank(this.proxyHost)) {
            httpRequestWithBody.proxy(this.proxyHost, this.proxyPort);
        }
        httpRequestWithBody.socketTimeout(this.readTimeout);
        httpRequestWithBody.connectTimeout(this.connTimeout);
        if (this.contentType != null) {
            httpRequestWithBody.contentType(this.contentType);
        }
        if (this.cookie != null) {
            httpRequestWithBody.cookie(this.cookie);
        }

        HttpClientDefaultConfig.setDefaultHeader(this.businessTierHeaderInfoProvider, httpRequestWithBody);
        httpRequestWithBody.headers(this.headers);

        String bodyJson = "";
        try {
            if (this.body != null) {
                bodyJson = JsonUtil.ObjectToJson(this.body);
            }
        } catch (JsonProcessingException e) {
            log.error(e.getMessage(), e);
        }
        return httpRequestWithBody.body(bodyJson);
    }

}

public interface HttpClientResponse<T> extends HttpResponse<T> {

            @Override

            }

            @Override

                return httpResponse.mapBody(function);
            }

            @Override
            public <V> HttpResponse<V> map(Function<T, V> function) {
                return httpResponse.map(function);
            }

            @Override
            public HttpResponse<T> ifSuccess(Consumer<HttpResponse<T>> consumer) {
                return httpResponse.ifSuccess(consumer);
            }

            @Override
            public HttpResponse<T> ifFailure(Consumer<HttpResponse<T>> consumer) {
                return httpResponse.ifFailure(consumer);
            }

            @Override
            public <E> HttpResponse<T> ifFailure(Class<? extends E> aClass, Consumer<HttpResponse<E>> consumer) {
                return httpResponse.ifFailure(aClass, consumer);
            }

            @Override
            public boolean isSuccess() {
                return httpResponse.isSuccess();
            }

            @Override
            public <E> E mapError(Class<? extends E> aClass) {
                return httpResponse.mapError(aClass);
            }

            @Override
            public Cookies getCookies() {
                return httpResponse.getCookies();
            }

            @Override
            public HttpRequestSummary getRequestSummary() {
                return null;
            }
        };
    }

    default T get() {
        return this.getBody();
    }
    default T orElseThrowGet() throws HttpClientApiException {
        if (this.isSuccess()) {
            return this.getBody();
        }
        Response<?> errorResponse = this.getErrorResponse();
        throw new HttpClientApiException(errorResponse, this.getStatus());
    }
    default <X extends Throwable> T orElseThrowGet(Supplier<? extends X> exceptionSupplier) throws X {
        if (this.isSuccess()) {
            return this.getBody();
        }
        throw exceptionSupplier.get();
    }

    default Response<?> getErrorResponse() {
        if (!this.isSuccess()) {
            Optional<UnirestParsingException> parsingError = this.getParsingError();
            Response<?> errorResponse = null;
            if (parsingError.isPresent()) {
                final String body = parsingError.get().getOriginalBody();
                errorResponse = new GsonBuilder().create().fromJson(body, Response.class);
            }
            return errorResponse;
        }
        return Response.SUCCESS;
    }

}

                        .path(this.url)

                        .build())
                .contentType(this.contentType != null ? MediaType.valueOf(this.contentType) : MediaType.APPLICATION_JSON)

 * 설명 :
 * 예제 :
 * <pre>
 *    Response<?> response = webClientAdaptor.businessTier(HttpClientConstant.Msa.BATCH)
 *        .post("/system/jobLauncher.json")
 *        .body(Map.of("btchJobId", "BATCHTESTJOB"))

@Slf4j
@Component
public class WebClientAdaptor {
    private final Environment env;
    private final BusinessTierHeaderInfoProvider businessTierHeaderInfoProvider;
    private final HttpClient httpClient;

    private WebClientAdaptor(Environment env, BusinessTierHeaderInfoProvider businessTierHeaderInfoProvider) {
        this.env = env;
        this.businessTierHeaderInfoProvider = businessTierHeaderInfoProvider;
        int readTimeout = env.getProperty("http.client.config.read-timeout", Integer.class, 100_000);
        int connTimeout = env.getProperty("http.client.config.connect-timeout", Integer.class, 3000);
        String proxyHost = env.getProperty("http.client.config.proxy-host");
        int proxyPort = env.getProperty("http.client.config.proxy-port", Integer.class, -1);
        Boolean verifySsl = env.getProperty("http.client.config.verify-ssl", Boolean.class, Boolean.TRUE);

        // HttpClient 설정
        this.httpClient = HttpClient.create()
                .option(ChannelOption.CONNECT_TIMEOUT_MILLIS, connTimeout)
                .responseTimeout(Duration.ofMillis(readTimeout))
                .doOnConnected(conn -> conn
                        .addHandlerLast(new ReadTimeoutHandler(readTimeout, TimeUnit.MILLISECONDS))
                        .addHandlerLast(new WriteTimeoutHandler(readTimeout, TimeUnit.MILLISECONDS)));

        // Proxy 설정
        if (StringUtils.isNotBlank(proxyHost)) {
            httpClient.proxy(ops -> ops.type(ProxyProvider.Proxy.HTTP).host(proxyHost).port(proxyPort));
        }

        // SSL 인증서 무시 설정
        if (!verifySsl) {
            try {
                // TrustManagerFactory를 기본값으로 설정 (모든 인증서를 신뢰)
                TrustManagerFactory trustManagerFactory = TrustManagerFactory.getInstance(TrustManagerFactory.getDefaultAlgorithm());
                trustManagerFactory.init((KeyStore) null);

                // SslContextBuilder를 사용하여 SslContext 생성
                SslContext sslContext = SslContextBuilder.forClient()
                        .trustManager(trustManagerFactory) // 신뢰 관리자 설정
                        .build();

                // SslProvider를 사용하여 SSL 설정 구성
                SslProvider sslProvider = SslProvider.builder()
                        .sslContext(sslContext)
                        .build();

                httpClient.secure(sslContextSpec -> sslContextSpec.sslContext(sslProvider.getSslContext()));
            } catch (NoSuchAlgorithmException | KeyStoreException | SSLException e) {
                log.error("SSL 인증서 무시 설정 중 오류가 발생했습니다.", e);
            }
        }
    }

    /**
     * Business Tier 호출
     */
    public WebClientBuilder businessTier(HttpClientConstant.Msa msa) {
        WebClient webClient = WebClient.builder()
                .clientConnector(new ReactorClientHttpConnector(this.httpClient))
                .defaultHeaders(httpHeaders -> {
                    HttpClientDefaultConfig.setDefaultHeader(this.businessTierHeaderInfoProvider, httpHeaders);
                })
                .baseUrl(Objects.requireNonNull(this.env.getProperty("http.client.business-tier." + msa.name().toLowerCase())))
                .build();

        return new WebClientBuilder(webClient);
    }

    /**
     * 외부 URL 호출
     */
    public WebClientBuilder external(String externalUrl) {
        WebClient webClient = WebClient.builder()
                .clientConnector(new ReactorClientHttpConnector(this.httpClient))
                .defaultHeaders(httpHeaders -> {
                    HttpClientDefaultConfig.setDefaultHeader(this.businessTierHeaderInfoProvider, httpHeaders);
                })
                .baseUrl(Objects.requireNonNull(this.env.getProperty("http.client.external-url." + externalUrl)))
                .build();

        return new WebClientBuilder(webClient);
    }

}

/**
 * 시스템명 : 
 * 업무구분명 : 공통 > Util > CryptoJS를 이용한 암복호화 유틸
 * 프로그램명 : CryptoJS를 이용한 암복호화 유틸

 * 2022. 10. 13.         정영탁          최초생성

@UtilityClass
public class CryptoJsUtil {

    /**
     * AES-256을 이용하여 암호화
     * @param plaintext 암호화할 문자열
     * @param passphrase 암복호화 비밀번호
     * @return 암호화된 문자열
     */
    public static String encryptAES(String plaintext, String passphrase) throws NoSuchAlgorithmException, NoSuchPaddingException, InvalidAlgorithmParameterException, InvalidKeyException, IllegalBlockSizeException, BadPaddingException {
    	if (StringUtils.isBlank(plaintext) || StringUtils.isBlank(passphrase)) {
            return "";
        }

        final int keySize = 256;
        final int ivSize = 128;

        // 텍스트를 Base64 형식으로 디코딩
        byte[] plaintextBytes = plaintext.getBytes(StandardCharsets.UTF_8);

        // Salt를 구한다. (Salted__ 시작되는 문자열)
        byte[] stBytes = "Salted__".concat(DateUtil.getCurrentDate(DateUtil.FORMATTER_HH_MM_SS)).getBytes(StandardCharsets.UTF_8);
        byte[] saltBytes = Arrays.copyOfRange(stBytes, 8, 16);

        // 비밀번호와 Salt에서 Key와 IV값을 가져옴.
        byte[] key = new byte[keySize / 8];
        byte[] iv = new byte[ivSize / 8];
        EvpKDF(passphrase.getBytes(StandardCharsets.UTF_8), keySize, ivSize, saltBytes, key, iv);

        // 암호화
        Cipher cipher = Cipher.getInstance("AES/CBC/PKCS5Padding");
        cipher.init(Cipher.ENCRYPT_MODE, new SecretKeySpec(key, "AES"), new IvParameterSpec(iv));
        byte[] receovedCiphertextBytes = cipher.doFinal(plaintextBytes);

        // 암호화된 텍스트 (Salt값과 암호화된 텍스트를 합한다.)
        byte[] ciphertextBytes = new byte[stBytes.length + receovedCiphertextBytes.length];
        System.arraycopy(stBytes, 0, ciphertextBytes, 0, stBytes.length);
        System.arraycopy(receovedCiphertextBytes, 0, ciphertextBytes, stBytes.length, receovedCiphertextBytes.length);

        // 암호화 문자열을 Base64 형식으로 인코딩
        return Base64.encodeBase64String(ciphertextBytes);
    }

    /**
     * AES-256을 이용하여 복호화
     * @param ciphertext 복호화할 문자열
     * @param passphrase 암복호화 비밀번호
     * @return 복호화된 문자열
     */
    public static String decryptAES(String ciphertext, String passphrase) throws NoSuchAlgorithmException, NoSuchPaddingException, InvalidAlgorithmParameterException, InvalidKeyException, IllegalBlockSizeException, BadPaddingException {
    	if (StringUtils.isBlank(ciphertext) || StringUtils.isBlank(passphrase)) {
            return "";
        }

        final int keySize = 256;
        final int ivSize = 128;

        // 텍스트를 Base64 형식으로 디코딩
        byte[] ctBytes = Base64.decodeBase64(ciphertext.getBytes(StandardCharsets.UTF_8));

        // Salt를 구한다. (생략된 8비트는 Salted__ 시작되는 문자열)
        byte[] saltBytes = Arrays.copyOfRange(ctBytes, 8, 16);

        // 암호화된 텍스트 (Salt값 이후가 암호화된 텍스트)
        byte[] ciphertextBytes = Arrays.copyOfRange(ctBytes, 16, ctBytes.length);

        // 비밀번호와 Salt에서 Key와 IV값을 가져옴.
        byte[] key = new byte[keySize / 8];
        byte[] iv = new byte[ivSize / 8];
        EvpKDF(passphrase.getBytes(StandardCharsets.UTF_8), keySize, ivSize, saltBytes, key, iv);

        // 복호화
        Cipher cipher = Cipher.getInstance("AES/CBC/PKCS5Padding");
        cipher.init(Cipher.DECRYPT_MODE, new SecretKeySpec(key, "AES"), new IvParameterSpec(iv));
        byte[] recoveredPlaintextBytes = cipher.doFinal(ciphertextBytes);

        return new String(recoveredPlaintextBytes, StandardCharsets.UTF_8);
    }

    /**
     * OpenSSL KDF
     */
    private static byte[] EvpKDF(byte[] password, int keySize, int ivSize, byte[] salt, byte[] resultKey, byte[] resultIv) throws NoSuchAlgorithmException {
    	return EvpKDF(password, keySize, ivSize, salt, 1, "MD5", resultKey, resultIv);
    }
    /**
     * OpenSSL KDF
     */
    private static byte[] EvpKDF(byte[] password, int keySize, int ivSize, byte[] salt, int iterations, String hashAlgorithm, byte[] resultKey, byte[] resultIv) throws NoSuchAlgorithmException {
    	keySize = keySize / 32;
    	ivSize = ivSize / 32;
    	int targetKeySize = keySize + ivSize;
    	byte[] derivedBytes = new byte[targetKeySize * 4];
    	int numberOfDerivedWords = 0;
    	byte[] block = null;
    	MessageDigest hasher = MessageDigest.getInstance(hashAlgorithm);
    	while (numberOfDerivedWords < targetKeySize) {
    		if (block != null) {
    			hasher.update(block);
    		}
    		hasher.update(password);
    		// Salting
    		block = hasher.digest(salt);
    		hasher.reset();
    		// Iterations : 키 스트레칭(key stretching)
    		for (int i = 1; i < iterations; i++) {
    			block = hasher.digest(block);
    			hasher.reset();
    		}
    		System.arraycopy(block, 0, derivedBytes, numberOfDerivedWords * 4, Math.min(block.length, (targetKeySize - numberOfDerivedWords) * 4));
    		numberOfDerivedWords += block.length / 4;
    	}
    	System.arraycopy(derivedBytes, 0, resultKey, 0, keySize * 4);
    	System.arraycopy(derivedBytes, keySize * 4, resultIv, 0, ivSize * 4);
    	return derivedBytes; // key + iv
    }

        dt = dt.plusDays(day);
        return fmt.print(dt);

    /**
     * yyyyMMdd를 yyyy, mm, dd로 분리해서 String[3]로 제공한다. String[0]는 년, String[1]은 월,
     * String[2] 일.
     */
    public static String[] splitDateStr(String yyyyMMdd) {
        String[] strDate = new String[3];
        strDate[0] = yyyyMMdd.substring(0, 4);
        strDate[1] = yyyyMMdd.substring(4, 6);
        strDate[2] = yyyyMMdd.substring(6);
        return strDate;
    }

    /**
     * yyyyMMdd를 yyyy, mm, dd로 분리해서 int[3]로 제공한다. int[0]는 년, int[1]은 월, int[2]
     * 일.
     */
    public static int[] splitDateInt(String yyyyMMdd) {
        String[] strDate = splitDateStr(yyyyMMdd);
        int[] intDate = new int[3];
        intDate[0] = Integer.parseInt(strDate[0]);
        intDate[1] = Integer.parseInt(strDate[1]);
        intDate[2] = Integer.parseInt(strDate[2]);
        return intDate;
    }

    /**
     * 'yyyyMMdd' 포맷으로 문자열을 제공한다.
     */
    public static String formatYMD8(int year, int month, int day) {
        StringBuilder sb = new StringBuilder();
        sb.append(year);
        if (month < 10)
            sb.append("0");
        sb.append(month);
        if (day < 10)
            sb.append("0");
        sb.append(day);
        return sb.toString();
    }

    /**
     * yyyyMMdd의 전일을 제공한다.
     */
    public static String previousDate(String yyyyMMdd) {
        int[] intDate = splitDateInt(yyyyMMdd);
        int day = intDate[2];
        int month = intDate[1];
        int year = intDate[0];

        if (day > 1) {
            return formatYMD8(year, month, day - 1);
        } else {
            return handleEndOfMonth(year, month);
        }
    }

    private static String handleEndOfMonth(int year, int month) {
        if (month > 1) {
            month -= 1;
        } else {
            year -= 1;
            month = 12;
        }
        Calendar cal = DateUtil.getCalendar(year, month, 1);
        int lastDayOfPreviousMonth = cal.getActualMaximum(Calendar.DAY_OF_MONTH);
        return formatYMD8(cal.get(Calendar.YEAR), cal.get(Calendar.MONTH) + 1, lastDayOfPreviousMonth);
    }

    /**
     * String 과 infmt, outfmt 을 받아 날짜를 출력해준다.
     * <p>
     * ex) getFormatDate("20080901","yyyyMMdd","yyyy-MM-dd") -> 2008-09-01
     * </p>
     */
    public static String getFormatDate(String date, String infmt, String outfmt) {
        try {
            // date 형식이 infmt 에 맞지 않다면 date 를 infmt 에 맞추어줌.
            StringBuffer sDate = new StringBuffer(date);
            int len = date.length();

            if (StringUtils.equals(infmt, "yyyyMMddHHmm") && len < 12) {
                for (int i = len; i < 12; i++) {
                    sDate.append("0");
                }
            } else if (StringUtils.equals(infmt, "yyyyMMddHHmmss") && len < 14) {
                for (int i = len; i < 14; i++) {
                    sDate.append("0");
                }
            } else if (StringUtils.equals(infmt, "yyyyMMdd") && len > 8) {
            	sDate.setLength(8);
            }

            SimpleDateFormat sdf = new SimpleDateFormat(infmt, Locale.US);
            Date d = sdf.parse(sDate.toString());
            sdf.applyPattern(outfmt);

            return sdf.format(d);
        } catch (ParseException e) {
            return "";
        }
    }

    /**
     * 현재 시간을 포맷에 맞추어 내용을 수정
     * @param pattern 날자 포맷 정의, String 형식으로 출력
     * @return 포맷팅된 문자열 출력
     */
	public static String getCurrentDateTime(@NonNull String pattern) {
		String newPattern = (StringUtils.isBlank(pattern)) ? FORMATTER_YYYY_MM_DD_HH_MM_SS : pattern;
		java.time.format.DateTimeFormatter formatter = java.time.format.DateTimeFormatter.ofPattern(newPattern);
		return LocalDateTime.now().format(formatter);
	}

    /**
     * 현재 일자 (년.월.일) 정보를 가져옴
     * @param pattern
     * @param addYear
     * @param addMonth
     * @param addDay
     * @return
     */
    public static String getCurrentDate(@NonNull String pattern, int addYear, int addMonth, int addWeek, int addDay) {
        String newPattern = StringUtils.defaultIfBlank(pattern, DateUtil.FORMATTER_YYYY_MM_DD);
        java.time.format.DateTimeFormatter formatter = java.time.format.DateTimeFormatter.ofPattern(newPattern);
        return LocalDate.now()
                .plusYears(addYear)
                .plusMonths(addMonth)
                .plusWeeks(addWeek)
                .plusDays(addDay)
                .format(formatter);
    }

    /**
     * 만 나이 계산
     * @param resno 주민등록번호
     * @return 만 나이
     */
    public static int getAmericanAge(@NonNull String resno) {
    	if (resno.length() < 13 || !StringUtils.isNumeric(resno)) {
    		return 0;
    	}

		String birthDate = extractBirthDateFromResno(resno);

    	LocalDate now = LocalDate.now();
        return calculateAgeFromBirthdate(now, birthDate);
    }

    /**
     * 만 나이 계산
     * @param resno 주민등록번호
     * @param baseDt  기준일 - isBlank 이면 오늘
     * @return 만 나이
     */
    public static int getAmericanAge(@NonNull String resno, @NonNull String baseDt) {
        if (resno.length() < 13 || !StringUtils.isNumeric(resno)) {
            return 0;
        }

        String birthDate = extractBirthDateFromResno(resno);

        if (StringUtils.isBlank(baseDt)) {
            baseDt = DateUtil.getCurrentDate(DateUtil.FORMATTER_YYYYMMDD);
        }

        Date dt = DateUtil.getDate(baseDt, DateUtil.FORMATTER_YYYYMMDD);
        LocalDate now = dt.toInstant().atZone(ZoneId.systemDefault()).toLocalDate();

        return calculateAgeFromBirthdate(now, birthDate);
    }

    private static String extractBirthDateFromResno(String resno) {
        String sexType = resno.substring(6, 7);
        String centuryPrefix;

        if ("1".equals(sexType) || "2".equals(sexType) || "5".equals(sexType) || "6".equals(sexType)) {
            centuryPrefix = "19";
        } else if ("9".equals(sexType) || "0".equals(sexType)) {
            centuryPrefix = "18";
        } else {
            centuryPrefix = "20";
        }

        return centuryPrefix.concat(resno.substring(0, 6));
    }

    private static int calculateAgeFromBirthdate(LocalDate now, String birthDate) {
        LocalDate parseBirthDate = LocalDate.parse(birthDate, java.time.format.DateTimeFormatter.ofPattern("yyyyMMdd"));

        int age = now.minusYears(parseBirthDate.getYear()).getYear();

        // 생일이 지났는지 판단하여, 나이 보정
        if (parseBirthDate.plusYears(age).isAfter(now)) {
            age = age - 1;
        }

        return age;
    }

    /**
     * String DateTime => LocalDateTime 타입으로 변환하여 가져오기
     * @param dateTime
     * @return 예) 2022-06-10T15:53:23
     * @throws ParseException 
     */
    public static String convertLocalDateTime(@NonNull String dateTime) throws ParseException {
    	SimpleDateFormat toDateFormat = new SimpleDateFormat("yyyyMMddHHmmss");
    	Date dateParse = toDateFormat.parse(dateTime);
    	
		SimpleDateFormat toSimpleDateFormat = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss");
		String dateFormat = toSimpleDateFormat.format(dateParse);
		
		java.time.format.DateTimeFormatter toDateTimeFormat = java.time.format.DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");
		return java.time.LocalDateTime.parse(dateFormat, toDateTimeFormat).toString();
    }
    
    /**
     * 현재시간 LocalDateTime 타입으로 가져오기
     * @return 예) 2022-06-10T15:53:23
     */
    public static String getNowLocalDateTime() {
		LocalDateTime result = LocalDateTime.now();
		return result.toString().substring(0, 19);
	}

/**
 * 시스템명 : 
 * 업무구분명 : 공통 > Util > DTO 간 데이터 전달을 위한
 * 설명 :
 *
 * @Class   DTOUtil.class
 * @author  정영탁

 * @version 1.0

@Slf4j
@UtilityClass
public class DTOUtil {
	private static final ModelMapper modelMapper;

	static {
		modelMapper = new ModelMapper();
		modelMapper.getConfiguration()
				.setSkipNullEnabled(true)
				.setMatchingStrategy(MatchingStrategies.STRICT);
	}

	/**
	 * srcObj의 각 필드의 값을 복사한 tgtClass의 새로운 인스턴스를 생성한다.
	 * tgtClass의 필드 명을 기준으로 srcObj에서 값을 복사한다.
	 */
	public static <I, O> O convertDTO(I srcObj, Class<O> tgtClass) {
		if (srcObj == null ) {
			return null;
		}
		return modelMapper.map(srcObj, tgtClass);
	}

	/**
	 * srcObj 리스트의 값을 복사한 tgtClass 타입의 List를 생성한다.
	 * tgtClass의 필드 명을 기준으로 srcObj에서 값을 복사한다.
	 */
	public static <I, O> List<O> convertDTOList(List<I> srcObjList, Class<O> tgtClass) {
		if (srcObjList == null ) {
			return null;
		}
		
		List<O> list = new ArrayList<>();
		for (I srcObj : srcObjList) {
			list.add(convertDTO(srcObj, tgtClass));
		}

		return list;
	}

	/**
	 * srcObj의 각 필드의 값을 tgtObj의 각 필드의 값으로 overwrite복사한 인스턴스를 생성한다.
	 * srcObj, tgtObj는 각 DTO의 생성된 인스턴스이다.
	 */
	public static <I, O> O copyDTO(I srcObj, O tgtObj) {
		if (srcObj == null ) {
			return null;
		}

		modelMapper.map(srcObj, tgtObj);

		return tgtObj;
	}

	/**
	 * 여러개의 srcObj의 각 필드의 값을 복사한 tgtClass의 새로운 인스턴스를 생성한다.
	 * tgtClass의 필드 명을 기준으로 srcObj에서 값을 복사한다.
	 */
	public static <I, O> O mergeDTO(List<I> srcObjList, Class<O> tgtClass) {
		if (srcObjList == null ) {
			return null;
		}

		O out = null;
		try {
			out = tgtClass.getDeclaredConstructor().newInstance();
			for (I srcObj : srcObjList) {
				modelMapper.map(srcObj, out);
			}
		} catch (Exception e) {
			log.error(e.getMessage());
		}

		return out;
	}

	/**
	 * srcObj의 각 필드의 값을 복사한 tgtClass의 새로운 인스턴스를 생성한다.
	 * tgtClass의 필드 명을 기준으로 srcObj에서 값을 복사한다.
	 */
	public static <I, O> O convertMap(I srcObj, Class<O> tgtClass) {
		if (srcObj == null ) {
			return null;
		}

        //init gson
        Gson gson = new GsonBuilder()
                .setFieldNamingStrategy(f -> CaseFormat.LOWER_CAMEL.to(CaseFormat.UPPER_UNDERSCORE, f.getName()))
                .create();

        O out = null;
		try {
			out = gson.fromJson(gson.toJson(srcObj), tgtClass);
		} catch (Exception e) {
			log.error(e.getMessage());
		}

		return out;
	}

}

     * @param object 데이터 객체
     * @param row 행 객체
     * @param header 헤더 정보
     * @param format 포맷 정보
     * @param cellStyle 셀 스타일
     * @throws IllegalAccessException 예외 발생 시
     * @throws InvocationTargetException 예외 발생 시
     */
    @SuppressWarnings("deprecation")
    private void createData(Sheet sheet, Object object, Row row, Map<String, String> header, Map<String, String> format, CellStyle cellStyle) throws IllegalAccessException, InvocationTargetException {
        String methodName = null;
        int cellIdx = 0;
        Object value = null;
        String cellValue = null;
        Cell cell;
        CellType cellType = CellType.STRING;

        for (String key : header.keySet()) {
            methodName = "get" + key.substring(0,1).toUpperCase() + key.substring(1);

            try {
                if (object instanceof Map) {
                    value = ((Map<?, ?>) object).get(key);
                } else {
                    value = MethodUtils.invokeMethod(object, true, methodName);
                }

                if(value == null) {
                    cellValue = "";
                } else {
                    if (ExcelConstant.EXCEL_FORMAT_MONEY.equals(format.get(key))) {
                        if (value instanceof BigDecimal t) {
                            t = t.setScale(0, RoundingMode.HALF_UP);
                            cellValue = t.toPlainString();
                        } else if (value instanceof BigInteger t) {
                            cellValue = new String(t.toByteArray());
                        } else {
                            cellValue = String.valueOf(value);
                        }
                        cellType = CellType.NUMERIC;
                    } else if(ExcelConstant.EXCEL_FORMAT_DATETIME.equals(format.get(key))) {
                        if (value instanceof Date) {
                            cellValue = DateUtil.getString((Date)value, DateUtil.FORMATTER_YYYY_MM_DD_HH_MM);
                        } else {
                            cellValue = DateUtil.getFormatDate(String.valueOf(value), DateUtil.FORMATTER_YYYYMMDDHHMMSS, DateUtil.FORMATTER_YYYY_MM_DD_HH_MM);
                        }
                    } else if(ExcelConstant.EXCEL_FORMAT_DATE.equals(format.get(key))) {
                        cellValue = FormatUtil.formatDate(String.valueOf(value), "-");
                    } else if(ExcelConstant.EXCEL_FORMAT_TIME.equals(format.get(key))) {
                        cellValue = FormatUtil.formatTime(String.valueOf(value), ":");
                    } else if(ExcelConstant.EXCEL_FORMAT_TIME_NO_SECOND.equals(format.get(key))) {
                        String v = String.valueOf(value);
                        if (v != null && v.length() < 6) v = v + "00";
                        cellValue = DateUtil.getFormatDate(v, DateUtil.FORMATTER_HHMMSS, DateUtil.FORMATTER_HH_MM);
                    } else if(ExcelConstant.EXCEL_FORMAT_INTEREST.equals(format.get(key))) {
                        cellValue = FormatUtil.formatInterest(String.valueOf(value));
                    } else if(ExcelConstant.EXCEL_FORMAT_ACNO.equals(format.get(key))) {
                        cellValue = FormatUtil.formatAcno(String.valueOf(value), false);
                    } else {
                        cellValue = HtmlUtils.htmlUnescape(String.valueOf(value));
                    }
                }
            } catch (NoSuchMethodException | IllegalAccessException | InvocationTargetException e) {
                cellValue = "";
            }

            cell = row.createCell(cellIdx);
            cell.setCellStyle(cellStyle);
            cell.setCellType(cellType);

            if (ExcelConstant.EXCEL_FORMAT_MONEY.equals(format.get(key)) && StringUtils.isNotBlank(cellValue)) {
                DataFormat df = workbook.createDataFormat();
                cellStyle.setDataFormat(df.getFormat("#,##0"));
                cell.setCellValue(new BigDecimal(cellValue).doubleValue());
            } else {
                cell.setCellValue(cellValue);
            }

            cellIdx++;
        }
    }

	/**
	 * 파일을 압축하여 반환
	 * @param inputStreamList 압축할 파일 목록
	 * @return 저장할 파일
	 */
	public static byte[] saveToZip(List<InputStream> inputStreamList, List<String> filenameList) throws Exception {
    	ByteArrayOutputStream byteArrayOutputStream = new ByteArrayOutputStream();

		if (inputStreamList == null || filenameList == null || inputStreamList.size() != filenameList.size()) {
			log.error("## [saveToZip] 입력값이 상이합니다.");
			return byteArrayOutputStream.toByteArray();
		}

		ZipOutputStream zos = new ZipOutputStream(byteArrayOutputStream);

		int length;
		ZipEntry ze;
		byte[] buf = new byte[8 * 1024];

		try {
			for (int i = 0; i < inputStreamList.size(); i++) {
				InputStream inputStream = inputStreamList.get(i);
				String filename = filenameList.get(i);

				ze = new ZipEntry(filename);
				zos.putNextEntry(ze);

				while ((length = inputStream.read(buf, 0, buf.length)) >= 0) {
					zos.write(buf, 0, length);
				}
				inputStream.close();
				zos.closeEntry();
			}
		} catch (IOException e) {
			log.error(e.getMessage());
		} catch (Exception e) {
			log.error(e.getMessage());
		} finally {
			if (zos != null)
				zos.close();
		}

		return byteArrayOutputStream.toByteArray();
	}

	public static void unzip(InputStream is, File destDir, String charsetName) throws IOException {
		ZipInputStream zis = null;
		ZipEntry entry;
		String name;
		File target;
		int nWritten = 0;
		BufferedOutputStream bos = null;
		byte[] buf = new byte[1024 * 8];

		try {
			zis = new ZipInputStream(is, Charset.forName(charsetName));
			while ((entry = zis.getNextEntry()) != null) {
				name = entry.getName();
				target = new File(destDir, name);
				if (entry.isDirectory()) {
					target.mkdirs();
				} else {
					target.createNewFile();
					try {
						bos = new BufferedOutputStream(new FileOutputStream(target));
						while ((nWritten = zis.read(buf)) >= 0) {
							bos.write(buf, 0, nWritten);
						}
					} catch (IOException e) {
						log.error(e.getMessage());
					} catch (Exception e) {
						log.error(e.getMessage());
					} finally {
						if (bos != null)
							bos.close();
					}
				}
			}
		} catch (IOException e) {
			log.error(e.getMessage());
		} catch (Exception e) {
			log.error(e.getMessage());
		} finally {
			if (zis != null)
				zis.close();
		}
	}

	/**
	 * 압축
	 */
	public static void zip(File src, OutputStream os, String charsetName, boolean includeSrc) throws IOException {
		ZipOutputStream zos = new ZipOutputStream(os, Charset.forName(charsetName));
		FileInputStream fis = null;

		int length;
		ZipEntry ze;
		byte[] buf = new byte[8 * 1024];
		String name;

		try {
			Stack<File> stack = new Stack<File>();
			File root;
			if (src.isDirectory()) {
				if (includeSrc) {
					stack.push(src);
					root = src.getParentFile();
				} else {
					File[] fs = src.listFiles();
					for (int i = 0; i < fs.length; i++) {
						stack.push(fs[i]);
					}
					root = src;
				}
			} else {
				stack.push(src);
				root = src.getParentFile();
			}

			while (!stack.isEmpty()) {
				File f = stack.pop();
				name = toPath(root, f);
				if (f.isDirectory()) {
					File[] fs = f.listFiles();
					for (int i = 0; i < fs.length; i++) {
						if (fs[i].isDirectory())
							stack.push(fs[i]);
						else
							stack.add(0, fs[i]);
					}
				} else {
					ze = new ZipEntry(name);
					zos.putNextEntry(ze);
					try {
						fis = new FileInputStream(f);
						while ((length = fis.read(buf, 0, buf.length)) >= 0) {
							zos.write(buf, 0, length);
						}
					} catch (IOException e) {
						log.error(e.getMessage());
					} catch (Exception e) {
						log.error(e.getMessage());
					}
					finally {
						if (fis != null)
							fis.close();
					}
					zos.closeEntry();
				}
			}
		} catch (IOException e) {
			log.error(e.getMessage());
		} catch (Exception e) {
			log.error(e.getMessage());
		} finally {
			zos.close();
		}
	}

	private static String toPath(File root, File dir) {
		String path = dir.getAbsolutePath();
		path = path.substring(root.getAbsolutePath().length()).replace(File.separatorChar, '/');
		if (path.startsWith("/"))
			path = path.substring(1);
		if (dir.isDirectory() && !path.endsWith("/"))
			path += "/";
		return path;
	}

    public static String formatNO(String no, boolean mask) {

        if (sb.length() == 10) { // 10자리 포멧

            if (sb.charAt(0) == '0' && sb.charAt(6) == '0' && sb.charAt(7) == '0') { // 13자리 포멧

    /**
     * 금액 문자열을 comma 표시 여부와 소수점 표시 여부를 확인하여 format된 문자열로 반환한다.
     * <p>
     * <ui>
     * <li> 금액 문자열이 음수(-)이면 반환된 값에도 음수(-)가 붙는다.</li>
     * <li> 금액 문자열에 소숫점이 있고 point 값이 있다면 둘 중 큰 숫자만큼 소수점 자리가 나온다.</li>
     * <li> 금액 문자열에 소숫점이 있고 point 값이 0이라면 금액 문자열의 소숫점이하가 0이면 절삭처리한다. 하지만 값이 있다면 절삭하지 않는다.</li>
     * </ui>
     * <pre>
     * 	"121,564,899.231" = formatMoney("0121564899.231", true, 0)
     * 	"121,564,899.231" = formatMoney("0121564899.231", true, 2)
     * 	"121564899.231" = formatMoney("0121564899.231", false, 2)
     * 	"121,564,899.2310" = formatMoney("0121564899.231", true, 4)
     * 	"121564899.2310" = formatMoney("0121564899.231", false, 4)
     * 	"-121,564,899.2310" = formatMoney("-0121564899.231", true, 4)
     * 	"-121564899.2310" = formatMoney("-0121564899.231", false, 4)
     * </pre>
     *
     * @param money  금액
     * @param comma  금액 표시시 천단위 구분자(,) 삽입여부
     * @param point  소수점의 자릿수
     */
    public static String formatMoney(String money, boolean comma, int point) {
        String pattern;
        if (comma) {
            pattern = "#,##0";
        } else {
            pattern = "###0";
        }
        int index = money.lastIndexOf('.');
        if (point < 1) {
            pattern += ".";
            index = money.length() - (index + 1);
            pattern += StringUtils.repeat('#', Math.max(index, 1));
        } else {
            pattern += ".";
            if (index < 0) {
                pattern += StringUtils.repeat('0', point);
            } else {
                index = money.length() - (index + 1);
                pattern += StringUtils.repeat('0', NumberUtils.max(index, point, 1));
            }
        }
        DecimalFormat decimalFormat = new DecimalFormat(pattern);
        double lAmt = Double.parseDouble(money.trim());
        return decimalFormat.format(lAmt);
    }

    public static String formatIPWithAsterisk(String ip) {
        if (StringUtils.isBlank(ip)) {

            }
        }

    }

    /**
     * UTF-8 인코딩으로 byte 수에 맞춰 문자열 잘라내는 메소드
     * 한글이 바이트 수에 걸쳐서 깨지는 경우 해당 한글 문자를 생략한다.
     */
    public static String cutWordUtf8(String word, int byteLength) {
    	String newWord = new String(word);
        try {
            double MAX_UTF8_CHAR_LENGTH = 4.0;
            if (newWord.length() > byteLength) {
                newWord = newWord.substring(0, byteLength);
            }
            if (newWord.length() > byteLength / MAX_UTF8_CHAR_LENGTH) {
                int residual;
                residual = newWord.getBytes("UTF-8").length - byteLength;
                if (residual > 0) {
                    int tempResidual = residual, start, end = newWord.length();
                    while (tempResidual > 0) {
                        start = end - ((int) Math.ceil((double) tempResidual / MAX_UTF8_CHAR_LENGTH));
                        tempResidual = tempResidual - newWord.substring(start, end).getBytes("UTF-8").length;
                        end = start;
                    }
                    newWord = newWord.substring(0, end);
                }
            }
            return newWord;
        } catch (UnsupportedEncodingException e) {
        	log.error(e.getMessage(), e);
        }
        return newWord;
    }

	public static String formatAcno(@NonNull String acno, boolean isMask) {
		final int LENGTH_ACNO = 14;

		if (StringUtils.isBlank(acno))
			return "";

		String str = acno.replace("-", "").trim();
		if (LENGTH_ACNO != str.length() || !StringUtils.isNumeric(str)) {

		if (isMask) {
			// 첫 5자리만 표시: 나머진 * 표시
			str = str.substring(0, 5) + StringUtils.repeat('*', LENGTH_ACNO - 5);

	}

    /**

     * 길이와 숫자만 들어왔는지 여부만 체크한다.
     * 예) FormatUtil.formatRnno("1234561234567", false ) => "123456-1234567"
     * 예) FormatUtil.formatRnno("1234561234567", true  ) => "123456-1******"
     * @param rnno 주민번호 ( 13 자리 )
     * @param isMask 마스크 여부
     * @return
     */
    public static String formatRnno( @NonNull String rnno , boolean isMask )
    {
        final int LENGTH_RNNO = 13;

        if ( StringUtils.isBlank( rnno )) return "";
        String str = rnno.replace("-", "").trim();
        if( LENGTH_RNNO != str.length() || !StringUtils.isNumeric( str ) ) {
            log.error("잘못된 주민번호");
            return rnno;
        }

        if(isMask)
        {
            str = str.substring(0,7) + StringUtils.repeat("*",6);
        }

        return FormatUtil.formatStringDelimiter( str, "-", new int[]{6,7}) ;
    }

    /**
     * 전화번호를 포맷팅한 문자열로 리턴한다.
     * 전화번호 검증은 숫자여부와 최소 길이여부만 체크한다.
     * <p/>
     * <pre>
     * FormatUtil.formatPhone("1231234", false);     => "123-1234"
     * FormatUtil.formatPhone("12341234", true);     => "1234-****"
     * FormatUtil.formatPhone("0212341234", false);  => "02-1234-1234"
     * FormatUtil.formatPhone("021231234", false);   => "02-123-1234"
     * FormatUtil.formatPhone("03112341234", true);  => "031-1234-****"
     * FormatUtil.formatPhone("0311231234", false);  => "031-123-1234"
     * </pre>
     * @param phoneno 전화번호
     * @param isMask 마스킹 여부
     * @return
     */
    public static String formatPhone(@NonNull String phoneno, boolean isMask )
    {
        String str = phoneno.replace("-", "").trim();

        if ( StringUtils.isBlank( str ) ) return "";
        if( 7 > str.length() || !StringUtils.isNumeric( str ) ) {
            log.error("잘못된 전화번호");
            return phoneno;
        }

        int len = str.length();
        String result = null;

        // 지역 번호 없는 경우
        if( len == 7 )
            result = FormatUtil.formatStringDelimiter( str, "-", new int[]{3,4}) ;
        // 지역 번호 없고 앞자리가 4자리인 경우
        else if( len == 8 )
            result = FormatUtil.formatStringDelimiter( str, "-", new int[]{4,4}) ;
        // 서울(02) 이고 전화번호 앞자리가 3자리인 경우
        else if( len == 9 && str.startsWith("02") )
            result = FormatUtil.formatStringDelimiter( str, "-", new int[]{2,3,4}) ;
        // 서울 이고, 전화번호 앞자리가 4자리인 경우
        else if( len == 10 && str.startsWith("02") )
            result = FormatUtil.formatStringDelimiter( str, "-", new int[]{2,4,4}) ;
        // 전화번호 앞자리가 3자리인 경우, 지역 번호 포함
        else if( len == 10 && !str.startsWith("02") )
            result = FormatUtil.formatStringDelimiter( str, "-", new int[]{3,3,4}) ;
        // 전화번호 앞자리가 4자리인 경우, 지역 번호 포함
        else if( len == 11 && !str.startsWith("02") )
            result = FormatUtil.formatStringDelimiter( str, "-", new int[]{3,4,4}) ;
        else {
            log.error("잘못된 전화번호");
            return phoneno;
        }


        // 마스킹 여부 확인
        if( isMask )
        {
            Matcher matcher = Pattern.compile( FORMATTER_PHONE_NUMBER ).matcher(result);
            if( matcher.find() ) {
                String replaceTaget = matcher.group(2);
                String maskStr = StringUtils.repeat('*', replaceTaget.length());

                result = result.replaceFirst(replaceTaget, maskStr );
            }
        }

        return result;
    }

    /**
     * email 주소를 마스킹.
     * <p>
     * email ID 앞 3자리 제외 마스킹 처리<br>
     * 입력값이 형식에 안 맞으면 "" 값을 리턴한다.
     * <pre>
     * ex) aaabbcc@email.com -> aaa****@email.com
     * </pre>
     */
    public static String formatEmail(@NonNull String email, boolean mask )
    {
        if ( StringUtils.isBlank( email )) return "";

        // email 검증 안 넣음
        if( mask )
        {
            String[] tokens = email.split("@");
            String emailId = tokens[0];
            String emailAdder = tokens[1];

            if( emailId.length() < 4 )
            {
                emailId = StringUtils.repeat('*', emailId.length() );
            }
            else {
                emailId = emailId.substring(0,3) + StringUtils.repeat('*', emailId.length()-3);
            }
            return emailId + "@" + emailAdder;
        }

        return email;
    }

    /**
     * 패스워드 길이만큼 '*' 표시함
     * <p/>
     * <pre>
     *     ****** -> FormatUtil.formatPassword( "123456" )
     * </pre>
     *
     * @param password
     * @return
     */
    public static String formatPassword( @NonNull String password )
    {
        return StringUtils.repeat("*", password.length());
    }

    /**
     * 고객 이름 마스킹
     * 홍길동 -> 홍**동
     * 맨앞 글자와 맨 뒤 글자만 표시 중간은 **
     */
    public static String getMaskedUserName( String name )
    {
        if( StringUtils.isBlank(name) )
        {
            return StringUtils.EMPTY;
        }

        // (사) (유) (주) 제거
        String newName = name.replaceAll("\\((.*?)\\)", "");

        if( StringUtils.isBlank(newName) )
        {
            return StringUtils.EMPTY;
        }

        String first =  String.valueOf(newName.charAt(0));
        String last =  (newName.length() > 1)? String.valueOf(newName.charAt(newName.length()-1)) : StringUtils.EMPTY;
        String middleToken = StringUtils.repeat("*", Math.min(2, Math.max(1, newName.length()-2)) ); // 최소 1개 최대 2개만 별표 표시

        String maskedName = first + middleToken + last;
        return maskedName;
    }

    /**
     * 사용자 ID
     * -> UserID 첫 5자 표시 나머지는 *
     * @param userId
     * @return
     */
    public static String getMaskedUserId( String userId )
    {
        String maskedUserId;


        if( StringUtils.isBlank( userId ) )
        {
            maskedUserId = StringUtils.EMPTY;
        }
        else {
            String newUserId = userId.trim();
            if( newUserId.length() > 5)
                maskedUserId = newUserId.substring(0,5) + StringUtils.repeat("*", newUserId.length() - 5);
            else {
                maskedUserId = newUserId.charAt(0) + StringUtils.repeat("*", newUserId.length()-1);
            }
        }
        return maskedUserId;
    }


    /**
     * OTP 시리얼 번호 마스킹
     * 12345678 -> 1234****
     * @param serialNo
     * @return

    public static String getMaskedOtpSerialNo( String serialNo )

        String maskedSerialNo;

        if( StringUtils.isBlank( serialNo) )
        {
            maskedSerialNo = StringUtils.EMPTY;
        }
        else {
            String newSerialNo = serialNo.trim();
            if( newSerialNo.length() > 4 )

            else
                maskedSerialNo = newSerialNo.charAt(0) + StringUtils.repeat("*", newSerialNo.length()-1);
        }
        return maskedSerialNo;
    }

	 * @param acno 계좌번호
	 * @return 기관코드(011=농협은행, 012=농협중앙회)
	 */
	public static String getCheckNHBankCd(String bankCd, String acno) {
		// 011,012가 아닐경우 농협은행이 아니므로, 입력된 은행코드를 그대로 반환
		if (!StringUtils.equalsAny(bankCd, "011", "012")) {
			return bankCd;
		}

		if (StringUtils.isBlank(acno)) {
			return bankCd;
		}

		// 농협은행,농협중앙회 기관코드 분류(계좌번호 패턴에 따른 분류)
		String newBankCd = "011";
		int len = acno.length();
		if (len == 10) {
			if (StringUtils.equalsAny(acno.substring(9,10), "9")) {
				newBankCd = "012";
			}
		} else if (len == 13) {
			if (StringUtils.equalsAny(acno.substring(12,13), "3", "4", "5", "9")) {
				newBankCd = "012";
			}
		} else if (len == 14) {
			if (StringUtils.equalsAny(acno.substring(0,3), "790", "791")) {
				newBankCd = "011";
			} else if (StringUtils.equalsAny(acno.substring(0,3), "792")) {
				newBankCd = "012";
			} else if (StringUtils.equalsAny(acno.substring(6,8), "64", "65")) {
				newBankCd = "011";
			} else if (StringUtils.equalsAny(acno.substring(6,8), "66", "67")) {
				newBankCd = "012";
			} else {
				newBankCd = "012";
			}
		}
		
		return newBankCd;
	}

package kr.fasol.obix.core.util;

import lombok.Getter;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;

/**
 * GUID 관리
 */
@Component
@Slf4j
@Getter
public class GuidContext {
    private String guid;
    private String childGuid;

    /**
     * GUID 설정
     */
    public void setGuid(String guid) {
        // GUID가 비어있는 경우에만 GUID를 저장해야 함.(기존값을 변경하면 안됨.)
        if (!StringUtils.hasLength(this.guid)) this.guid = guid;
    }

    /**
     * GUID 강제 재설정
     */
    public void setEnforceGuid(String guid) {
    	// GUID 강제 재설정
        this.guid = guid;
    }

    /**
     * ChildGuild 생성
     */
    public void createChildGuid() {
        // GUID를 사용하기 전에 처리. childGuid가 없으면 현재 guid가 childGuid. 있으면 guid의 hop을 붙여 childGuid를 생성.
        if (!StringUtils.hasLength(childGuid)) {
            childGuid = guid;
        } else if (StringUtils.hasLength(guid)) {
        	childGuid = GuidUtil.getGuidKey(guid) + GuidUtil.getGuidHop(childGuid);
        } else {
            childGuid = GuidUtil.getGuidKey(GuidUtil.createGuid()) + GuidUtil.getGuidHop(childGuid);
        }
    }

    /**
     * ChildGuild 생성
     */
    public synchronized void createNewChildGuid() {
        childGuid = GuidUtil.getGuidKey(GuidUtil.createGuid()) + GuidUtil.getGuidHop(guid);
    }

    /**
     * 시스템경로번호(NextHop) 재설정
     */
    public synchronized void setNextHop(String guid) {
        log.debug("prev guid info this.guid: {}, this.childGuid: {}, guid: {}", this.guid, this.childGuid, guid);

        // 보낸 GUID와 되돌아온 GUID의 Unique Key가 같아야 함. 그렇지 않으면 안됨.
        if (GuidUtil.getGuidKey(childGuid).equals(GuidUtil.getGuidKey(guid))) {
            this.guid = GuidUtil.nextGuid(GuidUtil.getGuidKey(this.guid) + GuidUtil.getGuidHop(guid));
            this.childGuid = GuidUtil.nextGuid(guid);
        } else {
            log.debug("Guid Process Error... childGuid: {}, guid: {}", childGuid, guid);
            this.guid = GuidUtil.nextGuid(this.guid);
            this.childGuid = GuidUtil.nextGuid(this.childGuid);
        }

        log.debug("next guid info this.guid: {}, this.childGuid: {}, guid: {}", this.guid, this.childGuid, guid);
    }
}

package kr.fasol.obix.core.util;

import jakarta.annotation.Nonnull;
import lombok.experimental.UtilityClass;
import lombok.extern.slf4j.Slf4j;
import org.apache.commons.lang3.StringUtils;

import java.net.InetAddress;
import java.net.UnknownHostException;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.concurrent.atomic.AtomicInteger;

/**
 * GUID 생성 규칙
 * GUID(36자리) = 일자(8) + 전문생성시간(9) + 채널코드(3) + 채널ID(7) + 일련번호(6) + 시스템경로번호(3)
 * <p>
 * 전문작성일자(8): 거래일자 (YYYYMMDD)
 * 전문작성시각(6): 거래시각 (hhmmssSSS)
 * 채널코드(3): MCI에서 구분하는 채널구분코드
 * 채널ID(7): 채널에서 서버 및 인스턴스 구분할 수 있는 값. 서버번호(5) + WAS Instance번호(2)
 * 일련번호(6)
 * 시스템경로번호(3): 증분
 */
@Slf4j
@UtilityClass
public class GuidUtil {

    private static final String DEFAULT_CHNL_CD_S3 = "OBX";
    private static final DateTimeFormatter DATE_FORMATTER = DateTimeFormatter.ofPattern("yyyyMMdd");
    private static final DateTimeFormatter TIME_FORMATTER = DateTimeFormatter.ofPattern("HHmmssSSS");
    private static final AtomicInteger MESG_SEQ_NO_COUNTER = new AtomicInteger(1);
    private static final String SYS_PATH_NO_S3 = "001";

    private static final String CHANNEL_SERVER_ID = StringUtils.leftPad(StringUtils.right(System.getProperty("application.name", getHostname()), 7), 7, "0").toUpperCase();

    public String createGuid() {
        final LocalDateTime now = LocalDateTime.now();

        final StringBuilder guidBuilder = new StringBuilder();
        guidBuilder.append(now.format(DATE_FORMATTER));
        guidBuilder.append(now.format(TIME_FORMATTER));
        guidBuilder.append(DEFAULT_CHNL_CD_S3);
        guidBuilder.append(CHANNEL_SERVER_ID);
        guidBuilder.append(String.format("%06d", getAndIncrement(MESG_SEQ_NO_COUNTER) % 1000_000));
        guidBuilder.append(SYS_PATH_NO_S3); // 생성시에는 001부터
        return guidBuilder.toString();
    }

    public String nextGuid(String guid) {
        final String nextGuid = getGuidKey(guid) + StringUtils.leftPad(String.valueOf(Integer.parseInt(getGuidHop(guid)) + 1), 3, "0");
        log.debug("guid: {}, nextGuid: {}", guid, nextGuid);
        return nextGuid;
    }

    private String getHostname() {
        try {
            return InetAddress.getLocalHost().getHostName();
        } catch (UnknownHostException e) {
            log.error(e.getMessage(), e);
        }
        return "Unknown";
    }

	// 정수형 오버플로우 방지
    private int getAndIncrement(AtomicInteger seqnoCounter) {
        int incrementValue = 0;
        int curInt = seqnoCounter.get();
        try {
            incrementValue = seqnoCounter.getAndIncrement();
            if (incrementValue < 0) {
                throw new Exception();
            }
        } catch (Exception e) {
            incrementValue = 0;
            seqnoCounter.set(incrementValue);
        }
        return curInt;
    }

    /**
     * GUID 에서 앞의 33자리 Unique Key 추출
     *
     * @param guid
     * @return
     */
    public String getGuidKey(@Nonnull String guid) {
        return (guid.length() > 33) ? guid.substring(0, 33) : guid;
    }

    /**
     * GUID 에서 뒤의 3자리 Hop 추출
     *
     * @param guid
     * @return
     */
    public String getGuidHop(@Nonnull String guid) {
        return (guid.length() == 36) ? guid.substring(33, 36) : SYS_PATH_NO_S3;
    }
}
