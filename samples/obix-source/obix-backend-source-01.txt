package kr.fasol.obix.backend.biz.common.service;

import kr.fasol.obix.backend.dao.comm.BizDayDAO;
import kr.fasol.obix.backend.dto.comm.HoldyBasQryDTO;
import kr.fasol.obix.core.util.DateUtil;
import lombok.RequiredArgsConstructor;
import org.apache.commons.lang3.time.DateUtils;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;

import java.text.SimpleDateFormat;
import java.util.*;
import java.util.stream.Collectors;

/**
 * 시스템명 : 
 * 업무구분명 : 공통 > 영업일조회
 * 프로그램명 : 영업일조회
 * 설명 :
 *
 * @Class BizDayService.class
 * @since 2021/04/15
 * @author 정영탁
 * @version 1.0

@Service
@RequiredArgsConstructor
public class BizDayService {

    private final BizDayDAO bizDayDAO;

    /**
     * 기준일로 부터 영업일을 조회한다.
     * @param baseDt 기준일
     * @param day - 0:당영업일, -1:전영업일, -2:전전영업일, 1:익영업일, 2:익익영업일
     * @return 영업일
     */
	public Date getBizDay(Date baseDt, int day) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);

		// 휴일목록 조회
		HoldyBasQryDTO req = new HoldyBasQryDTO();
		if (day < 0) {
			req.setStartDt(sdf.format(DateUtils.addMonths(baseDt, 0 - ((day / 15) + 1))));
			req.setEndDt(sdf.format(baseDt));
		} else {
			req.setStartDt(sdf.format(baseDt));
			req.setEndDt(sdf.format(DateUtils.addMonths(baseDt, ((day / 15) + 1))));
		}

		List<HoldyBasQryDTO> list = bizDayDAO.listHoldyBas(req);

		// 휴일 목록을 일자별 Map으로 변환
		Map<String, Object> bizDayMap = new HashMap<>();
		if (list != null && !list.isEmpty()) {
			bizDayMap = list.stream().collect(Collectors.toMap(HoldyBasQryDTO::getHoldyYmd, HoldyBasQryDTO::clone));
		}

		/*
		 * baseDt 기준 영업일 조회
		 */
		Calendar start = Calendar.getInstance();
		start.setTime(baseDt);

		if (day == 0) {
			while (true) {
				String targetDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(targetDt)) {
					break;
				}
				start.add(Calendar.DATE, 1);
			}
		} else {
			int bizDtCnt = 0;
			while (bizDtCnt < (day < 1 ? 0 - day : day)) {
				start.add(Calendar.DATE, (day < 1 ? -1 : 1));
				String targetDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(targetDt)) {
					bizDtCnt++;
				}
			}
		}

		return start.getTime();
	}

    /**
     * 기준일로 부터 영업일을 조회한다.
     * @param baseDt 기준일(yyyyMMdd)
     * @param day - 0:당영업일, -1:전영업일, -2:전전영업일, 1:익영업일, 2:익익영업일
     * @return 영업일(yyyyMMdd)
     */
	@Cacheable(cacheNames = "BIZDAY_LIST", key = "#baseDt + ':' + #day")
	public String getBizDay(String baseDt, int day) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);
		return sdf.format( getBizDay(sdf.parse(baseDt), day) );
	}


	/**
	 * 기준일로부터 대상일자가 몇 영업일인지 계산
	 * @param baseDt 기준일
	 * @param targetDt 대상일자
	 * @return - 0:당영업일, 1:익영업일, 2:익익영업일
	 */
	public int getBizDayDiff(Date baseDt, Date targetDt) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);

		// 휴일목록 조회
		HoldyBasQryDTO req = new HoldyBasQryDTO();
		req.setStartDt(sdf.format(baseDt));
		req.setEndDt(sdf.format(targetDt));

		List<HoldyBasQryDTO> list = bizDayDAO.listHoldyBas(req);

		// 휴일 목록을 일자별 Map으로 변환
		Map<String, Object> bizDayMap = new HashMap<>();
		if (list != null && !list.isEmpty()) {
			bizDayMap = list.stream().collect(Collectors.toMap(HoldyBasQryDTO::getHoldyYmd, HoldyBasQryDTO::clone));
		}


		int day = 0;
		if (!baseDt.equals(targetDt)) {
			// baseDt 기준 영업일 조회
			Calendar start = Calendar.getInstance();
			start.setTime(baseDt);

			while (true) {
				start.add(Calendar.DATE, 1);
				String diffDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(diffDt)) {
					day++;
				}
				if (targetDt.equals(sdf.parse(diffDt))) {
					 break;
				}
			}
		}

		return day;
	}

	/**
	 * 기준일로부터 대상일자가 몇 영업일인지 계산
	 * @param baseDt 기준일(yyyyMMdd)
	 * @param targetDt 대상일자(yyyyMMdd)
	 * @return - 0:당영업일, 1:익영업일, 2:익익영업일
	 */
	public int getBizDayDiff(String baseDt, String targetDt) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);
		return getBizDayDiff(sdf.parse(baseDt), sdf.parse(targetDt));
	}

}

@Service
@RequiredArgsConstructor
public class CmnCdService {

    private final CmnCdDAO cmnCdDAO;


	/**
	 * 코드그룹 조회
	 */
    @Cacheable(cacheNames = "CMN_CD_GROUP")
	public List<CmnCdGroupDTO> getListCmnCdGroup() throws Exception {
		return cmnCdDAO.listCmnCdGroup();
	}

    /**
     * 코드그룹 조회
     * @param cdGroupVal 코드그룹값
     */
	@Cacheable(cacheNames = "CMN_CD_GROUP", key = "#cdGroupVal")
    public CmnCdGroupDTO getCmnCdGroup(String cdGroupVal) throws Exception {
    	List<CmnCdGroupDTO> list = getListCmnCdGroup();
    	if (list != null && !list.isEmpty()) {
    		return list.stream().filter(x -> x.getCdGroupVal().equals(cdGroupVal)).findFirst().orElse(null);
    	}
    	return null;
    }

	/**
	 * 코드목록 조회
	 * @param req 코드그룹값
	 */
    @Cacheable(cacheNames = "CMN_CD", key = "#req.cdGroupVal")
	public List<CmnCdQryDTO> getListCmnCd(CmnCdQryDTO req) throws Exception {
		return cmnCdDAO.listCmnCdByCdGroupVal(req);
	}

	/**
	 * 코드목록 조회
	 * @param cdGroupVal 코드그룹값
	 */
	@Cacheable(cacheNames = "CMN_CD", key = "#cdGroupVal")
	public List<CmnCdQryDTO> getListCmnCd(String cdGroupVal) throws Exception {
		return getListCmnCd(CmnCdQryDTO.builder().cdGroupVal(cdGroupVal).build());
	}
	/**
	 * 코드목록 조회
	 * @param cdGroupVal 코드그룹값
	 * @param upCdVal 상위코드값
	 */
	public List<CmnCdQryDTO> getListCmnCd(String cdGroupVal, String upCdVal) throws Exception {
		List<CmnCdQryDTO> list = getListCmnCd(cdGroupVal);
		return list.stream().filter(x -> upCdVal.equals(x.getUpCdVal())).collect(Collectors.toList());
	}

	/**
	 * 코드정보 조회
	 * @param req 코드그룹값,코드값
	 */

		List<CmnCdQryDTO> list = getListCmnCd(req);
		return list.stream().filter(x -> x.getCdVal().equals(req.getCdVal())).findFirst().orElse(null);
	}

	/**
	 * 코드정보 조회
	 * @param cdGroupVal 코드그룹값

/**

 * 세부업무구분명 : 공통 > 에러 페이지
 * 프로그램명 : 에러 페이지

@Controller
@RequestMapping("${server.error.path:${error.path:/error}}")
public class ErrorPagesController extends AbstractErrorController {

	public ErrorPagesController(ErrorAttributes errorAttributes) {
		super(errorAttributes);
	}

	/**
	 * VIEW 페이지의 에러 정의
	 */
	@RequestMapping(produces = MediaType.TEXT_HTML_VALUE)
    public Object errorHtml(HttpServletRequest request, HttpServletResponse response) {

        /*
         * Error 구분에 따른 분기
         */
		String requestUri = (String) request.getAttribute(RequestDispatcher.ERROR_REQUEST_URI);
		if (StringUtils.trimToEmpty(requestUri).endsWith(".json")) {
			Response<?> errorResponse = error(request);
			// json 모드
			CustomMappingJackson2JsonView jsonView = new CustomMappingJackson2JsonView();
			jsonView.setAttributeResponse(errorResponse);
			return new ModelAndView(jsonView);
		}

		Object status = request.getAttribute(RequestDispatcher.ERROR_STATUS_CODE);
    	if (status != null) {
    		int statusCode = Integer.parseInt(status.toString());
    		if (statusCode == HttpStatus.NOT_FOUND.value()) {
    			response.setStatus(HttpStatus.OK.value());
    			return "/error/404";
    		} else if (statusCode == HttpStatus.FORBIDDEN.value()) {
    			response.setStatus(HttpStatus.OK.value());
    			return "/error/404";
    		} else if (statusCode == HttpStatus.INTERNAL_SERVER_ERROR.value()) {
    			response.setStatus(HttpStatus.OK.value());
    			return "/error/500";
    		}
    	}
    	return "/error/error";
    }

	/**
	 * VIEW 없는 페이지(API)의 에러 정의
	 */
	@RequestMapping
	public Response<?> error(HttpServletRequest request) {
    	Object status = request.getAttribute(RequestDispatcher.ERROR_STATUS_CODE);
    	if (status != null) {
    		int statusCode = Integer.parseInt(status.toString());
    		if (statusCode == HttpStatus.NOT_FOUND.value()) {
    			return Response.builder().statusAndMessage(CodeEnum.NOT_EXISTS.getCode(), HttpStatus.NOT_FOUND.getReasonPhrase()).build();
    		} else if (statusCode == HttpStatus.FORBIDDEN.value()) {
    			return Response.builder().statusAndMessage(CodeEnum.NOT_EXISTS.getCode(), HttpStatus.FORBIDDEN.getReasonPhrase()).build();
    		} else if (statusCode == HttpStatus.INTERNAL_SERVER_ERROR.value()) {
    			Exception exception = (Exception) request.getAttribute(RequestDispatcher.ERROR_EXCEPTION);
    			if (exception instanceof ServletException && ((ServletException) exception).getRootCause() instanceof UserException userException) {
                    String errorMessage = AppMessageUtil.getMessage(userException.getErrorCode());
					Object[] errorMessageValue = userException.getMessageValue();
					if (errorMessageValue != null && errorMessageValue.length > 0) {
						errorMessage = MessageFormat.format(errorMessage, errorMessageValue);
					}
					return Response.builder().statusAndMessage(userException.getErrorCode(), errorMessage).build();
    			}
    			return Response.builder().statusAndMessage(CodeEnum.NOT_EXISTS.getCode(), HttpStatus.INTERNAL_SERVER_ERROR.getReasonPhrase()).build();
    		}
    	}
		return Response.builder().statusAndMessage(CodeEnum.STATUS_BAD_REQUEST.getCode(), HttpStatus.BAD_REQUEST.getReasonPhrase()).build();
	}

}

        String serviceType = MDC.get(LoggingConstant.SERVICE_TYPE);
		String pathWithinHandlerMapping = (String) request.getAttribute(HandlerMapping.PATH_WITHIN_HANDLER_MAPPING_ATTRIBUTE);

		//final String requestURI = request.getRequestURI();
		final Map<String, String> requestHeaderMap = WebUtil.requestHeaderMap(request);
		//final Map<String, String> responseHeaderMap = WebUtil.responseHeaderMap(response);
		final Map<String, Object> parameterMap = WebUtil.parameterMap(request);

		/*
		 * WEB I/O 로그 출력
		 */
		if (log.isDebugEnabled()) {
			StringBuilder renderInfo = new StringBuilder();
			renderInfo.append("\n######################################################################");
			renderInfo.append("\n### FRAMEWORK RENDER INFO START");
			renderInfo.append("\n");

			renderInfo.append("SERVICE_ID : ").append(serviceId).append("\n");
			renderInfo.append("SERVICE_TYPE : ").append(serviceType).append("\n");

			// request header 출력
			renderInfo.append("HEADER : [").append("\n");
			for (Map.Entry<String, String> entry : requestHeaderMap.entrySet()) {

				renderInfo.append("\t").append(entry.getKey()).append(" = ").append(gson.toJson(entry.getValue())).append("\n");
			}
			renderInfo.append("]\n");

			if (handler instanceof HandlerMethod handlerMethod) {
				if (hasAnnotation(handlerMethod, RestController.class) || hasAnnotation(handlerMethod, RequestBody.class) || hasAnnotation(handlerMethod, ResponseBody.class)) {
					if (request instanceof MyRequestWrapper requestWrapper) {
						final String requestBody = new String(requestWrapper.getRawData(), Charset.forName(requestWrapper.getCharacterEncoding()));
						// request body 출력
						renderInfo.append("REQUEST-BODY : [").append("\n");
						if (!requestBody.isEmpty()) {
							renderInfo.append("\t").append(requestBody).append("\n");
						}
						renderInfo.append("]\n");
					}
				}
			}

			renderInfo.append("\n### FRAMEWORK RENDER INFO END");
			renderInfo.append("\n#####################################################################");

			log.debug(renderInfo.toString());
		}
    }

	private <A extends Annotation> boolean hasAnnotation(HandlerMethod handlerMethod, Class<A> annotationType) {
		if (handlerMethod == null || annotationType == null) {
			return false;
		}
		return Optional.ofNullable(handlerMethod.getMethodAnnotation(annotationType))
				.orElse(handlerMethod.getBeanType().getAnnotation(annotationType)) != null;
	}

 * 업무구분명 : 공통 > Util > Admin 사용자 비밀번호 검증

@UtilityClass
public class PswdValidUtil {

    private static final int USERID_MIN_LENGTH = 6;
    private static final int USERID_MAX_LENGTH = 13;

    private static final int PWSD_COMBI_2_MIN_LENGTH = 10;
    private static final int PWSD_COMBI_3_MIN_LENGTH = 8;

    private static final String PATTERN_SPECIAL_CHAR = "[!@#$%^&*()+=-]";
    private static final String PATTERN_SAME_CHAR = "(.)\\1\\1";

    private static final String USERID_PATTERN_REGEX = "^[0-9a-zA-z]{6,13}$";
    private static final String[] USERID_COMBI_PATTERNS = {"[0-9]", "[a-zA-z]"};

    private static final String[] PSWD_COMBI_PATTERNS = {"[0-9]", "[a-z]", "[A-Z]", PATTERN_SPECIAL_CHAR};

    /**
     * 비밀 번호 정책에 따른 비밀 번호 검증
     * 영문대문자, 영문소문자, 숫자, 특수문자의 4종류 조합으로 생성
     * - 2종류 조합시 10자리 이상, 3종류 조합시 8자리 이상
     * @param pswd 검증 대상 비밀번호
     * @return 검증 여부( 성공:true, 실패: false)
     */
    private static boolean isValidByPasswordPolicy(@NonNull final String pswd) {
        int matchCnt = 0;
        for (String iter : PSWD_COMBI_PATTERNS) {
            Pattern pattern = Pattern.compile(iter);
            Matcher matcher = pattern.matcher(pswd);
            if (matcher.find()) {
                matchCnt++;
            }
        }

        if (matchCnt == 2 && PWSD_COMBI_2_MIN_LENGTH <= StringUtils.length(pswd)) {
            return true;
        } else if (matchCnt >= 3 && PWSD_COMBI_3_MIN_LENGTH <= StringUtils.length(pswd)) {
            return true;
        }
        return false;
    }

    /**
     * 특수문자 포함여부
     * @param pswd 검증 대상 비밀번호
     * @return 검증 여부( 성공:true, 실패: false) - 특수문자가 포함되어야 함
     */
    private static boolean isValidSpecialChar(@NonNull final String pswd) {
        Pattern pattern = Pattern.compile(PATTERN_SPECIAL_CHAR);
        Matcher matcher = pattern.matcher(pswd);
        return matcher.find();
    }

    /**
     * 동일한 문자 체크 (예: aaa, 777)
     * @param pswd 검증 대상 비밀번호
     * @return 검증 여부( 성공:true, 실패: false ) - 동일한 문자/숫자가 있는 경우 false
     */
    public static boolean isValidDuplicate3Character(@NonNull final String pswd) {
        Pattern p = Pattern.compile(PATTERN_SAME_CHAR);
        Matcher m = p.matcher(pswd);
        if (!m.find()) return true;
        return false;
    }

    /**
     * 연속된 문자 체크 (예: abc, 123)
     * @param pswd
     * @return 검증 여부( 성공:true, 실패: false) - 연속된 글자가 있는 경우 false
     */
    private static boolean isValidContinuous3Charater(@NonNull final String pswd) {
        byte[] bytes = pswd.getBytes();

        for (int i = 1; i < bytes.length - 1; i++) {
            if (bytes[i] == (byte) ((bytes[i - 1] + bytes[i + 1]) / 2)
                    && 2 == Math.abs(bytes[i - 1] - bytes[i + 1])) {
                return false;
            }
        }
        return true;
    }

    private static String QWERTY_SEQUENCE = "qwertyuiop|asdfghjkl|zxcvbnm";

    /**
     * Qwerty 기반 배열 연속성 여부
     * - 4자 이상만 체크 한다.
     * - 가로만 해야 할 듯 하다. 세로까지 고려하기에 경우의 수가 많다.
     * @param pswd
     * @return
     */
    private static boolean isValidQwerySequence4(@NonNull final String pswd) {
        // forward, backward 둘다 되도록 처리함
        for (int i = 0; i < pswd.length() - 3; i++) {
            String str = pswd.substring(i, i + 4);

            if (!Pattern.matches("[a-z]{4}", str)) continue;

            if (QWERTY_SEQUENCE.contains(str)
                    || QWERTY_SEQUENCE.contains(StringUtils.reverse(str)))
                return false;
        }

        return true;
    }

    /**
     * 비밀번호만 가지고 검증하는 로직
     * @param pswd 검증 대상 비밀번호
     * @return 비밀번호 검증 여부 (true: 통과)
     */
    private static boolean isValidPassword(@NonNull final String pswd) {
        return (isValidSpecialChar(pswd)           // 비밀번호 검증 : 특수문자여부
                && isValidByPasswordPolicy(pswd)    // 비밀번호 검증 : 비밀번호 정책
                && isValidContinuous3Charater(pswd) // 비밀번호 검증 : 연속된 숫자/문자 3개 이상 여부
                && isValidDuplicate3Character(pswd) // 비밀번호 검증 : 동일 숫자/문자 3개 연속으로 있는경우
                && isValidQwerySequence4(pswd)
        );
    }

    /**
     * 비밀 번호와 유사성 유사성 여부
     * - 동일한 문자/숫자가 아이디/패스워드에 동일하게 들어가는지 확인
     * @param pswd 검증 대상 비밀번호
     * @param arg 비교대상 문자
     * @return 검증 여부( 성공:true, 실패: false) - 유사한 글자가 있는 경우 false
     */
    private static boolean isValidEqualCharacter(@NonNull final String pswd, @NonNull String arg) {
    	if (pswd.contains(arg)) { return false; }
    	return true;
    }

    /**
     * 비밀 번호와 유사성 유사성 여부
     * - 동일한 3글자 이상의 문자/숫자가 아이디/패스워드에 동일하게 들어가는지 확인
     * @param pswd 검증 대상 비밀번호
     * @param arg 비교대상 문자 ( 3글자 이상 )
     * @return 검증 여부( 성공:true, 실패: false) - 유사한 글자가 있는 경우 false
     */
    private static boolean isValidEqual3Character(@NonNull final String pswd, @NonNull String arg) {
    	int len = arg.length();
    	for (int i = 0; i < len - 2; i++) {
    		if (pswd.contains(arg.substring(i, i + 3))) return false;
    	}
    	return true;
    }

    /**
     * 아이디와 유사성 여부 체크 복수형
     * 복수의 문자열을 비밀번호와 넣어서 비교하기 위한 함수
     * @param pswd 검증 대상 비밀번호
     * @param args 복수의 문자열 Collection 형태로 전달
     * @return 검증 여부( 성공:true, 실패: false)  - 유사한 글자가 있는 경우 false
     */
    private static boolean isValidEqual3Character(@NonNull final String pswd, @NonNull List<String> args) {
        for (String arg : args) {
            if (!isValidEqual3Character(pswd, arg))
                return false;
        }
        return true;
    }

    /**
     * 비밀번호 유효일자 체크
     * - 비밀번호는 최근 변경일자 기준 60일 동안만 유효하다.
     * - 비밀번호 마지막 수정 일자를 기준으로 60일을 더한 일자와 오늘 일자를 변경한다.
     * @param lastChngDt 마지막 비밀번호 수정 일자
     * @return (lastChngDt+60) >= now : true // (lastChngDt+60) < now : false
     */
    private static int PSWD_VALID_DAYS = 60;

    @NonNull
    public static boolean isValidPswdChngDt(String lastChngDt) {
        if (StringUtils.isBlank(lastChngDt)) return true;

        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyyMMdd");
        LocalDate validPswdDt = LocalDate.parse(lastChngDt, formatter).plusDays(60);

        // 비밀번호 유효일자가 오늘보다 이후이면 true, 아니면 false;
        return validPswdDt.isAfter(LocalDate.now());
    }

    /**
     * 사용자 ID 유효성 체크
     * - 영문(대소문자 구분없음)숫자 혼용 6~13
     * - 영문 숫자 조합 필수
     * @param bnknUserId
     * @return
     */
    private static boolean isValidUserId(String bnknUserId) {
        // 조합 여부
        int matchCnt = 0;
        for (String iter : USERID_COMBI_PATTERNS) {
            Pattern pattern = Pattern.compile(iter);
            Matcher matcher = pattern.matcher(bnknUserId);
            if (matcher.find()) {
                matchCnt++;
            }
        }

        if (2 > matchCnt) return false;

        return Pattern.matches(USERID_PATTERN_REGEX, bnknUserId);
    }

    /**
     * 비밀번호 테스트
     * @param pswd 비밀번호
     * @param userId 사용자 아이디
     * @param cpno 사용자 전화번호
     * @throws UserException
     */
    public static boolean checkValidPswd(@NonNull String pswd, String userId, String cpno) throws UserException {
        //비밀번호 유효성 체크
        // TODO 비밀번호유효성검증 오류코드 정의 필요

        if (StringUtils.isBlank(pswd)) {
            throw new UserException("BB3007").setDebuggingMessage("비밀번호 미입력");
        }
        if (!isValidByPasswordPolicy(pswd)) {
            throw new UserException("BB3009").setDebuggingMessage("영대문자, 영소문자, 숫자, 특수문자를 조합해서 8자리 이상.");
        }
        if (!isValidSpecialChar(pswd)) {
            throw new UserException("BB3009").setDebuggingMessage("특수문자 미포함");
        }
        if (!isValidContinuous3Charater(pswd)) {
            throw new UserException("BB3010").setDebuggingMessage("연속된 문자(숫자)");
        }
        if (!isValidDuplicate3Character(pswd)) {
            throw new UserException("BB3011").setDebuggingMessage("동일한 문자(숫자)");
        }
        if (StringUtils.isNotBlank(userId)) {
            if (!isValidEqualCharacter(pswd, userId)) {
                throw new UserException("BB3018").setDebuggingMessage("아이디와 동일한 문자 포함");
            }
        }
        if (StringUtils.isNotBlank(cpno)) {
            String cpno1 = cpno.startsWith("010") ? cpno.substring(3) : cpno;
            if (!isValidEqual3Character(pswd, cpno1)) {
                throw new UserException("BB3013").setDebuggingMessage("사용자의 휴대전화와 동일한 문자(숫자) 3개이상 포함");
            }
        }
        if (!isValidQwerySequence4(pswd)) {
            throw new UserException("BB3014").setDebuggingMessage("키보드상의 연속된 배열");
        }
        return true;
    }

    /**
     * 비밀번호 유효성 체크 패스워드 Only
     *
     * @param pswd 신규비밀번호
     * @return
     * @throws Exception
     */
    public static boolean checkValidPswd(@NonNull final String pswd) throws Exception {
        return checkValidPswd(pswd, null, null);
    }

}

/**
 * 시스템명 : 
 * 업무구분명 : 설정 > DB의 Transaction 설정
 * 프로그램명 : DB의 Transaction 설정
 * 설명 :
 *
 * 클래스 : AppConfigTransaction.java
 * 작성자 : 정영탁
 * 작성일 : 2021. 12. 13.
 * --------------------------------------------------------------------
 * Modification Information
 * --------------------------------------------------------------------
 * 수정일               수정자            수정내용
 * --------------------------------------------------------------------
 * 2021. 12. 13.         정영탁          최초생성
 */
@Configuration
public class AppConfigTransaction {

    /**
     * TransactionManager 설정
     */
	@Bean(name={"txManager", "transactionManager"})
	public DataSourceTransactionManager txManager(@Qualifier("dataSource") DataSource dataSource) {
		DataSourceTransactionManager dataSourceTransactionManager = new DataSourceTransactionManager();
		dataSourceTransactionManager.setDataSource(dataSource);
		return dataSourceTransactionManager;
	}

	@Bean
	public TransactionInterceptor txAdvice(DataSourceTransactionManager txManager) {
		RuleBasedTransactionAttribute txAttribute = new RuleBasedTransactionAttribute();
		txAttribute.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRED);
		txAttribute.setRollbackRules(Collections.singletonList(new RollbackRuleAttribute(Exception.class)));

		HashMap<String, TransactionAttribute> txMethods = new HashMap<String, TransactionAttribute>();
		txMethods.put("*", txAttribute);

		NameMatchTransactionAttributeSource txAttributeSource = new NameMatchTransactionAttributeSource();
		txAttributeSource.setNameMap(txMethods);

		TransactionInterceptor txAdvice = new TransactionInterceptor();
		txAdvice.setTransactionAttributeSource(txAttributeSource);
		txAdvice.setTransactionManager(txManager);

		return txAdvice;
	}

	@Bean
	public Advisor txAdvisor(@Qualifier("txManager") DataSourceTransactionManager txManager) {
		AspectJExpressionPointcut pointcut = new AspectJExpressionPointcut();
		pointcut.setExpression(
				"execution(* kr.fasol..service.*.*(..))"
		);
		return new DefaultPointcutAdvisor(pointcut, txAdvice(txManager));
	}

}

@Configuration
@EnableCaching
public class CacheManagerConfig {

    @Bean
    public javax.cache.CacheManager jCacheManager() {
        CachingProvider cachingProvider = Caching.getCachingProvider();
        return cachingProvider.getCacheManager();
    }

    @Bean
    public org.springframework.cache.CacheManager cacheManager(javax.cache.CacheManager jCacheManager) {

        // Cache Entry Listener Configuration
        MutableCacheEntryListenerConfiguration<Object, Object> cacheEntryListenerConfig =
                new MutableCacheEntryListenerConfiguration<>(
                        FactoryBuilder.factoryOf(CustomCacheEntryListener.class),
                        null, // CacheEntryEventFilter (필터가 필요 없는 경우 null)
                        true, // 동기 설정
                        true  // 오래된 값 필요 여부
                );

        javax.cache.configuration.Configuration<Object, Object> configuration = new MutableConfiguration<>()
                .addCacheEntryListenerConfiguration(cacheEntryListenerConfig) // 캐시에 리스너 구성 추가
                .setTypes(Object.class, Object.class)
                .setExpiryPolicyFactory(CreatedExpiryPolicy.factoryOf(Duration.ONE_DAY))
                .setStoreByValue(false)
                .setStatisticsEnabled(true);

        // Cache 생성
        jCacheManager.createCache("BIZ_MENU_LIST", configuration);
        jCacheManager.createCache("BIZDAY_LIST", configuration);
        jCacheManager.createCache("CMN_CD_GROUP", configuration);
        jCacheManager.createCache("CMN_CD", configuration);

        return new JCacheCacheManager(jCacheManager);
    }

}

import org.springframework.security.config.annotation.web.configurers.HeadersConfigurer;

/**
 * 시스템명 : 
 * 업무구분명 : 공통 > 보안
 * 프로그램명 : 
 * 설명 :

 * 수정일               수정자            수정내용
 * --------------------------------------------------------------------
 * 2023. 7. 19.        정영탁			  최초작성
 */
@Configuration
public class WebSecurityConfig {

	@Bean
	public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {

		http
				.csrf(CsrfConfigurer::disable)
				.formLogin(Customizer.withDefaults())
				.authorizeHttpRequests(authorizeRequest ->
						authorizeRequest
								.anyRequest().permitAll()
				)
				.headers(
						headersConfigurer ->
								headersConfigurer
										.frameOptions(
												HeadersConfigurer.FrameOptionsConfig::sameOrigin
										)
				);

		return http.build();
	}

	@Bean
	public WebSecurityCustomizer webSecurityCustomizer() {
		return (web) ->
				web
						.httpFirewall(new DefaultHttpFirewall())
						.ignoring()
						.requestMatchers(
								PathRequest.toStaticResources().atCommonLocations()
						);
	}

}

package kr.fasol.obix.backend.util;

import jakarta.servlet.http.HttpServletRequest;
import kr.fasol.obix.core.base.validation.ByteSize;
import kr.fasol.obix.core.config.BusinessTierHeaderInfoProvider;
import kr.fasol.obix.core.enums.AppHeader;
import kr.fasol.obix.core.enums.LoggingConstant;
import kr.fasol.obix.core.util.GuidContext;
import kr.fasol.obix.core.util.GuidUtil;
import kr.fasol.obix.core.util.WebUtil;
import lombok.Getter;
import lombok.extern.slf4j.Slf4j;
import org.apache.commons.lang3.StringUtils;
import org.slf4j.MDC;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Scope;
import org.springframework.context.annotation.ScopedProxyMode;
import org.springframework.stereotype.Component;
import org.springframework.web.context.WebApplicationContext;

import java.time.LocalDateTime;

/**
 * request별로 정보를 담아놓으려고 생성함
 * 현재는 Guid만 담고 있는데 더 필요한 값이 생긴다면 클래스 명을 변경해야 함.
 */
@Slf4j
@Component
@Scope(scopeName = WebApplicationContext.SCOPE_REQUEST, proxyMode = ScopedProxyMode.INTERFACES)
@Getter
public class BusinessTierHeaderInfoProviderImpl implements BusinessTierHeaderInfoProvider {

    // Header
    @ByteSize(max = 36)
    private final String guid;                          // 전문의 Unique한 고유번호

    @ByteSize(max = 39)
    private final String frstTrnmIpad;                  // 최초 전송 IP 주소
    @ByteSize(max = 3)
    private final String frstTrnmSysCd;                 // 최초 전송 시스템 코드

    // Common
    @ByteSize(max = 4)
    private final String txBrcd;                        // 거래부점코드
    @ByteSize(max = 1)
    private final String chnlTycd;                      // 채널유형코드
    @ByteSize(max = 3)
    private final String chnlDtlsClcd;                  // 채널세부분류코드
    @ByteSize(max = 3)
    private final String langDvcd;                      // 언어구분코드
    // Bypass client app Header.
    private final String deviceId;
    private final String pid;
    @ByteSize(max = 3)
    private final String clphOsDvcd;                    // 휴대전화운영체계 구분코드

    private final String userAgent;
    private final GuidContext guidContext;
    private final LocalDateTime requestDateTime;

    @Autowired
    public BusinessTierHeaderInfoProviderImpl(HttpServletRequest request, GuidContext guidContext) {
        this.guidContext = guidContext;

        // guidContext에 GUID가 없으면 생성하고 저장한다.
        if (StringUtils.isEmpty(this.guidContext.getGuid())) {
            this.guid = StringUtils.defaultIfBlank(MDC.get(LoggingConstant.REQUEST_GUID), GuidUtil.createGuid());
            this.guidContext.setGuid(this.guid);
        } else {
            log.debug("guidContext.getGuid is not empty. guidContext.getGuid(): {}", this.guidContext.getGuid());
            this.guid = this.guidContext.getGuid();
        }
        log.debug("guid: {}", guid);

        // 헤더에서 X-OBIX-IP (L4에서 전달해 준 IP)가 있으면 사용. 없으면 X-Forward-For 사용. 그것도 없으면 RemoteAddr 을 사용한다. (없으면 계정계 통신시 오류 발생)
        this.frstTrnmIpad = StringUtils.defaultIfBlank(getHeader(request, AppHeader.FRST_TRNM_IPAD), StringUtils.defaultIfBlank(getHeader(request, AppHeader.X_FORWARDED_FOR), WebUtil.getClientIp(request)));
        this.frstTrnmSysCd = "OBIX";

        this.txBrcd = "0000";
        this.chnlTycd = "M";    // 채널유형코드(I:인터넷, M:모바일)
        this.chnlDtlsClcd = "BEA";  // 채널세부분류코드(MPA: Front, ADM: 관리자, BEA: Back-End)
        this.langDvcd = StringUtils.defaultIfBlank(getHeader(request, AppHeader.LANG_DVCD),"KOR");;

        this.deviceId = "";
        this.pid = getHeader(request, AppHeader.PID);

        // 휴대전화운영체계 구분코드(IOS, AOS, ETC)
        this.clphOsDvcd = "ETC";

        this.userAgent = getHeader(request, AppHeader.USER_AGENT);
        this.requestDateTime = LocalDateTime.now();
    }

    private String getHeader(HttpServletRequest request, AppHeader appHeader) {
        return request.getHeader(appHeader.getFieldName());
    }

    @Override
    public String getSysOprtEnvDvcd() {
        throw new UnsupportedOperationException("getSysOprtEnvDvcd() method is unsupported.");
    }

    @Override
    public String getTxTypDvcd() {
        throw new UnsupportedOperationException("getTxTypDvcd() method is unsupported.");
    }

    @Override
    public void beforeSend() {
        guidContext.createChildGuid();
    }

    @Override
    public void setGuid(String guid) {
        guidContext.setNextHop(guid);
    }

    @Override
    public String getGuid() {
        return guidContext.getGuid();
    }

    @Override
    public String getChildGuid() {
        return guidContext.getChildGuid();
    }

}

* Description : 페이징을 위한 쿼리
* - MariaDB에서는 ROW_NUMBER() OVER()를 사용할 수 없어서 변수를 사용하여 페이징 처리
* - MySQL 8.0.2 이상부터 ROW_NUMBER() OVER()를 사용할 수 있음
* - PostgreSQL에서는 ROW_NUMBER() OVER()를 사용할 수 있음
-->
<mapper namespace="COMMON">

    <sql id="pagingPreSQL">
        SELECT *
        FROM (
            SELECT
                @rownum := @rownum + 1 AS ROW_NUM,
                #{pageNo} AS PAGE_NO,
                #{pageSize} AS PAGE_SIZE,
                AA.*
            FROM (
    </sql>

    <sql id="pagingPostSQL">
                LIMIT 18446744073709551615
            ) AA
            CROSS JOIN (SELECT @rownum := 0) BB
        ) SUB
        WHERE ROW_NUM BETWEEN ((#{pageNo} - 1) * #{pageSize} + 1) AND (#{pageNo} * #{pageSize})
    </sql>
