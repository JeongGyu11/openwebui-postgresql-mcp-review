// 사용자 권한 정보 조회
function ASYAU40201() {
	console.log('ASYAU40201 호출', '사용자 권한 정보 조회');

	var param = {
			userId : g_userId
	};

	/* 사용자 권한 트리 조회 */
	httpAjax("ASYAU40201.json", param, function (data) {
		var result = data.result.list;

		// jsTree 체크 처리
		for(var idx in result) {
			var obj = result[idx];

			obj.state = {
				opened : true,
				selected : (obj.useYn == 'Y')
			};
		}

		$('#jstree').jstree({
			"core": {
				"data": result,
				"checkbox": { 
					"keep_selected_style": false,  // default: false
					"three_state": true,           // default: true
					"whole_node": true             // default: true
				},
			},
			"checkbox" : {
				'three_state': false
			},
			"plugins" : [
				"checkbox",
				"contextmenu",
				"types",
				"unique",
				"changed"
			]
		})
		.bind('loaded.jstree', function(e, data) {
			 //font-awesome 폴더 아이콘 제어
             $('#jstree').on("open_node.jstree", function(e, data) {
                 if(data.node.original.screnTranDvCd == '0') {
                     var node = data.instance.get_node(data.node.id);
                     data.instance.set_icon(node, "far fa-folder-open new-fs16");
                 }
             });
             $('#jstree').on("close_node.jstree", function(e, data) {  
                 if(data.node.original.screnTranDvCd == '0') {
                     var node = data.instance.get_node(data.node.id);
                     data.instance.set_icon(node, "far fa-folder new-fs16");
                 }
             });
	            
			 $.each(result, function(i, item) {
                 // 아이콘 설정 - 카테고리('0')는 폴더 / 화면거래는 파일로 구분
                 if (item.screnTranDvCd == '0') {
                	 data.instance.set_icon(item.id, "far fa-folder new-fs16");
                 } else if(item.screnTranDvCdNm.indexOf("화면") != -1) {
                     data.instance.set_icon(item.id, "far fa-window-maximize new-fs16");
                 } else {
                     data.instance.set_icon(item.id, "far fa-save new-fs16");
                 }
                 // 해당 메뉴권한이 없으면 skip
                 if (item.menuAuthCd == 'N') {
                     return true;
                 }
                 //// 하위메뉴가 존재하지 않는 경우에만 select 처리 (하위메뉴 존재 시 권한이 없는 메뉴까지 모두 select되기 때문)
                 if (data.instance.get_node(item.id).children.length == 0) {
                	 data.instance.select_node(item.id);
                 }
                 // 전체허용메뉴와 그룹권한메뉴는 Disabled 처리
                 if (item.menuAuthCd == 'A' || item.menuAuthCd == 'G') {
                	 data.instance.disable_node(item.id);
                 }
             });
			 
			 $('#jstree').on("changed.jstree", function(e, data) {
                if(data.action == "select_node") {
                    // 수정 시 오리지널 그룹 권한과 비교하여 타입 변경
                    $.each(data.changed.selected, function(i, item) {
                        var node = data.instance.get_node(item);
                        if(node.original.menuAuthCd == 'A' || node.original.menuAuthCd == 'G') {
                            data.instance.set_type(item, "default");
                        } else {
                            data.instance.set_type(item, "changed");
                            data.event.currentTarget.blur(); // focus 상태 없애기(bootstrap 클래스 color 때문)
                        }
                    });
                } else if(data.action == "deselect_node") {
                    // 수정 시 오리지널 그룹 권한과 비교하여 타입 변경
                    $.each(data.changed.deselected, function(i, item) {
                        var node = data.instance.get_node(item);
                        if(node.original.menuAuthCd == 'N') {
                            data.instance.set_type(item, "default");
                        } else {
                            data.instance.set_type(item, "changed");
                            data.event.currentTarget.blur(); // focus 상태 없애기(bootstrap 클래스 color 때문)
                        }
                    });
                }
            });
		})
	});
}

function ASYAU40202() {
	console.log('ASYAU40202 호출', '사용자 권한 정보 저장');

	var params = {
			userId : g_userId
	};

	// parent node도 등록하기 위하여 selected node와 같이 가져온다. 
	var idx = 0;
	$.each($('#jstree').find('.jstree-undetermined, .jstree-clicked'), function(i, node) {
		var id = $(node).closest('.jstree-node').attr('id');
		if ($('#jstree').jstree(true).is_disabled(id)) {
			// 전체허용메뉴 및 그룹권한메뉴로 disable된 메뉴는 skip한다.
		} else {
			params["screnList["+idx+"].screnTranId"] = id;
			params["screnList["+idx+"].userId"] = g_userId;
			idx++;
		}
	});

	httpAjax("ASYAU40202.json", params, function (data) {
		biz.showAlert("등록 되었습니다.");
		popup.closeLayerPopup(true);
	});
}

function fncBatchListTable(result) {

	var html = "";
	$("#tBatchlist").empty();
	
	if (btchJobList != null && btchJobList.length > 0) {
		
		for(var i in btchJobList){
			var obj = btchJobList[i];
			
			if (obj === btchJobList[btchJobList.length - 1]) {	//마지막 요소에 border-bottom 추가
				html += '<tr class="text-center border-bottom">';
			} else {

			}
			html += '	<td>';
			html += 		(pages.pageNo-1)*pages.pageSize+Number(i)+1 //No.
			html += '	</td>';
			html += '	<td>'+ obj.btchJobId + '</td>';
			html += '	<td>';
			html += '		<a href="#" onclick="popASYOP30100(`'+ obj.btchJobId +'`);return false;">' + util.nvl(obj.btchJobNm,'') + '</a>';
			html += '	</td>';
			html += '	<td>' + util.nvl(obj.btchExePgmNm,'') + '</td>';
			html += '	<td>' + util.nvl(obj.btchCyclNm,'') +'</td>';
			html += '	<td>' + util.nvl(obj.btchExeDesigDd,'') + '</td>';
			html += '	<td>' + util.nvl(obj.btchJobTtm,'') + '</td>';
			html += '	<td>' + util.nvl(obj.autoExeYn,'') + '</td>';

	var param = {};

/*
 * @desc 배치관리 설정정보 등록
 */
function ASYOP30102(){

	// validation 체크
	if (!validation.checkFormValidate(document.btchCreateForm)) {
		return false;
	}

	var updateBtchJobObject = new Object();
	
	var btchCyclCd = $('#popupBtchCyclCd').val();
	
	updateBtchJobObject.btchJobId = $('#popupBtchJobId').val()	// 배치작업코드
	updateBtchJobObject.btchJobNm = $('#popupBtchJobNm').val();				//배치작업명
	updateBtchJobObject.btchExePgmNm = $('#popupBtchExePgmNm').val();	// 배치실행프로그램명
	updateBtchJobObject.btchExePrmt = $('#popupBtchExePrmt').val();	// 배치실행파라미터
	updateBtchJobObject.reExePosblYn = $('#popupReExePosblYn').val();		// 재실행가능여부
	updateBtchJobObject.autoExeYn = $('#popupAutoExeYn').val();		// 자동실행여부
	updateBtchJobObject.btchCyclCd = $('#popupBtchCyclCd').val();	// 배치주기코드
	if( btchCyclCd =="32" ){	//월중 처리
		updateBtchJobObject.btchExeDesigDd = $('#popupBtchExeDesigDd').val(); //배치날짜코드
	}else{
		updateBtchJobObject.btchExeDesigDd = "";
	}
	if( btchCyclCd == "90" ){	//CRON표현식
		updateBtchJobObject.cronExrs = $('#popupCronExrs').val(); //CRON표현
	} else {
		updateBtchJobObject.btchJobTtm = $('#popupBtchJobTtmHr').val() + $('#popupBtchJobTtmMin').val(); // 배치작업시간
	}
	updateBtchJobObject.aplyStartDt = $('#popupAplyStartDt').val();
	updateBtchJobObject.aplyEndDt = $('#popupAplyEndDt').val();
	updateBtchJobObject.descCtnt = $('#popupDescCtnt').val();
	updateBtchJobObject.useYn = $('#popupUseYn').val();

	/*--------------------------------------------------------------------------------*/
	/* httpAjax - 배치 작업 설정정보 저장                                             */
	/*--------------------------------------------------------------------------------*/
	
	httpAjax('ASYOP30102.json', updateBtchJobObject, function(data){
 		console.log("성공");
 		console.log(data);
 		
 		var result = data.result.processResult;
		if(result > 0){
			var closeJson = new Object();
			closeJson.result = result;
			
			popup.closeLayerPopup(closeJson);
		}		 
	}); 
}

* @desc 조회로직 - 수정으로 들어왔을 경우 호출
*/
function fncSetBtchInfo(Btch){
	fncPopupBtchCyclCd(Btch.btchCyclCd);
	document.querySelector('#popupBtchJobId').value = Btch.btchJobId;		//배치작업코드

	document.querySelector('#popupBtchExePgmNm').value = util.nvl(Btch.btchExePgmNm,'');		//배치실행프로그램명
	document.querySelector('#popupBtchExePrmt').value = util.nvl(Btch.btchExePrmt,'');		//배치실행파라미터
	document.querySelector('#popupReExePosblYn').value = Btch.reExePosblYn;		//재실행가능여부
	document.querySelector('#popupAutoExeYn').value = Btch.autoExeYn;		//자동실행여부
	document.querySelector('#popupBtchCyclCd').value = Btch.btchCyclCd;//배치주기코드
	console.log("Btch.btchExeDesigDd",Btch.btchExeDesigDd);
	if(Btch.btchExeDesigDd != "" && Btch.btchExeDesigDd != null){
		document.querySelector('#popupBtchExeDesigDd').value = Btch.btchExeDesigDd;//배치주기코드
	}
	document.querySelector('#popupBtchJobTtmHr').value = (Btch.btchJobTtm ? Btch.btchJobTtm.substr(0,2) : '');	//배치작업시간
	document.querySelector('#popupBtchJobTtmMin').value = (Btch.btchJobTtm ? Btch.btchJobTtm.substr(2,2) : '');	//배치작업시간
	document.querySelector('#popupCronExrs').value = util.nvl(Btch.cronExrs,'');			//CRON표현
	$('#popupAplyStartDt').datepicker('setDate', date.calcDateSimple(util.nvl(Btch.aplyStartDt,''), null, 'date')); //적용시작일자
	$('#popupAplyEndDt').datepicker('setDate', date.calcDateSimple(util.nvl(Btch.aplyEndDt,''), null, 'date')); //적용종료일자
	document.querySelector('#popupDescCtnt').value = util.nvl(Btch.descCtnt,'');			//비고
	document.querySelector('#popupUseYn').value = util.nvl(Btch.useYn,'');
}

// keyup 이벤트 설정
function setAutoExeYnKeyup(){
	$("#popupAutoExeYn").keyup(function(){
           $(this).val($(this).val().replace(/[^nyNY]/g, ''));		// 영어 n,N,y,Y가 아닐 경우 빈칸으로 대체
           $(this).val($(this).val().replace(/[a-zA-Z]{2,}/g, $(this).val()[0]));		// 영어가 두 글자 이상일 경우 한 글자만 사용
           $(this).val($(this).val().toUpperCase());		// 소문자 입력시 대문자로 대체
	});
}

function fncPopupBtchCyclCd(btchCyclCd){

	}
	if( btchCyclCd == '' || btchCyclCd == "90" ){	//CRON표현식
		$("#selectPopupBtchExeDesigTm").hide();
	} else {
		$("#selectPopupBtchExeDesigTm").show();
	}
}

function fncSearchListBtchExeTxn(){

	//pagination 기본
	var pageInfo = {

		rows: 10
	}

		dateClassification: "",	// 조회일자구분

	if( $('#dateClassification').val() !== "") {
		searchInfo.dateClassification = $('#dateClassification').val();
	}
	if( periodStart !== "") {
		searchInfo.btchStartDtm = periodStart;
	}
	if( periodEnd !== "") {
		searchInfo.btchFinDtm = periodEnd;
	}
	if( $('#btchExeStusCd').val() !== "") {

	if( $('#btchJobIdOrNm').val() !== "") {
		searchInfo.btchJobNm = $('#btchJobIdOrNm').val();
	}

    $.extend(searchInfo, pageInfo);	//병합
	
	if (searchInfo.dateClassification === "" && searchInfo.btchStartDtm === "" && searchInfo.btchFinDtm === "" && searchInfo.btchExeStusCd === "" && searchInfo.btchJobNm === "") { //아무내용도 검색하지 않았을 경우

    function fn_StartTail() {
    	if (!$('#applicationName').val()) {
    		alert('로그파일 경로를 선택해 주세요.');
    		$('#applicationName').focus();
    		return false;
    	}
    	if (!$('#clientIp').val()) {
    		alert('조회할 접속자 IP를 입력해 주세요.');
    		$('#clientIp').focus();
    		return false;
    	}

    	executeTail();
    }

    function fn_StopTail() {
    	biz.timeout.resetTimer();
		$('#tailStatus').text( 'Stop' );
    }

    function executeTail() {
    	biz.timeout.startTimer($('#tailRefreshTime').val(), function(minutes, seconds) {
    		$('#tailStatus').text( '[' + seconds + '초후에 Refresh]' );
    		if (seconds == 0) {

    			// 로그 조회
				httpAjax('logTailProcess', { applicationName : $('#applicationName').val(), clientIp : $('#clientIp').val(), preEndPoint : $('#preEndPoint').val() }, function(data){
					// success
					if (data.result['logData']) {
						$('#preEndPoint').val( (data.result['endPoint'] || '0') );
						$('#logViewerArea').append( getSearchWord( data.result['logData'].replaceAll('\n', '<br/>') ) );
						$('#logViewerArea').scrollTop($('#logViewerArea').prop('scrollHeight'));
					}

	    			// Timer 재시작
	    			setTimeout(executeTail, 1000);
				}, null, {
					useProgress : false
				});
    		}
    	});
    }

    function getSearchWord(str) {
    	var searchWord = $('#searchWord').val();
    	if (searchWord.length > 0) {
	    	str = str.replaceAll(searchWord, '<span style="color:red;">' + searchWord + '</span>');
    	}
    	str = str.replaceAll(' ', '&nbsp;');
    	str = str.replaceAll('\t', '&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;');
    	return str;
    }

    function fn_LogFileDownload() {
    	if (!$('#applicationName').val()) {
    		alert('로그파일 경로를 선택해 주세요.');
    		$('#applicationName').focus();
    		return false;
    	}
    	if (!$('#clientIp').val()) {
    		alert('조회할 접속자 IP를 입력해 주세요.');
    		$('#clientIp').focus();
    		return false;
    	}

		// 로그파일 목록 조회
		httpAjax('logListProcess', { applicationName : $('#applicationName').val(), clientIp : $('#clientIp').val() }, function(data){
			// success
			var _html = '';
			for (var i = 0; i < data.result.length; i++) {
				_html += '<a href="javascript:fileDownload(\'' + data.result[i] + '\');">' + data.result[i] + '</a><br/>';
			}
			biz.showAlert(_html, null, '닫기', '다운로드할 로그파일을 선택해 주세요.', {width: '400px'});
		});
    }

	// 파일 다운로드
	function fileDownload(filename) {
		navigate('logFileDownload.file', {applicationName : $('#applicationName').val(), filename : filename}, { useProgress : false });
	}
