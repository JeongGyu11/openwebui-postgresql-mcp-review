@Controller
@RequestMapping(value = "/admin/cm/so")
@Tag(name = "공통 > 로그인")
public class ACMSOController extends AbstractBaseController {

	/** AdminLoginService */
	@Resource(name = "ACMSOService")
	private ACMSOService aCMSOService;


	@Operation(summary = "로그인 페이지")
	@RequestMapping(value = "/ACMSO10000.view")
	public String ACMSO10000(
			@RequestParam(name="returnUrl", required=false) String returnUrl,
			@RequestParam(name="returnParam", required=false) String returnParam,
			ModelMap map, HttpServletRequest request) throws Exception {

		/*
		 * Return URL 설정
		 * - 로그인 후 돌아올 URL을 세션에 저장
		 */
		if (StringUtils.isNotBlank(returnUrl) && returnUrl.contains("/admin/cm/so/ACMSO") == false) {
			SessionUtil.put(SessionEnum.NEXT_USER_URL, returnUrl);
		}
		if (StringUtils.isNotBlank(returnParam)) {
			SessionUtil.put(SessionEnum.NEXT_USER_URL_PARAM, returnParam);
		}

		// request attribute 설정
		map.put("currentLocaleCode", RequestContextUtils.getLocale(request));

		return getDefaultViewName(request);
	}

	@Operation(summary = "로그인 수행")
	@RequestMapping(value = "/ACMSO10001.json")
	public Response<?> loginProcess(@ModelAttribute ACMSO10001DTO input, HttpServletRequest request) throws Exception {

		/*
		 * 기존 로그인세션 정보가 남아있다면 제거한다.
		 */
		if (AdminSessionHelper.isLogin()) {
			AdminSessionHelper.destroy();
		}

		// 비밀번호 복호화(Crypto.JS)
		if (StringUtils.isNotBlank(input.getUserPwd())) {
			input.setUserPwd( CryptoJsUtil.decryptAES(input.getUserPwd(), input.getUserId()) );
		}

		/*
		 * 로그인 수행
		 */
		ACMSO10001DTO output = aCMSOService.ACMSO10001(input);

		/*
		 * 로그인 후 되돌아갈 페이지(returnUrl) 지정
		 * 1. 세션에 기록된 returnUrl이 있으면(메뉴를 클릭하여 이동한 경우), 해당 페이지로 이동
		 * 2. 로그인 버튼을 클릭하여 이도한 경우, 기본 페이지로 이동
		 */
		String returnUrl = SessionUtil.getString(SessionEnum.NEXT_USER_URL);
		String returnParam = SessionUtil.getString(SessionEnum.NEXT_USER_URL_PARAM);
		SessionUtil.remove(SessionEnum.NEXT_USER_URL);
		SessionUtil.remove(SessionEnum.NEXT_USER_URL_PARAM);
		if (returnUrl != null) {
			output.setReturnUrl(returnUrl);
			output.setReturnUrlParam(returnParam);
		}

		return Response.ok(output);
	}

	@Operation(summary = "로그아웃 수행")
	@RequestMapping(value = "/ACMSO10002.json")
	public Response<?> ACMSO10002(HttpServletRequest request) throws Exception {

		/*
		 * 로그아웃 수행
		 */
		if (AdminSessionHelper.isLogin()) {
			AdminSessionHelper.destroy();
		}

		return Response.ok(null);
	}

	/**
	 * 사용자 정보 조회
	 */
	public ACMMN10101DTO ACMMN10101(ACMMN10101DTO input) throws Exception {
		// 사용자 정보 조회
		AdmUserBasQryDTO res = admUserBasDAO.selectAdmUserBas(DTOUtil.convertDTO(input, AdmUserBasQryDTO.class));

		// 사용자 패스워드 만기일 지정
		if (StringUtils.isBlank(res.getLastPwdChngDt())) {
			res.setPwdExptDt(DateUtil.getCurrentDate());
		} else {
			res.setPwdExptDt(DateUtil.plusDays(res.getLastPwdChngDt(), 90));
		}
		// 사용자 그룹명 조회 후 메인그룹 지정
		List<AdmGroupUserRelQryDTO> rellist = admUserBasDAO.listAllAdmGroupByUser(DTOUtil.convertDTO(input, AdmGroupUserRelQryDTO.class));
		for (int i = 0; i < rellist.size(); i++) {
			if (rellist.get(i).getGroupDvCd().equals(CodeConstant.GROUP_DV_CD.메인그룹.getCode())) {
				res.setGroupNm(rellist.get(i).getGroupNm());
			}
		}
		if (StringUtils.isBlank(res.getGroupNm()) && rellist.size() > 0) {
			res.setGroupNm(rellist.get(0).getGroupNm());
		}

		return ACMMN10101DTO.builder().dto(res).build();
	}

    /**
	 * 프로그램명 : 비밀번호 변경 처리
	 */
    public int ACMMN10102(ACMMN10102DTO req) throws Exception {
		// 비밀번호 유효성 체크
		PswdValidUtil.checkValidPswd(req.getUserPwd(), req.getUserId(), null);

		// 입력받은 신규 비밀번호 암호화
		req.setUserPwd(ObixSecurityUtil.SHA256(req.getUserPwd()));

		// 세션 설정
		req.setUserId(AdminSessionHelper.getAdminUserId());

		// 등록자/수정자 기본값 설정
		req.setFrstRegiUserId(AdminSessionHelper.getAdminUserId());
		req.setLastChngUserId(AdminSessionHelper.getAdminUserId());

		// 현재 비밀번호 검증
		AdmUserBasQryDTO userdto = admUserBasDAO.selectAllAdmUserBas(DTOUtil.convertDTO(req, AdmUserBasQryDTO.class));
		if (!userdto.getUserPwd().equals(ObixSecurityUtil.SHA256(req.getNowUserPwd()))) {
			// 현재 비밀번호가 틀렸습니다.
			throw new UserException("ADM005").setDebuggingMessage("현재 비밀번호 오류");
		}

		req.setAdmUserStusCd(CodeConstant.ADM_USER_STUS_CD.초기화.getCode());
		// 최근 5회 변경 비밀번호 중 동일 비밀번호 건수 가져오기
		if (adminLoginDAO.getAdmUserSamePwdCnt(DTOUtil.convertDTO(req, AdmUserBasDTO.class)) > 0) {
			// 최근에 사용한 비밀번호로 변경할 수 없습니다.
			throw new UserException("ADM004").setDebuggingMessage("최근 비밀번호 이미 존재");
		}

		// 비밀번호 재설정 이력 생성
		adminLoginDAO.insertAdmUserPwdChngTxn(DTOUtil.convertDTO(req, AdmUserPwdChngTxnDTO.class));

		// 비밀번호 재설정
		req.setAdmUserStusCd(CodeConstant.ADM_USER_STUS_CD.정상.getCode());
		return adminLoginDAO.updateAdmUserBasPwd(DTOUtil.convertDTO(req, AdmUserBasDTO.class));
    }

    /**
	 * 프로그램명 : 즐겨찾기 정보 추가
	 */
	public void ACMMN10201(ACMMN10201DTO req) throws Exception {
		AdmUserFvrtDtlDTO newFvrt = DTOUtil.convertDTO(req, AdmUserFvrtDtlDTO.class);
		AdmUserFvrtDtlDTO oldFvrt = admUserFvrtDtlDAO.selectAdmUserFvrtDtl(newFvrt);
		
		int result = 1;
		if(oldFvrt == null) {
			result *= admUserFvrtDtlDAO.insertAdmUserFvrtDtl(newFvrt);
		} else {
			result *= admUserFvrtDtlDAO.updateAdmUserFvrtDtl(newFvrt);
		}
		if(result != 1) {
			throw new UserException("ADM006").setDebuggingMessage("[즐겨찾기 정보 추가 실패] newFvrt = " + newFvrt.toJSONString());
		}

	/**
	 * 사용자 설정 인증
	 */
	public ACMSO10101DTO ACMSO10101(ACMSO10101DTO input) throws Exception {
		ACMSO10101DTO output = new ACMSO10101DTO();

		AdmUserBasQryDTO admUserBasQryDTO = admUserBasDAO.selectAdmUserBas(DTOUtil.convertDTO(input, AdmUserBasQryDTO.class));

		// 등록이 안된 사용자일 경우 에러 처리
		if (admUserBasQryDTO == null) {
			throw new UserException("ADM001").setDebuggingMessage("등록되지 않은 사용자");
		}

		// 운영 환경에서만 체크 함
		if (ProfileUtil.acceptsProfiles(Constants.ENV_PRODUCTION, Constants.ENV_DR)) {
			if (!StringUtils.equals(input.getUserNm(), admUserBasQryDTO.getUserNm()) || !StringUtils.equals(input.getCpNo(), admUserBasQryDTO.getCpNo())) {
				throw new UserException("ADM001").setDebuggingMessage("값이 유효하지 않는 사용자");
			}
		}

		// 사용자 상태에 따른 에러 처리
		if (CodeConstant.ADM_USER_STUS_CD.정상.getCode().equals(admUserBasQryDTO.getAdmUserStusCd()) || CodeConstant.ADM_USER_STUS_CD.초기화.getCode().equals(admUserBasQryDTO.getAdmUserStusCd())) {
			//TODO 사용자인증을 위한 정보 발송
			userCertificationMailSend(admUserBasQryDTO);

			// 정상 or 초기화 상태
			output = DTOUtil.convertDTO(admUserBasQryDTO, ACMSO10101DTO.class);
			return output;
		}
		else if (CodeConstant.ADM_USER_STUS_CD.잠김.getCode().equals(admUserBasQryDTO.getAdmUserStusCd())) {
			// 잠김 상태
			throw new UserException("ADM002").setDebuggingMessage("잠김 상태의 사용자");
		}
		else {
			// 그 이외의 상태
			throw new UserException("ADM003").setDebuggingMessage("이용할 수 없는 상태의 사용자");
		}
	}

	/**
	 * 사용자 인증을 위한 메일 발송
	 */
	private void userCertificationMailSend(AdmUserBasDTO admUserBasDTO) throws Exception {
		// 6자리 인증번호를 Random하게 생성
		String approvalNum = getRandomString(6);

		log.info("### 사용자 인증을 위한 메일 발송 ###");
		log.info("### 사용자ID = {}", admUserBasDTO.getUserId());
		log.info("### OTP 번호 = {}", approvalNum);

		// 승인번호를 세션에 저장
		SessionUtil.put(SessionEnum.SELF_CERT_SESSION, approvalNum);

		// 메일 발송
		try {
			Map<String, Object> mailDataMap = new HashMap<>();
			mailDataMap.put("approvalNum", approvalNum);

			mailSenderUtil.sendMessage(MailSenderUtil.MailMessageDto.builder()
					.fromEmail(EnvironmentUtil.getProperty("MAIL.FROM.EMAIL", ""))
					.fromName(EnvironmentUtil.getProperty("MAIL.FROM.NAME", ""))
					.toEmail(admUserBasDTO.getEmail())
					.subject("[ObiX 관리 시스템] 비밀번호 재설정을 위한 승인번호")
					.text(getTemplateHtml("user_certification", mailDataMap))
					.text("인증번호 : " + approvalNum)
					.build());
		} catch (Exception e) {
			log.error("[이메일 발송][발송실패]", e);
		}
	}
	/**
     * Random한 승인번호 생성
     */
    private static String getRandomString(int size) {
		char[] digits = "0123456789".toCharArray();
		Random r = new Random(System.currentTimeMillis());

		StringBuilder buf = new StringBuilder();
		r.setSeed(System.currentTimeMillis());
		for (int i = 0; i < size; i++) {
			buf.append(digits[r.nextInt(10)]);
		}

		return buf.toString();
    }
	/**
	 * 메일 발송 템플릿 반환
	 */
	public String getTemplateHtml(String templateName, Map<String, Object> mailDataMap) {
		try {
			// 메일 본문 템플릿 설정
			Context context = new Context(Locale.KOREA);
			context.setVariables(mailDataMap);
			return templateEngine.process(templateName, context);
		} catch (Exception e) {
			log.error("[메일 템플릿 생성 실패]", e);
		}
		return null;
	}

	/**
	 * 비밀번호 재설정 처리
	 */
	public boolean ACMSO10301(ACMSO10301DTO input) throws Exception {
		// 등록자/수정자 기본값 설정
		input.setFrstRegiUserId(AdminSessionHelper.getAdminUserId());
		input.setLastChngUserId(AdminSessionHelper.getAdminUserId());

		//비밀번호 유효성 체크
		PswdValidUtil.checkValidPswd(input.getUserPwd(), input.getUserId(), null);

		//입력받은 신규 비밀번호 암호화
		input.setUserPwd(ObixSecurityUtil.SHA256(input.getUserPwd()));

		input.setAdmUserStusCd(CodeConstant.ADM_USER_STUS_CD.초기화.getCode());

		//최근 5회 변경 비밀번호 중 동일 비밀번호 건수 가져오기
		if (adminLoginDAO.getAdmUserSamePwdCnt(DTOUtil.convertDTO(input, AdmUserBasDTO.class)) > 0) {
			// 최근에 사용한 비밀번호로 변경할 수 없습니다.
			throw new UserException("ADM004").setDebuggingMessage("최근 5회 비밀번호 이미 존재");
		}

		//비밀번호 재설정 이력 생성
		adminLoginDAO.insertAdmUserPwdChngTxn(DTOUtil.convertDTO(input, AdmUserPwdChngTxnDTO.class));

		//비밀번호 재설정
		input.setAdmUserStusCd(CodeConstant.ADM_USER_STUS_CD.정상.getCode());
		AdmUserBasDTO admUserBasDTO = DTOUtil.convertDTO(input, AdmUserBasDTO.class);
		adminLoginDAO.insertAdmUserBasHst(admUserBasDTO);
		adminLoginDAO.updateAdmUserBasPwd(admUserBasDTO);

		return true;
	}

		input.setFrstRegiUserId(AdminSessionHelper.getAdminUserId());
		input.setLastChngUserId(AdminSessionHelper.getAdminUserId());

		//비밀번호 유효성 체크
		PswdValidUtil.checkValidPswd(input.getUserPwd(), input.getUserId(), null);

		//입력받은 신규 비밀번호 암호화
		input.setUserPwd(ObixSecurityUtil.SHA256(input.getUserPwd()));

		//현재 비밀번호 검증
		AdmUserBasQryDTO userdto = admUserBasDAO.selectAllAdmUserBas(DTOUtil.convertDTO(input, AdmUserBasQryDTO.class));
		if (!userdto.getUserPwd().equals(ObixSecurityUtil.SHA256(input.getNowUserPwd()))) {
			//현재 비밀번호가 틀렸습니다.
			throw new UserException("ADM005").setDebuggingMessage("현재 비밀번호 오류");
		}

		input.setAdmUserStusCd(CodeConstant.ADM_USER_STUS_CD.초기화.getCode());

		//최근 5회 변경 비밀번호 중 동일 비밀번호 건수 가져오기
		if (adminLoginDAO.getAdmUserSamePwdCnt(DTOUtil.convertDTO(input, AdmUserBasDTO.class)) > 0) {
			// 최근에 사용한 비밀번호로 변경할 수 없습니다.
			throw new UserException("ADM004").setDebuggingMessage("최근 비밀번호 이미 존재");
		}

		//비밀번호 재설정 이력 생성
		adminLoginDAO.insertAdmUserPwdChngTxn(DTOUtil.convertDTO(input, AdmUserPwdChngTxnDTO.class));

		//비밀번호 재설정
		input.setAdmUserStusCd(CodeConstant.ADM_USER_STUS_CD.정상.getCode());
		AdmUserBasDTO admUserBasDTO = DTOUtil.convertDTO(input, AdmUserBasDTO.class);
		adminLoginDAO.insertAdmUserBasHst(admUserBasDTO);
		adminLoginDAO.updateAdmUserBasPwd(admUserBasDTO);

		return true;
	}

	 * - 사용자 접근 화면 남김
	 * - 사용자 메뉴의 view 화면만 남김
