package kr.fasol.obix.core.base;

import com.fasterxml.jackson.annotation.JsonIgnore;
import lombok.Data;

import java.io.Serial;
import java.io.Serializable;
import java.util.ArrayList;
import java.util.List;

/**
 * 페이징 정보 구현체
 */
@Data
public class PageInfoImpl implements IPageInfo, Serializable {
    @Serial
    private static final long serialVersionUID = 6626382060530239164L;

    /*
     * 페이징 처리
     */
    private int pageNo = 1;		    // 페이지 번호
    private int totalCount;			// 전체 데이터 수
    private int pageSize = 10;		// 페이지 당 출력할 항목 수
    private int pageBlock = 10;		// 페이지바에 출력할 페이지 수
    private int rowNum = 0;         // 게시물의 순번

    /**
     * 쿼리 페이징을 위한 페이번호 설정
     */
    @JsonIgnore
    public int getPageNoIndex() {
        return (pageSize * (pageNo - 1));
    }

    /**
     * 총 페이지 수
     */
    @JsonIgnore
    public int getTotalPage() {
        return ((totalCount - 1) / pageSize) + 1;
    }

    /**
     * 페이징 출력 시작번호
     */
    @JsonIgnore
    public int getStartPage() {
        return ((pageNo - 1) / pageBlock) * pageBlock + 1;
    }

    /**
     * 페이징 출력 끝번호
     */
    @JsonIgnore
    public int getEndPage() {
        return ((getStartPage() + (pageBlock - 1)) < getTotalPage() ? getStartPage() + (pageBlock - 1) : getTotalPage());
    }

    /** 페이징에 출력할 페이지 번호 */
    @JsonIgnore
    public List<Integer> getPages() {
        List<Integer> list = new ArrayList<>();
        for (int i = getStartPage(); i <= getEndPage(); i++) {
            list.add(i);
        }
        return list;
    }

    /**
     * 처음 링크의 페이지 번호
     */
    @JsonIgnore
    public int getFirstPage() {
        return 1;
    }
    /**
     * 이전 링크의 페이지 번호
     */
    @JsonIgnore
    public int getPrevPage() {
        if (getStartPage() > 1) {
            return getStartPage() - 1;
        } else {
            return 1;
        }
    }
    /**
     * 다음 링크의 페이지 번호
     */
    @JsonIgnore
    public int getNextPage() {
        if (getEndPage() < getTotalPage()) {
            return getEndPage() + 1;
        } else {
            return getTotalPage();
        }
    }
    /**
     * 마지막 링크의 페이지 번호
     */
    @JsonIgnore
    public int getLastPage() {
        return getTotalPage();
    }

}

    public static final Response<?> ERROR = Response.builder().status("CB1000", "처리가 정상적으로 되지 않았습니다.").build();
    public static final Response<?> BAD_REQUEST = Response.builder().status("CB1003", "잘못된 요청입니다.").build();

    private String status;
    private String message;
    private Object common;
    //private ErrorInfo errorInfo;
    private T result;
    private String detailMessage;

    public Response() {
    }
    public Response(String status, String message, T result) {
        this.status = status;
        this.message = message;
        this.result = result;
    }
    public Response(String status, String message, Object common, T result) {
        this.status = status;
        this.message = message;
        this.common = common;
        this.result = result;
    }
    public Response(String status, String message, Object common, T result, ErrorInfo errorInfo, String detailMessage) {
        this.status = status;
        this.message = message;
        this.common = common;
        this.result = result;
        //this.errorInfo = errorInfo;
        this.detailMessage = detailMessage;
    }

    public static <T> Response<T> ok(T body) {
        return new ResponseBuilder<T>().result(body).build();
    }

    public static <T> ResponseBuilder<T> builder() {
        return new ResponseBuilder<>();
    }

    public static <T> Map<String, ?> convertMap(Response<T> response) {
        Map<String, Object> result = new LinkedHashMap<>();
        result.put("status", response.getStatus());
        result.put("message", response.getMessage());
        result.put("common", response.getCommon());
        if (response.getResult() != null) {
            result.put("result", JsonUtil.convertValue(response.getResult(), new TypeReference<T>() {}));
        }
        result.put("detailMesssage", response.getDetailMessage());
        result.put("success", response.isSuccess());
        return result;
    }

    public boolean isSuccess() {
        return StringUtils.equals(status, "CB0000");
    }

    @Getter
    @Setter
    @Builder
    public static class ErrorInfo {
        private String errCd;
        private String errMsgCtnt;
        private String errMsgDesc;
        private String errMngmCtnt;
        private String errLoctCtnt;
    }

    public static class ResponseBuilder<T> {
        private String status = "CB0000";
        private String message = "성공하였습니다.";
        private Object common;
        private String detailMessage;
        private T result;
        private ErrorInfo errorInfo;

        ResponseBuilder() {
        }

        public Response.ResponseBuilder<T> status(String code, String message) {
            this.status = code;
            this.message = message;
            return this;
        }

        public Response.ResponseBuilder<T> statusAndMessage(String status, String message) {
            return status(status, message);
        }

        public Response.ResponseBuilder<T> common(Object common) {
            this.common = common;
            return this;
        }

        public Response.ResponseBuilder<T> errorInfo(ErrorInfo errorInfo) {
            this.errorInfo = errorInfo;
            return this;
        }

		public Response.ResponseBuilder<T> result(T result) {
            this.result = result;
            return this;
        }

        public Response.ResponseBuilder<T> detailMessage(String detailMessage) {
            this.detailMessage = detailMessage;
            return this;
        }

        public Response<T> build() {
            return new Response<>(status, message, common, result, errorInfo, detailMessage);
        }
    }
}

package kr.fasol.obix.core.base.validation;

import jakarta.validation.ConstraintValidator;
import jakarta.validation.ConstraintValidatorContext;
import lombok.extern.slf4j.Slf4j;
import org.apache.commons.lang3.StringUtils;

import java.io.UnsupportedEncodingException;

@Slf4j
public class ByteSizeConstraintValidator implements ConstraintValidator<ByteSize, String> {

    private int min;
    private int max;
    private String charset;

    @Override
    public void initialize(ByteSize constraintAnnotation) {
        charset = constraintAnnotation.charset();
        min = constraintAnnotation.min();
        max = constraintAnnotation.max();
    }

    @Override
    public boolean isValid(String value, ConstraintValidatorContext context) {
        final String v = StringUtils.defaultString(value, StringUtils.EMPTY);
        try {
            final int len = v.getBytes(charset).length;
            return len >= min && len <= max;
        } catch (UnsupportedEncodingException e) {
            log.error(e.getMessage(), e);
        }
        return false;
    }
}

public abstract class AsyncThreadConfiguration {

	/**
	 * ThreadPool Task Executor 생성
	 */
	public Executor configThreadPoolTaskExecutor(int corePoolSize, int maxPoolSize, int queueCapacity, int keepAliveSeconds, int awaitTerminationSeconds, String threadNamePrefix) {
		ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
		executor.setCorePoolSize(corePoolSize);		// 기본적으로 실행을 대기하고 있는 Thread의 갯수
		executor.setMaxPoolSize(maxPoolSize);		// 동시 동작하는, 최대 Thread 갯수
		executor.setQueueCapacity(queueCapacity);		// MaxPoolSize를 초과하는 요청이 Thread 생성 요청시 해당 내용을 Queue에 저장하게 되고, 사용할 수 있는 Thread 여유 자리가 발생하면 하나씩 꺼내져서 동작하게 된다.
		executor.setKeepAliveSeconds(keepAliveSeconds);	// 하나의 요청에 대한 최대 연결 시간(단위:초)
		executor.setRejectedExecutionHandler(new ThreadPoolExecutor.CallerRunsPolicy());	// shutdown 상태가 아니라면 ThreadPoolTaskExecutor에 요청한 thread에서 직접 처리
		executor.setAllowCoreThreadTimeOut(true);	// KeepAliveSeconds로 설정된 시간만큼 Task가 할당되지 않은 Thread를 core thread로 정리할지 말지 여부 설정
		executor.setWaitForTasksToCompleteOnShutdown(false);	// 종료시 queue에 남아있는 모든 작업이 완료될 때까지 대기할지 여부
		executor.setAwaitTerminationSeconds(awaitTerminationSeconds);	// shutdown 최대 대기시간
		executor.setThreadNamePrefix(threadNamePrefix);	// Spring이 생성하는 Thread의 접두사 지정
		executor.setTaskDecorator(new LoggingTaskDecorator());
		executor.initialize();
		return executor;
	}

    /**
     * Logging을 위한 TaskDecorator 구현체
     */
    public static class LoggingTaskDecorator implements TaskDecorator {
		@Override
		public Runnable decorate(Runnable runnable) {
			Map<String, String> callerThreadContext = MDC.getCopyOfContextMap();
			return () -> {
				if (callerThreadContext != null) {
					MDC.setContextMap(callerThreadContext);
				}
				runnable.run();
			};
		}
    }

}

/**
 * 프로그램명 : SQL Insert/Update 시의 Audit 컬럼 기본값 설정 Interceptor
 */
@Slf4j
@RequiredArgsConstructor
@Intercepts(@Signature(type = Executor.class, method = "update", args = {MappedStatement.class, Object.class}))
public class AuditInterceptor implements Interceptor {

    private final String applicationName;

    @Override
    public Object intercept(Invocation invocation) throws Throwable {
        ServletRequestAttributes servletRequestAttributes = (ServletRequestAttributes) RequestContextHolder.getRequestAttributes();
        String pageId = "";

        // request header에서 추출
        if (servletRequestAttributes != null) {
            HttpServletRequest request = servletRequestAttributes.getRequest();
            pageId = request.getHeader("PAGE-ID");
            if (StringUtils.isBlank(pageId)) {
                pageId = applicationName;
            }
        } else {
            pageId = applicationName;
        }

        log.debug("### [AuditInterceptor] [pageId]: {}", pageId);

        MappedStatement mappedStatement = (MappedStatement) invocation.getArgs()[0];
        SqlCommandType sqlCommandType = mappedStatement.getSqlCommandType();
        Object parameter = invocation.getArgs()[1];
        if (parameter instanceof AbstractBaseDTO dto) {
            if (SqlCommandType.INSERT.equals(sqlCommandType)) {
                if (StringUtils.isBlank(dto.getFrstRegiUserId())) { dto.setFrstRegiUserId(pageId); }
            } else if (SqlCommandType.UPDATE.equals(sqlCommandType)) {
                if (StringUtils.isBlank(dto.getLastChngUserId())) { dto.setLastChngUserId(pageId); }
            }
        }
        return invocation.proceed();
    }

}

@Slf4j
@Intercepts({
        @Signature(type = Executor.class, method = "update", args = {MappedStatement.class, Object.class})
        ,@Signature(type = Executor.class, method = "query" , args = {MappedStatement.class, Object.class, RowBounds.class, ResultHandler.class})
        ,@Signature(type = StatementHandler.class, method = "prepare", args = {Connection.class, Integer.class})
})
public class SqlLogInterceptor implements Interceptor {
    private static final Pattern PARAM_PLACEHOLDER = Pattern.compile("\\?");

    @Override
    public Object intercept(Invocation invocation) throws InvocationTargetException, IllegalAccessException, NoSuchFieldException {
        if (invocation.getTarget() instanceof Executor) {
            return interceptExecutor(invocation);
        } else {
            return interceptStatementHandler(invocation);
        }
    }

    /**
     * Executor를 가로채서 SQL-ID 추출
     */
    private Object interceptExecutor(Invocation invocation) throws InvocationTargetException, IllegalAccessException {
        MappedStatement mappedStatement = (MappedStatement) invocation.getArgs()[0];
        MDC.put("SQL-ID", mappedStatement.getId());
        MDC.put("SQL-COMMAND-TYPE", (mappedStatement.getSqlCommandType() != null ? mappedStatement.getSqlCommandType().name() : ""));

        // implement pre processing if need
        Object returnObject = invocation.proceed();
        // implement post processing if need

        int resultCount = getResultCount(returnObject);
        log.debug("SQL-결과건수 : {}", resultCount);

        return returnObject;
    }

    /**
     * SQL 로그 출력
     */
    public Object interceptStatementHandler(Invocation invocation) throws InvocationTargetException, IllegalAccessException {
        String sqlId = MDC.get("SQL-ID");
        String sqlCommandType = MDC.get("SQL-COMMAND-TYPE");
        MDC.remove("SQL-ID");
        MDC.remove("SQL-COMMAND-TYPE");

        StatementHandler statementHandler = (StatementHandler) invocation.getTarget();
        MetaObject metaObject = SystemMetaObject.forObject(statementHandler);
        BoundSql boundSql = (BoundSql) metaObject.getValue("delegate.boundSql");

        String originalSql = boundSql.getSql(); // 원본 SQL
        Object parameterObject = boundSql.getParameterObject(); // 파라미터

        // SQL에 파라미터 값 매핑
        String filledSql = replacePlaceholders(originalSql, parameterObject, boundSql);

        if (log.isDebugEnabled()) {
            log.debug("SQL-COMMAND-TYPE : {}", sqlCommandType);
            log.debug("SQL-ID : {}", sqlId);
            log.debug("Executing SQL: {}", filledSql);
        }

        // implement pre processing if need
        Object returnObject = invocation.proceed();
        // implement post processing if need

        return returnObject;
    }

    /**
     * SQL에 파라미터 값 매핑
     */
    private String replacePlaceholders(String sql, Object parameterObject, BoundSql boundSql) {
        List<Object> parameters = extractParameters(parameterObject, boundSql);

        Matcher matcher = PARAM_PLACEHOLDER.matcher(sql);
        StringBuilder sb = new StringBuilder();
        int index = 0;

        try {
            while (matcher.find() && index < parameters.size()) {
                Object value = parameters.get(index++);
                String replacement = (value instanceof String) ? "'" + value + "'" : value == null ? "NULL" : value.toString();
                matcher.appendReplacement(sb, Matcher.quoteReplacement(replacement));
            }
            matcher.appendTail(sb);
            return sb.toString();
        } catch (Exception e) {
            log.error("Failed to replace placeholders in SQL: {}", sql, e);
            return sql;
        }
    }

    /**
     * 파라미터 추출
     */
    private List<Object> extractParameters(Object parameterObject, BoundSql boundSql) {
        List<Object> parameters = new ArrayList<>();
        if (parameterObject instanceof Map) {
            @SuppressWarnings("unchecked")
            Map<String, Object> paramMap = (Map<String, Object>) parameterObject;
            paramMap.forEach((key, value) -> {
                if (!key.startsWith("__frch")) { // MyBatis 내부 키 필터링
                    parameters.add(value);
                }
            });
        } else if (parameterObject != null) {
            MetaObject metaObject = SystemMetaObject.forObject(parameterObject);
            boundSql.getParameterMappings().stream()
                    .map(ParameterMapping::getProperty)
                    .filter(metaObject::hasGetter)
                    .forEach(name -> parameters.add(metaObject.getValue(name)));
        }
        return parameters;
    }

    /**
     * 쿼리 실행결과 건수 반환
     */
    private int getResultCount(Object returnObject) {
        if (returnObject instanceof List<?>) {
            return ((List<?>) returnObject).size();
        } else if (returnObject instanceof Object[]) {
            return ((Object[]) returnObject).length;
        } else if (returnObject instanceof Long) {
            return ((Long) returnObject).intValue();
        } else if (returnObject instanceof Integer) {
            return (Integer) returnObject;
        } else if (returnObject != null) {
            return 1;
        }
        return 0;
    }

    @Override
    public Object plugin(Object target) {
        return Plugin.wrap(target, this);
    }

    @Override
    public void setProperties(Properties properties) {
        // 필요시 설정값 사용
    }

}

package kr.fasol.obix.core.config.db.type;

import org.apache.ibatis.type.BaseTypeHandler;
import org.apache.ibatis.type.JdbcType;
import org.apache.ibatis.type.MappedJdbcTypes;
import org.apache.ibatis.type.TypeHandler;

import java.io.StringReader;
import java.sql.*;

@MappedJdbcTypes({JdbcType.CLOB, JdbcType.NCLOB})
public class CustomClobTypeHandler extends BaseTypeHandler<String> implements TypeHandler<String> {

	@Override
	public void setNonNullParameter(PreparedStatement ps, int i, String parameter, JdbcType jdbcType) throws SQLException {
		if (jdbcType == JdbcType.CLOB || jdbcType == JdbcType.NCLOB) {
			StringReader reader = new StringReader(parameter);
			ps.setCharacterStream(i, reader, parameter.length());
		} else {
			ps.setString(i, parameter);
		}
	}

	@Override
	public String getNullableResult(ResultSet rs, String columnName) throws SQLException {
		Clob clob = rs.getClob(columnName);
		try {
			return toString(clob);
		} finally {
			if (clob != null) {
				clob.free();
			}
		}
	}

	@Override
	public String getNullableResult(ResultSet rs, int columnIndex) throws SQLException {
		Clob clob = rs.getClob(columnIndex);
		try {
			return toString(clob);
		} finally {
			if (clob != null) {
				clob.free();
			}
		}
	}

	@Override
	public String getNullableResult(CallableStatement cs, int columnIndex) throws SQLException {
		Clob clob = cs.getClob(columnIndex);
		try {
			return toString(clob);
		} finally {
			if (clob != null) {
				clob.free();
			}
		}
	}

	private String toString(Clob clob) throws SQLException {
		return clob == null ? null : clob.getSubString(1, (int) clob.length());
	}
}

@Data
@EqualsAndHashCode(callSuper=false)
public abstract class BaseException extends Exception implements Serializable {
    @Serial
    private static final long serialVersionUID = -6522849858380060421L;

    private String errorCode;
    private String messageCode;
    private String[] messageValue;
    private String returnServiceId;
    private String debuggingMessage;
    private String errorPageUrl;
    private Map<String, Object> httpResponseHeaderMap;
    private String exceptionClassName;
    private String exceptionThrowingMethodName;
    private int exceptionThrowingLineNumber;
    private boolean useOriginTemplate;
    private boolean isUseOriginTemplateSetByUser;
    private Map<String, Object> exceptionCommon;
    private Map<String, Object> exceptionCause;
    private boolean ignoreMetaReturnServiceId;
    private boolean doNotWriteErrorLog;

    public static enum TYPE {
        COMMON, WEB, BIZ, USER, AUTH, SERVICE, CRITICAL;
    }

    public BaseException() {
        this.useOriginTemplate = true;
        this.isUseOriginTemplateSetByUser = false;
        this.ignoreMetaReturnServiceId = false;
        this.doNotWriteErrorLog = false;

        setMessageCode(null);
        setErrorCode(null);
        this.messageValue = null;
        getClass(this);
    }

    public BaseException(String _errorCode, Throwable _e) {
        this(_errorCode, _errorCode, null, _e);
    }

    public BaseException(String _errorCode) {
        this(_errorCode, _errorCode, null);
    }

    public BaseException(String _errorCode, String[] _messageValue, Throwable _e) {
        this(_errorCode, _errorCode, _messageValue, _e);
    }

    public BaseException(String _errorCode, String[] _messageValue) {
        this(_errorCode, _errorCode, _messageValue);
    }

    public BaseException(String _errorCode, String _messageCode, String[] _messageValue, Throwable _e) {
        super(_e);

        this.useOriginTemplate = true;
        this.isUseOriginTemplateSetByUser = false;
        this.ignoreMetaReturnServiceId = false;
        this.doNotWriteErrorLog = false;

        setMessageCode(_messageCode);
        setErrorCode(_errorCode);
        setMessageValue(_messageValue);
        getClass(_e);
    }

    public BaseException(String _errorCode, String _messageCode, String[] _messageValue) {
        this.useOriginTemplate = true;
        this.isUseOriginTemplateSetByUser = false;
        this.ignoreMetaReturnServiceId = false;
        this.doNotWriteErrorLog = false;
        setMessageCode(_messageCode);
        setErrorCode(_errorCode);
        setMessageValue(_messageValue);
        getClass(this);
    }

    private void getClass(Throwable e) {
        StackTraceElement[] se = e.getStackTrace();
        if ((se != null) && (se.length > 0)) {
            this.exceptionClassName = se[0].getClassName();
            this.exceptionThrowingMethodName = se[0].getMethodName();
            this.exceptionThrowingLineNumber = se[0].getLineNumber();
        }
    }

    public BaseException setDebuggingMessage(String debuggingMessage) {
        this.debuggingMessage = debuggingMessage;
        return this;
    }

    public TYPE getType() {
        return TYPE.COMMON;
    }

    public String toString() {
        return super.getClass().getName() + "  [errorCode=" + this.errorCode + ", messageCode=" + this.messageCode + ", messageValue=" + Arrays.toString(this.messageValue) + ", debuggingMessage=" + this.debuggingMessage + ", exceptionClassName=" + this.exceptionClassName + ", exceptionThrowingMethodName=" + this.exceptionThrowingMethodName + ", exceptionThrowingLineNumber=" + this.exceptionThrowingLineNumber + "]";
    }

    public void setUseOriginTemplate(boolean useOriginTemplate) {
        this.useOriginTemplate = useOriginTemplate;
        this.isUseOriginTemplateSetByUser = true;
    }

    public void setMessageValue(String[] messageValue) {
    	if (messageValue == null) {
    		this.messageValue = null;
    	} else {
    		this.messageValue = messageValue.clone();
    	}
    	
    };

    public String[] getMessageValue() {
    	if (this.messageValue == null) {
    		return null;
    	} else {
    		return this.messageValue.clone();
    	}
    	
    };
    
}

package kr.fasol.obix.core.filter;

import ch.qos.logback.classic.ClassicConstants;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import kr.fasol.obix.core.enums.LoggingConstant;
import kr.fasol.obix.core.util.GuidUtil;
import kr.fasol.obix.core.util.WebUtil;
import lombok.extern.slf4j.Slf4j;
import org.apache.commons.lang3.StringUtils;
import org.apache.logging.log4j.ThreadContext;
import org.slf4j.MDC;
import org.springframework.http.MediaType;
import org.springframework.web.filter.OncePerRequestFilter;
import org.springframework.web.util.ContentCachingResponseWrapper;

import java.io.IOException;
import java.nio.charset.Charset;
import java.util.Map;
import java.util.UUID;
import java.util.stream.Collectors;

/**
 * 로깅을 위한 Filter
 * HTTP 요청 및 응답을 로깅하여 추적 ID와 GUID를 설정합니다.
 * 요청 본문과 응답 본문을 로깅합니다.
 *
 * @author 정영탁
 */
@Slf4j
public class LoggingFilter extends OncePerRequestFilter {

    /**
     * HTTP 요청을 필터링하여 로깅을 설정합니다.
     *
     * @param request  HTTP 요청 객체
     * @param response HTTP 응답 객체
     * @param filterChain 필터 체인 객체
     * @throws ServletException 서블릿 예외
     * @throws IOException 입출력 예외
     */
    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain) throws ServletException, IOException {
        MDC.put(LoggingConstant.TRACE_ID, UUID.randomUUID().toString());

        // 로그에 GUID를 남기기 위한 설정
        String guid = GuidUtil.createGuid();
        MDC.put(LoggingConstant.REQUEST_GUID, guid);

        insertIntoMDC(request);

        // RequestMapping 정보 조회
        checkRequestMappingInfo(request);

        // log4j 로깅을 위한 설정
        ThreadContext.put(LoggingConstant.REQUEST_GUID, guid);
        ThreadContext.put(LoggingConstant.REQUEST_REMOTE_HOST, WebUtil.getClientIp(request));

        doFilterWrapped(request, response, filterChain);
        MDC.clear();
    }

    /**
     * 요청과 응답을 래핑하여 필터링합니다.
     *
     * @param request  HTTP 요청 객체
     * @param response HTTP 응답 객체
     * @param filterChain 필터 체인 객체
     * @throws IOException 입출력 예외
     * @throws ServletException 서블릿 예외
     */
    public void doFilterWrapped(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain) throws IOException, ServletException {
        MyRequestWrapper requestWrapper = new MyRequestWrapper(request);
        ContentCachingResponseWrapper responseWrapper = new ContentCachingResponseWrapper(response);
        try {
            logRequest(requestWrapper);
            doFilter(requestWrapper, responseWrapper, filterChain);
        } finally {
            logResponse(responseWrapper);
            responseWrapper.copyBodyToResponse();
        }
    }

    /**
     * HTTP 요청을 로깅합니다.
     *
     * @param requestWrapper 래핑된 HTTP 요청 객체
     * @throws IOException 입출력 예외
     */
    public void logRequest(MyRequestWrapper requestWrapper) throws IOException {
        String queryString = requestWrapper.getQueryString();
        String prefix = "Request";
        log.info("Request : {} uri=[{}] content-type=[{}]",
                requestWrapper.getMethod(),
                queryString == null ? requestWrapper.getRequestURI() : requestWrapper.getRequestURI() + "?" + queryString,
                requestWrapper.getContentType()
        );
        if (requestWrapper.getContentType() == null) {
            return;
        }
        if (needBodyPayload(requestWrapper.getContentType())) {
            String body = new String(requestWrapper.getRawData(), Charset.forName(requestWrapper.getCharacterEncoding()));
            logPayload(prefix, body);
        } else {
            logPayload(prefix, requestWrapper.getParameterMap());
        }
    }

    /**
     * HTTP 응답을 로깅합니다.
     *
     * @param response 래핑된 HTTP 응답 객체
     * @throws IOException 입출력 예외
     */
    public void logResponse(ContentCachingResponseWrapper response) throws IOException {
        //logPayload("Response", response.getContentInputStream());
    }

    /**
     * 요청 본문을 로깅할 필요가 있는지 확인합니다.
     *
     * @param contentType 콘텐츠 타입
     * @return 요청 본문을 로깅할 필요가 있는지 여부
     */
    private boolean needBodyPayload(String contentType) {
        return (contentType.equalsIgnoreCase(MediaType.APPLICATION_JSON_VALUE) ||
                contentType.equalsIgnoreCase(MediaType.APPLICATION_JSON_UTF8_VALUE) ||
                contentType.equalsIgnoreCase(MediaType.APPLICATION_XML_VALUE));
    }

    /**
     * 요청 또는 응답의 페이로드를 로깅합니다.
     *
     * @param prefix 로깅할 페이로드의 접두사
     * @param contentString 로깅할 페이로드 문자열
     * @throws IOException 입출력 예외
     */
    private void logPayload(String prefix, String contentString) throws IOException {
        log.info("{} Payload: {}", prefix, contentString);
    }

    /**
     * 요청 또는 응답의 파라미터를 로깅합니다.
     *
     * @param prefix 로깅할 파라미터의 접두사
     * @param parameter 로깅할 파라미터 맵
     * @throws IOException 입출력 예외
     */
    private void logPayload(String prefix, Map<String,String[]> parameter) throws IOException {
        log.info("{} Payload: {}", prefix, getParameterString(parameter));
    }

    /**
     * 파라미터 맵을 문자열로 변환합니다.
     *
     * @param parameter 파라미터 맵
     * @return 파라미터 문자열
     */
    private String getParameterString(Map<String,String[]> parameter){
        return parameter.entrySet().stream()
                .map(e -> "["+e.getKey() + " : " + String.join(", ",e.getValue())+"]")
                .collect(Collectors.joining(", "));
    }

    /**
     * MDC에 요청 정보를 삽입합니다.
     *
     * @param request HTTP 요청 객체
     */
    private void insertIntoMDC(HttpServletRequest request) {
        MDC.put(ClassicConstants.REQUEST_REMOTE_HOST_MDC_KEY, WebUtil.getClientIp(request));

        MDC.put(ClassicConstants.REQUEST_REQUEST_URI, request.getRequestURI());
        StringBuffer requestURL = request.getRequestURL();
        if (requestURL != null) {
            MDC.put(ClassicConstants.REQUEST_REQUEST_URL, requestURL.toString());
        }

            serviceType = pathWithinHandlerMapping.substring(pathWithinHandlerMapping.lastIndexOf('.') + 1);
        } else {
            serviceId = pathWithinHandlerMapping;
        }
        MDC.put(LoggingConstant.SERVICE_ID, serviceId);
        MDC.put(LoggingConstant.SERVICE_TYPE, serviceType);
        request.setAttribute(LoggingConstant.SERVICE_ID, serviceId);
        request.setAttribute(LoggingConstant.SERVICE_TYPE, serviceType);

/**
 * MyRequestWrapper 클래스는 HttpServletRequest를 래핑하여 XSS 공격을 방지하기 위한 기능을 제공합니다.
 * @author 정영탁
 */
public class MyRequestWrapper extends HttpServletRequestWrapper {
    private byte[] rawData;
    private HttpServletRequest request;
    private ResettableServletInputStream servletStream;

    public MyRequestWrapper(HttpServletRequest request) {
        super(request);
        this.request = request;
        this.servletStream = new ResettableServletInputStream();
    }

    public void resetInputStream(byte[] data) {
        servletStream.stream = new ByteArrayInputStream(data);
    }

    @Override
    public ServletInputStream getInputStream() throws IOException {
        if (rawData == null) {
            rawData = IOUtils.toByteArray(this.request.getReader());
            servletStream.stream = new ByteArrayInputStream(rawData);
        }
        return servletStream;
    }

    @Override
    public BufferedReader getReader() throws IOException {
        if (rawData == null) {
            rawData = IOUtils.toByteArray(this.request.getReader());
            servletStream.stream = new ByteArrayInputStream(rawData);
        }
        return new BufferedReader(new InputStreamReader(servletStream));
    }

    @Override
    public String[] getParameterValues(String name) {
        String[] values = super.getParameterValues(name);
        if (values == null) {
            return null;
        }
        int count = values.length;
        String[] encodedValues = new String[count];
        for (int i = 0; i < count; i++) {
            encodedValues[i] = cleanXSS(values[i]);
        }
        return encodedValues;
    }

    @Override
    public String getParameter(String name) {
        String value = super.getParameter(name);
        if (value == null) {
            return null;
        }
        return cleanXSS(value);
    }

    public byte[] getRawData() throws IOException {
        if (rawData == null) {
            rawData = IOUtils.toByteArray(this.request.getReader());
            servletStream.stream = new ByteArrayInputStream(rawData);
        }
        return rawData;
    }

    private String cleanXSS(String value) {
        return XssUtil.escapeXss(value);
    }

    /**
     * ResettableServletInputStream 클래스는 ServletInputStream을 재설정할 수 있는 기능을 제공합니다.
     */
    private static class ResettableServletInputStream extends ServletInputStream {
        private InputStream stream;

        @Override
        public int read() throws IOException {
            return stream.read();
        }
        @Override
        public boolean isFinished() {
            return false;
        }
        @Override
        public boolean isReady() {
            return false;
        }
        @Override
        public void setReadListener(ReadListener listener) {
        }
    }

}

/**
 * 시스템명 :
 * 업무구분명 : 공통 > Filter
 * 프로그램명 : XSS(크로스 사이트 스크립트) 필터링
 * 설명 : HTTP 요청에서 JSON 데이터를 필터링하여 XSS 공격을 방지하는 필터입니다.
 *
 * 클래스 : XssProtectFilter.java
 * 작성자 : 정영탁
 * 작성일 : 2022. 1. 3.
 * --------------------------------------------------------------------
 * Modification Information
 * --------------------------------------------------------------------
 * 수정일               수정자            수정내용
 * --------------------------------------------------------------------
 * 2022. 1. 3.         정영탁          최초생성
 */
@Slf4j
public class XssProtectFilter extends OncePerRequestFilter {

	/**
	 * HTTP 요청을 필터링하여 JSON 데이터에서 XSS 공격을 방지합니다.
	 *
	 * @param request  HTTP 요청 객체
	 * @param response HTTP 응답 객체
	 * @param filterChain 필터 체인 객체
	 * @throws ServletException 서블릿 예외
	 * @throws IOException 입출력 예외
	 */
	@Override
	protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain) throws ServletException, IOException {
		MyRequestWrapper requestWrapper = new MyRequestWrapper(request);

		// 요청의 Content-Type이 application/json이 아닌 경우 필터를 통과시킵니다.
		if (requestWrapper.getContentType() == null || !requestWrapper.getContentType().contains("application/json")) {
			filterChain.doFilter(requestWrapper, response);
			return;
		}

		// 요청 본문을 문자열로 변환합니다.
		String body = IOUtils.toString(requestWrapper.getInputStream(), Charset.forName(requestWrapper.getCharacterEncoding()));
		if (StringUtils.isNotBlank(body)) {
			// JSON 문자열을 Map 객체로 변환합니다.
			Map<String, Object> oldJsonObject = JsonUtil.JsonToObject(body, new TypeReference<Map<String, Object>>() {});
			Map<String, Object> newJsonObject = new HashMap<>();

			// JSON 데이터에서 문자열 값을 XSS 필터링합니다.
			oldJsonObject.forEach((key, value) -> {
				if (value instanceof String) {
					newJsonObject.put(key, XssUtil.escapeXss((String) value));
				} else {
					newJsonObject.put(key, value);
				}
			});

			// 필터링된 JSON 데이터를 요청 본문에 다시 설정합니다.
			requestWrapper.resetInputStream(JsonUtil.ObjectToJson(newJsonObject).getBytes());
		}

		// 필터 체인을 통해 요청을 계속 처리합니다.
		filterChain.doFilter(requestWrapper, response);
	}
}

    private static final String DEFAULT_EXCEPTION_ATTRIBUTE = "exception";

     */
    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ModelAndView handle(HttpServletRequest request, HttpServletResponse response, HttpMessageNotReadableException ex) {
        UserException exception = new UserException(CodeEnum.VALIDATION_FAILED.getCode());
        exception.setDebuggingMessage("Required request body is missing");

    }
    @ExceptionHandler(MissingServletRequestParameterException.class)
    public ModelAndView handle(HttpServletRequest request, HttpServletResponse response, MissingServletRequestParameterException ex) {
        UserException exception = new UserException(CodeEnum.VALIDATION_FAILED.getCode());
        exception.setDebuggingMessage(ex.getMessage());
        return doResolveException(request, response, null, exception);
    }

    /**
     * 404 NOT_FOUND에 대한 Exception Handler
     */
    @ExceptionHandler(NoHandlerFoundException.class)
    public ModelAndView handle(HttpServletRequest request, HttpServletResponse response, NoHandlerFoundException ex) {
        UserException exception = new UserException(CodeEnum.NOT_EXISTS.getCode());

    /**
     * Exception 발생시 수행
     */
    @Override
    public ModelAndView resolveException(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) {
        if (shouldApplyTo(request, handler)) {
            logException(ex, request);
            prepareResponse(ex, response);
            return doResolveException(request, response, handler, ex);
        }

        return null;
    }

    /**
     * Exception Message 결과를 response.header에 셋팅 후, View에 반환
     */
    protected ModelAndView doResolveException(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) {

        ExceptionConstant.ERROR_TYPE errorType = findErrorType(ex);
    	log.error("[" + errorType + "][" + ex.getClass().getName() + "]\n" + ex.toString(), ex);

        // viewName name of the View to render
        String viewName = determineViewName(ex, request);

        String msg = "";
        String errorCode = "";
        String orgErrorCode = "";
        String orgErrorMessage = "";
        String convertedErrorCode = "";
        String convertedErrorMsg = "";
        String debuggingMessage = "";
        //String additionalInfo = "";
        String returnServiceId = "";

        ExceptionResultVO exceptionResultVO = new ExceptionResultVO();
        exceptionResultVO.setException(ex);
        exceptionResultVO.setErrorViewName(viewName);
        exceptionResultVO.setBaseBizClass(this.baseBizClass);

        try {
            /*
             * 에러 메시지 정의
             */
            if (ex instanceof UserException userException) {
				errorCode = userException.getErrorCode();
                orgErrorCode = (userException.getOrgErrorCode() != null ? userException.getOrgErrorCode() : userException.getErrorCode());
                orgErrorMessage = userException.getOrgErrorMessage();
                debuggingMessage = userException.getDebuggingMessage();
                returnServiceId = userException.getReturnServiceId();
            } else if (ex instanceof CriticalException criticalException) {
				errorCode = criticalException.getErrorCode();
                orgErrorCode = (criticalException.getOrgErrorCode() != null ? criticalException.getOrgErrorCode() : criticalException.getErrorCode());
                orgErrorMessage = criticalException.getOrgErrorMessage();
                exceptionResultVO.setOrgErrorType(criticalException.getOrgErrorType());
                debuggingMessage = criticalException.getDebuggingMessage();
                returnServiceId = criticalException.getReturnServiceId();
            } else if (ex instanceof ChannelException channelException) {
                errorCode = channelException.getCode();
                orgErrorMessage = channelException.getDetailMessage();
                debuggingMessage = channelException.getDebuggingMessage();
                returnServiceId = channelException.getReturnServiceId();
            } else if (ex instanceof BaseException baseException) {
				errorCode = baseException.getErrorCode();
                orgErrorCode = baseException.getErrorCode();
                debuggingMessage = baseException.getDebuggingMessage();
                returnServiceId = baseException.getReturnServiceId();
            } else if (ex instanceof RetryableException retryableException) {
                // Feign Service의 Exception에 대한 처리
				exceptionResultVO.setOrgErrorMessage(retryableException.getMessage());
                errorCode = CodeEnum.API_SERVER_FAIL.getCode();
            } else if (ex instanceof FeignException feignException) {
                // Feign Service의 Exception에 대한 처리
				exceptionResultVO.setOrgErrorMessage(feignException.getMessage());
                errorCode = CodeEnum.API_SERVER_FAIL.getCode();
            } else if (ex instanceof HttpMessageNotReadableException || ex instanceof MissingServletRequestParameterException) {
                errorCode = CodeEnum.VALIDATION_FAILED.getCode();
            	debuggingMessage = ex.getMessage();
            } else if (ex instanceof HttpRequestMethodNotSupportedException || ex instanceof HttpMediaTypeNotAcceptableException || ex instanceof NoHandlerFoundException) {
                errorCode = CodeEnum.STATUS_BAD_REQUEST.getCode();
            	debuggingMessage = ex.getMessage();
            } else if (ex instanceof TransactionSystemException transactionSystemException) {
				if (transactionSystemException.contains(QueryTimeoutException.class)) {
	            	errorCode = CodeEnum.DB_TIMEOUT.getCode();
            	} else {
            		// 일반 SQLException은 CriticalException에서 제외
	            	errorCode = CodeEnum.DB_UNKNOWN.getCode();
                    errorType = ExceptionConstant.ERROR_TYPE.SQL;
            	}
            	exceptionResultVO.setOrgErrorType("DB");
                debuggingMessage = ex.getMessage();
            }
            else if (CriticalExceptionType.getTypeByException(ex) != null) {
            	CriticalExceptionType criticalExceptionType = CriticalExceptionType.getTypeByException(ex);
            	errorCode = criticalExceptionType.getErrorCode();
            	debuggingMessage = criticalExceptionType.getDebuggingMessage();
                exceptionResultVO.setOrgErrorType(criticalExceptionType.getOrgErrorType());
            }
            else {
                errorCode = CodeEnum.STATUS_UNKNOWN_ERROR.getCode();
                orgErrorMessage = ex.getMessage();
            }

            exceptionResultVO.setErrorCode(errorCode);
            exceptionResultVO.setConvertedErrorCode(convertedErrorCode);
            exceptionResultVO.setOrgErrorCode(orgErrorCode);
            exceptionResultVO.setReturnServiceId(returnServiceId);

            if (msg.isEmpty()) {
                msg = findErrorMessage(request, ex, errorCode, orgErrorMessage);
            }
            exceptionResultVO.setErrorMessage(msg);
            exceptionResultVO.setOrgErrorMessage(orgErrorMessage);
            convertedErrorMsg = msg;
            exceptionResultVO.setConvertedErrorMessage(convertedErrorMsg);
            exceptionResultVO.setErrorType(errorType.getErrorType());

            /*
             * EMS 연동을 위한 Critical Error 출력
             */
            if (errorType == ExceptionConstant.ERROR_TYPE.CRITICAL) {
            	criticalLog.error("{} [{}]{}", exceptionResultVO.getOrgErrorType(), exceptionResultVO.getErrorCode(), exceptionResultVO.getErrorMessage());
            }

            // ExceptionResultVO에 프로그램정보 추가
            exceptionResultVO = this.additionalErrorInfomationMaker.setAdditionalErrorInfo(request, ex, exceptionResultVO);

            request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_CODE.getKeyName(), errorCode);
            request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.CONVERTED_ERROR_CODE.getKeyName(), convertedErrorCode);
            request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ORGIN_ERROR_CODE.getKeyName(), orgErrorCode);
            request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.CONVERTED_ERROR_MESSAGE.getKeyName(), convertedErrorMsg);
            request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_MESSAGE.getKeyName(), convertedErrorMsg);
            request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_TYPE.getKeyName(), errorType);
            //request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_PROGRAM_NAME.getKeyName(), exceptionResultVO.getProgramName());
            //request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_PROGRAM_LINE.getKeyName(), exceptionResultVO.getProgramLine());
            request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_SERVER_INFO.getKeyName(), exceptionResultVO.getServerInfo());
            request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_RETURN_SERVICE_ID.getKeyName(), exceptionResultVO.getReturnServiceId());
            if (!ProfileUtil.isProduction()) {
                request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_GUID.getKeyName(), exceptionResultVO.getErrorGuid());
            }

            if (log.isErrorEnabled()) {
                StringBuilder logBuilder = new StringBuilder();
                logBuilder.append('\n').append("\tExceptionResultVO  ErrorCode : ").append(exceptionResultVO.getErrorCode());
                logBuilder.append('\n').append("\tExceptionResultVO  ErrorMessage : ").append(exceptionResultVO.getErrorMessage());
                logBuilder.append('\n').append("\tExceptionResultVO  ErrorCodeChanged : ").append(exceptionResultVO.isErrorCodeChanged());
                logBuilder.append('\n').append("\tExceptionResultVO  ConvertedErrorCode : ").append(exceptionResultVO.getConvertedErrorCode());
                logBuilder.append('\n').append("\tExceptionResultVO  ConvertedErrorMessage : ").append(exceptionResultVO.getConvertedErrorMessage());
                logBuilder.append('\n').append("\tExceptionResultVO  OrgErrorCode : ").append(exceptionResultVO.getOrgErrorCode());
                logBuilder.append('\n').append("\tExceptionResultVO  OrgErrorMessage : ").append(exceptionResultVO.getOrgErrorMessage());
                logBuilder.append('\n').append("\tExceptionResultVO  ErrorGuid : ").append(exceptionResultVO.getErrorGuid());
                logBuilder.append('\n').append("\tExceptionResultVO  ProgramName : ").append(exceptionResultVO.getProgramName());
                logBuilder.append('\n').append("\tExceptionResultVO  ProgramLine : ").append(exceptionResultVO.getProgramLine());
                logBuilder.append('\n').append("\tExceptionResultVO  ServerInfo : ").append(exceptionResultVO.getServerInfo());
                logBuilder.append('\n').append("\tExceptionResultVO  DebuggingMessage : ").append(debuggingMessage);
                logBuilder.append('\n').append("\tExceptionResultVO  ReturnServiceId : ").append(exceptionResultVO.getReturnServiceId());
                log.error(logBuilder.toString());
            }

            /*
             * RestController일 경우, 오류처리
             */
            if (handler instanceof HandlerMethod handlerMethod) {
                if (getAnnotation(handlerMethod, RestController.class) != null || getAnnotation(handlerMethod, ResponseBody.class) != null) {
                    log.error("### RestController일 경우, 오류 처리");
                    return new DefaultExceptionOutputMaker().makeOutput(request, response, exceptionResultVO);
                }
            }

            // view 모드가 아닐 경우
            if (Constants.NOTVIEW_SERVICE_TYPE.contains(Constants.CONST_SERVICE_TYPE.getTypeByCode(MDC.get(LoggingConstant.SERVICE_TYPE)))) {
                log.error("### view 모드가 아닐 경우, 오류 처리");
                DefaultExceptionOutputMaker outputMaker = new DefaultExceptionOutputMaker();
                return (ModelAndView) outputMaker.makeOutput(request, response, exceptionResultVO);
            }

        } catch (Exception er) {
            log.error("Critical Exception", er);
            try {
                response.sendError(500);

                response.setHeader(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_PAGE.getKeyName(), errorCode);
                response.setHeader(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_MESSAGE.getKeyName(), URLEncoder.encode(msg, "UTF-8"));
                request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_CODE.getKeyName(), errorCode);
                request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_MESSAGE.getKeyName(), msg);
            } catch (Exception eeee) {
                log.error("Exception Processor dual error", eeee);
            }
        } catch (Error er) {
            log.error("Critical ERROR ", er);
            try {
                response.sendError(500);

                response.setHeader(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_PAGE.getKeyName(), errorCode);
                response.setHeader(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_MESSAGE.getKeyName(), URLEncoder.encode(msg, "UTF-8"));
                request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_CODE.getKeyName(), errorCode);
                request.setAttribute(ExceptionConstant.ERROR_PROPERTIES_KEY.ERROR_MESSAGE.getKeyName(), msg);
            } catch (Exception eeee) {
                log.error("Exception Processor dual error", eeee);
            }
        }

        if (viewName != null) {
            Integer statusCode = determineStatusCode(request, viewName);
            if (statusCode != null) {
                applyStatusCodeIfPossible(request, response, statusCode.intValue());
            }
            log.error("### view 모드일 경우, 오류 처리");
            return getModelAndView(viewName, ex, request);
        }

        return null;
    }

    /**
     * Exception 에러 종류 반환
     */
    private ExceptionConstant.ERROR_TYPE findErrorType(Exception e) {
        if (e instanceof UserException) {
            return ExceptionConstant.ERROR_TYPE.USER;
        } else if (e instanceof CriticalException) {
            return ExceptionConstant.ERROR_TYPE.CRITICAL;	// CriticalException으로 분류
        } else if (e instanceof ChannelException) {
            return ExceptionConstant.ERROR_TYPE.CHANNEL;
        } else if (e instanceof RetryableException) {
            // Feign Service의 Exception에 대한 처리
            return ExceptionConstant.ERROR_TYPE.CHANNEL;
        } else if (e instanceof FeignException) {
            // Feign Service의 Exception에 대한 처리
            return ExceptionConstant.ERROR_TYPE.CHANNEL;
        } else if (e instanceof SQLException) {
            return ExceptionConstant.ERROR_TYPE.SQL;
        } else if (CriticalExceptionType.getTypeByException(e) != null) {
            return ExceptionConstant.ERROR_TYPE.CRITICAL;	// CriticalException으로 분류
        }
        return ExceptionConstant.ERROR_TYPE.SYSTEM;
    }

    protected String determineViewName(Exception ex, HttpServletRequest request) {
        String viewName = null;
        // Check for specific exception mappings.
        if (this.exceptionMappings != null) {
            viewName = findMatchingViewName(this.exceptionMappings, ex);
        }
        // Return default error view else, if defined.
        if (viewName == null && this.defaultErrorView != null) {
            viewName = this.defaultErrorView;
        }
        return viewName;
    }

    protected String findMatchingViewName(Properties exceptionMappings, Exception ex) {
        String viewName = null;
        String dominantMapping = null;
        int deepest = Integer.MAX_VALUE;
        for (Enumeration<?> names = exceptionMappings.propertyNames(); names.hasMoreElements(); ) {
            String exceptionMapping = (String) names.nextElement();
            int depth = getDepth(exceptionMapping, ex);
            if (depth >= 0 && (depth < deepest || (depth == deepest &&
                    dominantMapping != null && exceptionMapping.length() > dominantMapping.length()))) {
                deepest = depth;
                dominantMapping = exceptionMapping;
                viewName = exceptionMappings.getProperty(exceptionMapping);
            }
        }
        return viewName;
    }

    protected int getDepth(String exceptionMapping, Exception ex) {
        return getDepth(exceptionMapping, ex.getClass(), 0);
    }

    private int getDepth(String exceptionMapping, Class<?> exceptionClass, int depth) {
        if (exceptionClass.getName().contains(exceptionMapping)) {
            // Found it!
            return depth;
        }
        // If we've gone as far as we can go and haven't found it...
        if (exceptionClass == Throwable.class) {
            return -1;
        }
        return getDepth(exceptionMapping, exceptionClass.getSuperclass(), depth + 1);
    }

    protected Integer determineStatusCode(HttpServletRequest request, String viewName) {
        if (this.statusCodes.containsKey(viewName)) {
            return this.statusCodes.get(viewName);
        }
        return this.defaultStatusCode;
    }

    protected void applyStatusCodeIfPossible(HttpServletRequest request, HttpServletResponse response, int statusCode) {
        if (!WebUtils.isIncludeRequest(request)) {
            response.setStatus(statusCode);
            request.setAttribute(WebUtils.ERROR_STATUS_CODE_ATTRIBUTE, statusCode);
        }
    }

    protected ModelAndView getModelAndView(String viewName, Exception ex, HttpServletRequest request) {
        return getModelAndView(viewName, ex);
    }

    protected ModelAndView getModelAndView(String viewName, Exception ex) {
        ModelAndView mv = new ModelAndView(viewName);
        if (this.exceptionAttribute != null) {
            mv.addObject(this.exceptionAttribute, ex);
        }
        return mv;
    }

    public void setExceptionMappings(Properties exceptionMappings) {
        this.exceptionMappings = exceptionMappings;
    }
    public void setDefaultErrorView(String defaultErrorView) {
        this.defaultErrorView = defaultErrorView;
    }
    public void setDefaultStatusCode(Integer defaultStatusCode) {
        this.defaultStatusCode = defaultStatusCode;
    }
    public void setBaseBizClass(String baseBizClass) {
        this.baseBizClass = baseBizClass;
    }

    private <A extends Annotation> A getAnnotation(HandlerMethod handlerMethod, Class<A> annotationType) {
        if (handlerMethod == null || annotationType == null) {
            return null;
        }
        return Optional.ofNullable(handlerMethod.getMethodAnnotation(annotationType))
                .orElse(handlerMethod.getBeanType().getAnnotation(annotationType));
    }

    /**
     * Exception의 오류코드에 대한 오류메시지 반환
     */
    private String findErrorMessage(HttpServletRequest request, Exception e, String errorCode, String orgErrorMessage) {
        String msg = "";
        Object[] args = null;

        if (e instanceof UserException userException) {
			args = userException.getMessageValue();
        } else if (e instanceof CriticalException criticalException) {
			args = criticalException.getMessageValue();
        } else if (e instanceof ChannelException channelException) {
			args = channelException.getMessageValue();
        } else if (e instanceof BaseException baseException) {
            args = baseException.getMessageValue();
        }

        /*
         * 오류코드 조회
         * if (errorCode)
         *     AppMessageUtil (error_*.properties)에서 조회
         * else
         *     errorCode.toString
         */
        if (StringUtils.isNotBlank(errorCode)) {
	        final String errorMessage = AppMessageUtil.getMessage(errorCode, orgErrorMessage);
	        if (StringUtils.isNotBlank(errorMessage)) {
	            if (args != null && args.length > 0) {
	                msg = MessageFormat.format(errorMessage, args);
	            } else {
	                msg = errorMessage;
	            }
	        }
        }

        return msg;
    }

}

@Slf4j
public class DefaultAddionalErrorInformationMaker {

    private String serverInfo = "";

    public DefaultAddionalErrorInformationMaker() {
        String address = "";
        String containerName = null;
        try {
            address = InetAddress.getLocalHost().getHostAddress();
            if (address != null) {
                String[] adrs = address.split("\\.");
                if ((adrs != null) && (adrs.length > 0)) {
                    address = adrs[0] + "." + adrs[(adrs.length - 1)];
                }
            }
            containerName = WasUtil.getContainerName();
            if ((containerName != null) && (containerName.length() > 2)) {
                containerName = containerName.substring(containerName.length() - 2);
            }
        } catch (Exception e) {
            log.error("[DefaultAddionalErrorInformationMaker]", e);
        }

        if (StringUtils.isNotBlank(address)) {
            this.serverInfo = address;
        }

        if (containerName != null) {
            this.serverInfo = this.serverInfo + "-" + containerName;
        }
    }

    /**
     * Exception 정보에 프로그램정보 추가
     */
    public ExceptionResultVO setAdditionalErrorInfo(HttpServletRequest request, Exception e, ExceptionResultVO exceptionResultVO) {
        log.trace("START ADDITIONAL ERROR INFO");
        exceptionResultVO.setServerInfo(this.serverInfo);

    	StackTraceElement[] se = e.getStackTrace();
        if ((se != null) && (se.length > 0)) {
            exceptionResultVO.setOrgClassName(se[0].getClassName());
            exceptionResultVO.setOrgMethodName(se[0].getMethodName());
            exceptionResultVO.setOrgLineNumber(se[0].getLineNumber());
        	exceptionResultVO.setProgramLine(String.valueOf(se[0].getLineNumber()));
        }

        String lastClassName = "";
        int lastLineNumber = -1;
        String methodName = "";
        for (StackTraceElement st : se) {
        	if (st.getClassName().contains(exceptionResultVO.getBaseBizClass())) {
        		lastLineNumber = st.getLineNumber();
        		lastClassName = st.getClassName();
        		methodName = st.getMethodName();
        		log.trace("FIND ORIGIN BIZ CLASS : {}", lastClassName);
        		break;
        	}
        }

        exceptionResultVO.setBizClassName(lastClassName);
        exceptionResultVO.setBizLineNumber(lastLineNumber);
        exceptionResultVO.setBizMethodName(methodName);

        if (e instanceof BaseException) {
            exceptionResultVO.setExceptionCause(((BaseException) e).getExceptionCause());
            exceptionResultVO.setExceptionCommon(((BaseException) e).getExceptionCommon());
        }

        String serviceId = MDC.get(LoggingConstant.SERVICE_ID);
        String className = null;
        if (lastLineNumber != -1) {
        	className = exceptionResultVO.getBizClassName();
        } else {
        	className = StringUtils.trimToEmpty(exceptionResultVO.getOrgClassName());
        }
        String[] classname = className.split("\\.");
        String programName = serviceId + ":" + classname[(classname.length - 1)];
        exceptionResultVO.setProgramName(programName);

        log.trace("ProgramName : {}", exceptionResultVO.getProgramName());
        log.trace("ProgramLine : {}", exceptionResultVO.getProgramLine());
        log.trace("SERVER INFO : {}", exceptionResultVO.getServerInfo());
        log.trace("END ADDITIONAL ERROR INFO");

        return exceptionResultVO;
    }

}

                    .logLevel(Logger.Level.FULL)

    public Client client() {

                .setMaxConnPerRoute(300)
                .setMaxConnTotal(300)
                .addInterceptorFirst((HttpResponseInterceptor) (response, context) -> {

                    log.debug("Response Header GUID: {}", guid);
                    businessTierHeaderInfoProvider.setGuid(guid);
                });

        if (env.acceptsProfiles(Profiles.of(Constants.ENV_LOCAL, Constants.ENV_DEV))) {
            builder.disableContentCompression();

    public ErrorDecoder errorDecoder() {
        return (methodKey, response) -> {
            Response<?> commonResponse = null;
            try {
                final String body = IOUtils.toString(response.body().asInputStream(), StandardCharsets.UTF_8);
                commonResponse = gson.fromJson(body, Response.class);
            } catch (IOException e) {
                log.error("Error", e);
            }
            return new HttpClientApiException(commonResponse, response.status());
        };
    }

        return new GsonDecoder(gson);

        log.debug("request before send to bt = {}", template);

        Boolean verifySsl = env.getProperty("http.client.config.verify-ssl", Boolean.class, Boolean.TRUE);

        // Unirest 기본설정
        Config config = new Config();

        // SSL 인증서 무시 설정
        if (!verifySsl) {
            try {
                SSLContext sslContext = new SSLContextBuilder().loadTrustMaterial(null, new TrustSelfSignedStrategy() {
                    public boolean isTrusted(X509Certificate[] chain, String authType) {
                        return true;
                    }
                }).build();
                config.sslContext(sslContext);
                config.verifySsl(false);
            } catch (NoSuchAlgorithmException | KeyStoreException | KeyManagementException e) {
                log.error("SSL 인증서 무시 설정 중 오류가 발생했습니다.", e);
            }
        }

        primaryInstance = new UnirestInstance(config);
    }

    /**
     * Business Tier 호출
     */
    public BusinessTierClientBuilder businessTier(HttpClientConstant.Msa msa) {
        return new BusinessTierClientBuilder(primaryInstance, msa, gson, env, businessTierHeaderInfoProvider, readTimeout, connTimeout, proxyHost, proxyPort);
    }

    /**
     * 외부 URL 호출
     */
    public ExternalClientBuilder external(String externalUrl) {
        return new ExternalClientBuilder(primaryInstance, externalUrl, gson, env, businessTierHeaderInfoProvider, readTimeout, connTimeout, proxyHost, proxyPort);
    }

}
