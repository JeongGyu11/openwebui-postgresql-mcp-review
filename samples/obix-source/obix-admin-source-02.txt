	/**
	 * @desc 화면거래 미등록 URL 현황 조회
	 * @param ASYAU10101DTO
	 */
	public ASYAU10101DTO ASYAU10101(ASYAU10101DTO input) throws Exception {

		// 서버와 DB에 각각 등록된 url 리스트(+ 컨트롤러이름/화면거래아이디)
		List<UrlCompareDTO> serverSide = new ArrayList<>();
		List<UrlCompareDTO> clientSide = new ArrayList<>();

		// 서버 컨트롤러 매핑된 url 가져오기
		Map<RequestMappingInfo, HandlerMethod> urlHandlerMap = requestMappingHandlerMapping.getHandlerMethods();
		for (Map.Entry<RequestMappingInfo, HandlerMethod> element : urlHandlerMap.entrySet()) {
			String urlPattern = element.getKey().getPatternValues().stream().findFirst().get();
			String controllerName = element.getValue().getBeanType().getName();
			if (!StringUtils.containsAny(controllerName, ErrorPagesController.class.getName())) {
				String screnTranNm = StringUtils.EMPTY;

				// @Operation Annotation에 설정된 Description 조회
				Operation operation = (Operation) element.getValue().getMethod().getAnnotation(Operation.class);
				if (operation != null && StringUtils.isNotBlank(operation.summary())) {
					screnTranNm = operation.summary();
				}

				UrlCompareDTO urlInfo = UrlCompareDTO.builder()
						.urlPattern(urlPattern)
						.controllerName(controllerName)
						.screnTranNm(screnTranNm)
						.build();
				serverSide.add(urlInfo);
			}
		}

		// db 등록된 url 가져오기
		List<UrlCompareDTO> screnTranBasList = admScrenTranBasDAO.listValidAdmScrenTranBas();
		for (UrlCompareDTO element : screnTranBasList) {
			if ("Y".equals(element.getUseYn())) {
				clientSide.add(element);
			}
		}

		//컨트롤러-DB 일치하여 URL매핑이 잘 된 리스트
		List<UrlCompareDTO> mappedList = new ArrayList<>(); 
		for(UrlCompareDTO element : serverSide) {
			if(screnTranBasList.contains(element)) {
				mappedList.add(element);
			}
		}
		
		//서버.클라이언트 리스트에서 매핑 잘 된 리스트 삭제
		serverSide.removeAll(mappedList);		
		clientSide.removeAll(mappedList);
		serverSide.sort(null);

		return ASYAU10101DTO.builder()
				.serverOnlyList(serverSide)
				.clientOnlyList(clientSide)
				.build();
	}

				admGroupAthIn.setLastChngUserId(AdminSessionHelper.getAdminUserId());

				admGroupAthResult += admGroupAthDAO.mergeAdmGroupAth(admGroupAthIn);

	/**
	 * @desc 사용자 설정정보 저장
	 * @param ASYAU40102DTO
	 */
	public ASYAU40102DTO ASYAU40102(ASYAU40102DTO input) throws Exception {
		ASYAU40102DTO output = new ASYAU40102DTO();

		input.setUserPwd(ObixSecurityUtil.SHA256(input.getUserId()));

		// 등록자/수정자 기본값 설정
		input.setFrstRegiUserId(AdminSessionHelper.getAdminUserId());
		input.setLastChngUserId(AdminSessionHelper.getAdminUserId());

		//신규 유저 or 기존 유저 체크
    	AdmUserBasQryDTO res = admUserBasDAO.selectAdmUserBas(DTOUtil.convertDTO(input, AdmUserBasQryDTO.class));
    	//기존 유저
    	if(res != null) {
    		//사용자 설정정보 이력 저장
    		admUserBasDAO.insertAdmUserBasHst(DTOUtil.convertDTO(input, AdmUserBasDTO.class));
    		//사용자 그룹 설정정보 이력 저장
    		admGroupUserRelDAO.insertAdmGroupUserRelHst(DTOUtil.convertDTO(input, AdmGroupUserRelDTO.class));
    		
    		//상태 변경이 있을 시 상태 변경 내역 저장
    		if(!res.getAdmUserStusCd().equals(input.getAdmUserStusCd())) {
    			input.setBefUserStusCd(res.getAdmUserStusCd());
    			adminLoginDAO.insertAdmUserStusChngTxn(DTOUtil.convertDTO(input, AdmUserStusChngTxnDTO.class));
    		}
    		//기존 유저 UPDATE
    		admUserBasDAO.updateAdmUserBas(DTOUtil.convertDTO(input, AdmUserBasDTO.class));
    		
    		//기존 사용자 그룹 초기화
    		admGroupUserRelDAO.resetAdmGroupUserRel(DTOUtil.convertDTO(input, AdmGroupUserRelDTO.class));
    	//신규 유저
    	}else{
    		//신규 유저 INSERT
    		input.setAdmUserStusCd(CodeConstant.ADM_USER_STUS_CD.초기화.getCode());
    		admUserBasDAO.insertAdmUserBas(DTOUtil.convertDTO(input, AdmUserBasDTO.class));
    	}

    	List<AdmGroupUserRelDTO> list = input.getGroupList();
    	for(int i=0; i<list.size(); i++) {
    		list.get(i).setFrstRegiUserId(input.getFrstRegiUserId());
    		list.get(i).setLastChngUserId(input.getLastChngUserId());
    	}
    	Map<String, List<AdmGroupUserRelDTO>> map = new HashMap<String, List<AdmGroupUserRelDTO>>();
    	map.put("list", list);
    	
    	//사용자 그룹 머지
    	admGroupUserRelDAO.mergeBulkAdmGroupUserRel(map);

    	output.setResult("OK");
		return output;
	}

     * @param day - 0:당영업일, -1:전영업일, -2:전전영업일, 1:익영업일, 2:익익영업일
     * @return 영업일
     */
	public Date getBizDay(Date baseDt, int day) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);

		// 휴일목록 조회
		HoldyBasQryDTO req = new HoldyBasQryDTO();
		if (day < 0) {
			req.setStartDt(sdf.format(DateUtils.addMonths(baseDt, 0 - ((day / 15) + 1))));
			req.setEndDt(sdf.format(baseDt));
		} else {
			req.setStartDt(sdf.format(baseDt));
			req.setEndDt(sdf.format(DateUtils.addMonths(baseDt, ((day / 15) + 1))));
		}

		List<HoldyBasQryDTO> list = bizDayDAO.listHoldyBas(req);

		// 휴일 목록을 일자별 Map으로 변환
		Map<String, Object> bizDayMap = new HashMap<>();
		if (list != null && !list.isEmpty()) {
			bizDayMap = list.stream().collect(Collectors.toMap(HoldyBasQryDTO::getHoldyYmd, HoldyBasQryDTO::clone));
		}

		/*
		 * baseDt 기준 영업일 조회
		 */
		Calendar start = Calendar.getInstance();
		start.setTime(baseDt);

		if (day == 0) {
			while (true) {
				String targetDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(targetDt)) {
					break;
				}
				start.add(Calendar.DATE, 1);
			}
		} else {
			int bizDtCnt = 0;
			while (bizDtCnt < (day < 1 ? 0 - day : day)) {
				start.add(Calendar.DATE, (day < 1 ? -1 : 1));
				String targetDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(targetDt)) {
					bizDtCnt++;
				}
			}
		}

		return start.getTime();
	}

    /**
     * 기준일로 부터 영업일을 조회한다.
     * @param baseDt 기준일(yyyyMMdd)
     * @param day - 0:당영업일, -1:전영업일, -2:전전영업일, 1:익영업일, 2:익익영업일
     * @return 영업일(yyyyMMdd)
     */
	@Cacheable(cacheNames = "BIZDAY_LIST", key = "#baseDt + ':' + #day")
	public String getBizDay(String baseDt, int day) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);
		return sdf.format( getBizDay(sdf.parse(baseDt), day) );
	}


	/**
	 * 기준일로부터 대상일자가 몇 영업일인지 계산
	 * @param baseDt 기준일
	 * @param targetDt 대상일자
	 * @return - 0:당영업일, 1:익영업일, 2:익익영업일
	 */
	public int getBizDayDiff(Date baseDt, Date targetDt) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);

		// 휴일목록 조회
		HoldyBasQryDTO req = new HoldyBasQryDTO();
		req.setStartDt(sdf.format(baseDt));
		req.setEndDt(sdf.format(targetDt));

		List<HoldyBasQryDTO> list = bizDayDAO.listHoldyBas(req);

		// 휴일 목록을 일자별 Map으로 변환
		Map<String, Object> bizDayMap = new HashMap<>();
		if (list != null && !list.isEmpty()) {
			bizDayMap = list.stream().collect(Collectors.toMap(HoldyBasQryDTO::getHoldyYmd, HoldyBasQryDTO::clone));
		}


		int day = 0;
		if (!baseDt.equals(targetDt)) {
			// baseDt 기준 영업일 조회
			Calendar start = Calendar.getInstance();
			start.setTime(baseDt);

			while (true) {
				start.add(Calendar.DATE, 1);
				String diffDt = sdf.format(start.getTime());
				int week = start.get(Calendar.DAY_OF_WEEK);
				if (week != Calendar.SATURDAY && week != Calendar.SUNDAY && !bizDayMap.containsKey(diffDt)) {
					day++;
				}
				if (targetDt.equals(sdf.parse(diffDt))) {
					 break;
				}
			}
		}

		return day;
	}

	/**
	 * 기준일로부터 대상일자가 몇 영업일인지 계산
	 * @param baseDt 기준일(yyyyMMdd)
	 * @param targetDt 대상일자(yyyyMMdd)
	 * @return - 0:당영업일, 1:익영업일, 2:익익영업일
	 */
	public int getBizDayDiff(String baseDt, String targetDt) throws Exception {
		SimpleDateFormat sdf = new SimpleDateFormat(DateUtil.FORMATTER_YYYYMMDD);
		return getBizDayDiff(sdf.parse(baseDt), sdf.parse(targetDt));
	}

@Controller
@RequestMapping("${server.error.path:${error.path:/error}}")
public class ErrorPagesController extends AbstractErrorController {

	@Value("${server.error.path:${error.path:/error}}")
	private String ERROR_PATH;

	public ErrorPagesController(ErrorAttributes errorAttributes) {
		super(errorAttributes);
	}

	/**
	 * VIEW 페이지의 에러 정의
	 */
	@RequestMapping(produces = MediaType.TEXT_HTML_VALUE)
	public Object errorHtml(HttpServletRequest request, HttpServletResponse response) {

		/*
		 * Error 구분에 따른 분기
		 */
		String requestUri = (String) request.getAttribute(RequestDispatcher.ERROR_REQUEST_URI);
		if (StringUtils.trimToEmpty(requestUri).endsWith(".json")) {
			Response<?> errorResponse = error(request);
			// json 모드
			CustomMappingJackson2JsonView jsonView = new CustomMappingJackson2JsonView();
			jsonView.setAttributeResponse(errorResponse);
			return new ModelAndView(jsonView);
		}

		Object status = request.getAttribute(RequestDispatcher.ERROR_STATUS_CODE);
		if (status != null) {
			int statusCode = Integer.valueOf(status.toString());
			if (statusCode == HttpStatus.NOT_FOUND.value()) {
				response.setStatus(HttpStatus.OK.value());
				return "/error/404";
			} else if (statusCode == HttpStatus.FORBIDDEN.value()) {
				response.setStatus(HttpStatus.OK.value());
				return "/error/404";
			} else if (statusCode == HttpStatus.INTERNAL_SERVER_ERROR.value()) {
				response.setStatus(HttpStatus.OK.value());
				return "/error/500";
			}
		}
		return "/error/error";
	}

	/**
	 * VIEW 없는 페이지(API)의 에러 정의
	 */
	@RequestMapping
	public Response<?> error(HttpServletRequest request) {
		Object status = request.getAttribute(RequestDispatcher.ERROR_STATUS_CODE);
		if (status != null) {
			int statusCode = Integer.valueOf(status.toString());
			if (statusCode == HttpStatus.NOT_FOUND.value()) {
				return Response.builder().statusAndMessage(CodeEnum.NOT_EXISTS.getCode(), HttpStatus.NOT_FOUND.getReasonPhrase()).build();
			} else if (statusCode == HttpStatus.FORBIDDEN.value()) {
				return Response.builder().statusAndMessage(CodeEnum.NOT_EXISTS.getCode(), HttpStatus.FORBIDDEN.getReasonPhrase()).build();
			} else if (statusCode == HttpStatus.INTERNAL_SERVER_ERROR.value()) {
				Exception exception = (Exception) request.getAttribute(RequestDispatcher.ERROR_EXCEPTION);
				if (exception instanceof ServletException && ((ServletException) exception).getRootCause() instanceof UserException) {
					UserException userException = (UserException) ((ServletException) exception).getRootCause();
					String errorMessage = AppMessageUtil.getMessage(userException.getErrorCode());
					Object[] errorMessageValue = userException.getMessageValue();
					if (errorMessageValue != null && errorMessageValue.length > 0) {
						errorMessage = MessageFormat.format(errorMessage, errorMessageValue);
					}
					return Response.builder().statusAndMessage(userException.getErrorCode(), errorMessage).build();
				}
				return Response.builder().statusAndMessage(CodeEnum.NOT_EXISTS.getCode(), HttpStatus.INTERNAL_SERVER_ERROR.getReasonPhrase()).build();
			}
		}
		return Response.builder().statusAndMessage(CodeEnum.STATUS_BAD_REQUEST.getCode(), HttpStatus.BAD_REQUEST.getReasonPhrase()).build();
	}

	public String getErrorPath() {
		return ERROR_PATH;
	}

}

            String bizUrlLoginService = EnvironmentUtil.getProperty("biz.url.login.service");
            String requestUri = request.getRequestURI();
        	String requestParam = new Gson().toJson(WebUtil.parameterMap(request));
        	String redirectUrl = String.format("%s.%s", bizUrlLoginService, Constants.CONST_SERVICE_TYPE.VIEW.getType());

        	if (log.isDebugEnabled()) {
                log.debug("로그인 되어 있지 않아 현재 URL을 저장하고 로그인 화면으로 이동합니다.");
                log.debug("SERVICE_ID : {}, SERVICE_TYPE : {}", serviceId, serviceType);
                log.debug("biz-url.login.service : {}", bizUrlLoginService);
                log.debug("접근했던 URL: {}", requestUri);
                log.debug("접근했던 URL의 파라미터: {}", requestParam);
            }

			/*
			 * ServiceType이 (FRAG, JSON)이면 REQUIRED_LOGIN 에러로 반환
			 */
            if (Constants.CONST_SERVICE_TYPE.FRAG.getType().equals(serviceType) || Constants.CONST_SERVICE_TYPE.JSON.getType().equals(serviceType)) {
                AuthenticationException authenticationException = new AuthenticationException("REQUIRED_LOGIN");
                authenticationException.setDebuggingMessage("[AdminRequiredLoginInterceptor][REQUIRED_LOGIN]");
                authenticationException.setReturnServiceId(bizUrlLoginService);
                throw authenticationException;
            } else {
                // 로그인 후 돌아올 URL을 세션에 저장
            	SessionUtil.put(SessionEnum.NEXT_USER_URL, requestUri);
            	SessionUtil.put(SessionEnum.NEXT_USER_URL_PARAM, requestParam);

                log.debug("sendRedirect : {}", redirectUrl);
				response.sendRedirect(redirectUrl);
            }

            return false;
        }

        return true;
    }

 * 세부업무구분명 : 공통 > Custom Tag > 날짜 포멧변경 출력
 * 설명 :
 *
 * ex)
 * <u:formatDate infmt="yyyyMMdd" outfmt="yyyy.MM.dd" value="20071012"/>
 *
 * <b>Attributes</b>
 * - infmt        : 입력 포멧(기본값: yyyyMMdd)
 * - outfmt       : 출력 포멧(기본값: yyyy.MM.dd)
 * - value        : 입력 값
 *
 *

@Slf4j
public class FormatDateTag extends TagSupport {

    private HttpServletRequest request;
    private String var;
    private int scope;
    private Object value;
    private String infmt;
    private String outfmt;

    public FormatDateTag() {
        this.var = null;
        this.scope = 1;
    }

    @Override
    public int doEndTag() throws JspException {
        JspWriter out = pageContext.getOut();
        request = (HttpServletRequest) pageContext.getRequest();

		try {
			String result = "";

			String inputFmt = StringUtils.defaultIfBlank(this.infmt, "yyyyMMdd");
			String outputFmt = StringUtils.defaultIfBlank(this.outfmt, "yyyy.MM.dd");

			/*
			 * 입력값을 outfmt의 형식으로 변환
			 */
			if (this.value == null) {
				result = "";
			} else if (this.value instanceof Date) {
				result = DateUtil.getString((Date) this.value, outputFmt);
			} else {
				result = (String) this.value;

				if (StringUtils.isBlank(result)) {
					result = StringUtils.EMPTY;
				} else {
					try {
						SimpleDateFormat sdf = new SimpleDateFormat(inputFmt);
						Date d = sdf.parse(result);
						sdf.applyPattern(outputFmt);
						result = sdf.format(d);
					} catch (ParseException e) {
						result = (String) this.value;
					}
				}
			}

            /*
             * Output에 결과 전송
             */
            if (this.var != null) {
                if (this.scope == 1) {
                    VariableMapper vm = this.pageContext.getELContext().getVariableMapper();
                    if (vm != null) {
                        vm.setVariable(this.var, null);
                    }
                }
                this.pageContext.setAttribute(this.var, result, this.scope);
            } else {
                out.append(result);
            }

        } catch(Exception e) {
            throw new JspException(e);
        }
        return EVAL_PAGE;
    }

    }

        AdmScrenTranBasQryDTO screnTranDTO = menuList.stream().filter(x -> StringUtils.trimToEmpty(x.getUrlAddr()).contains(screnTranId)).findFirst().orElse(null);
        if (screnTranDTO != null) {
            locationList.add((AdmScrenTranBasQryDTO) screnTranDTO.clone());
            while (screnTranDTO != null && StringUtils.isNotBlank(screnTranDTO.getUpScrenTranId())) {
                final String upScrenTranId = screnTranDTO.getUpScrenTranId();
                screnTranDTO = menuList.stream().filter(x -> x.getScrenTranId().equals(upScrenTranId)).findFirst().orElse(null);
                if (screnTranDTO != null) {
                    locationList.add((AdmScrenTranBasQryDTO) screnTranDTO.clone());
                }
            }
        }

        List<AdmScrenTranBasQryDTO> outputList = new ArrayList<>();
        if (!locationList.isEmpty()) {
            for (int i = locationList.size() - 1; i >= 0; i--) {
                outputList.add(locationList.get(i));
            }
        }
        return outputList;
    }

    /**
     * 메뉴ID를 기준으로 메뉴정보가 존재하는 지 여부
     * @param menuList 전체 메뉴 목록
     * @param screnTranId : 메뉴ID
     */
    public static boolean hasMenu(List<AdmScrenTranBasQryDTO> menuList, String screnTranId) {
		boolean isAuthUri = menuList.stream()
				.filter(x -> x.getScrenTranId().equals(screnTranId))
				.findFirst().isPresent();
		return isAuthUri;
    }

    /**
     * 메뉴ID를 기준으로 하위의 메뉴가 존재하는 지 여부
     * @param menuList 전체 메뉴 목록
     * @param screnTranId : 메뉴ID
     */
    public static boolean hasSubMenu(List<AdmScrenTranBasQryDTO> menuList, String screnTranId) {
        if (menuList == null || menuList.isEmpty() || StringUtils.isEmpty(screnTranId)) {
            return false;
        }

		boolean isAuthUri = menuList.stream()
				.filter(x -> StringUtils.trimToEmpty(x.getScrenTranId()).equals(screnTranId))
				.filter(x -> "Y".equals(x.getMenuRepstYn()))
				.findFirst().isPresent();
		return isAuthUri;
    }

    /**
     * 메뉴ID를 기준으로 메뉴정보 DTO를 반환
     * @param menuList 전체 메뉴 목록
     * @param screnTranId : 메뉴ID
     */
    public static AdmScrenTranBasQryDTO getMenuData(List<AdmScrenTranBasQryDTO> menuList, String screnTranId) {
        if (menuList == null || menuList.isEmpty() || StringUtils.isEmpty(screnTranId)) {
            return null;
        }

		return menuList.stream().filter(x -> x.getScrenTranId().equals(screnTranId)).findFirst().orElse(null);
    }

    /**
     * 메뉴ID를 기준으로 메뉴명을 반환
     * @param menuList 전체 메뉴 목록
     * @param screnTranId : 메뉴ID
     */
    public static String getMenuName(List<AdmScrenTranBasQryDTO> menuList, String screnTranId) {
        if (menuList == null || menuList.isEmpty() || StringUtils.isEmpty(screnTranId)) {
            return null;
        }

		return menuList.stream().filter(x -> x.getScrenTranId().equals(screnTranId)).findFirst().orElseGet(AdmScrenTranBasQryDTO::new).getScrenTranNm();
    }

	/**
	 * 날짜 포멧변경 출력
	 */
	public static String formatDate(Object value, String pattern) {
		String output = "";
		if (value == null) {
			output = "";
		} else if (value instanceof Date) {
			output = DateUtil.getString((Date) value, pattern);
		} else {
			output = (String) value;
			if (StringUtils.isNotBlank(output)) {
				SimpleDateFormat sdf = new SimpleDateFormat("yyyyMMdd");
				Date d = null;
				try {
					d = sdf.parse(output);
					sdf.applyPattern(pattern);
					output = sdf.format(d);
				} catch (ParseException e) {
					output = "";
				}
			}
		}
		return output;
	}

	/**
	 * 숫자 포멧변경 출력
	 */
	public static String formatNumber(String value, String pattern) {
		String output = value;

		if (output != null && !output.isEmpty()) {
			DecimalFormat decimalFormat = new DecimalFormat(pattern);
			double lAmt = Double.parseDouble(output.trim());
			output = decimalFormat.format(lAmt);
		}

		return output;
	}
