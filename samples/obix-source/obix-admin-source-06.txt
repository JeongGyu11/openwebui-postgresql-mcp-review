/**
* 화면거래목록 출력
*/
function fncScrenTranList(data){
	var result = data.result.grid;
	var html = "";
	$("#tScrenTranInfo").empty();
	console.log(result);
	if(result!=null && result.length > 0){
		var idx = 1;
		for(var i in result){
			var obj = result[i];
			if(obj.screnTranLvel > 0){
				html += '<tr class="text-center border-bottom">';
				html += '	<td>' + idx + '</td>';				// Idx
				html += ' 	<td>' + biz.getCodeName($admBzwkLclsCd, obj.admBzwkLclsCd) + '</td>';		// 대분류코드
				html += '	<td>' + biz.getCodeName($admBzwkMclsCd, obj.admBzwkMclsCd) + '</td>';		// 중분류코드
				html += '	<td><a href="javascript:popASYAU10200(`'+obj.screnTranId+'`);">' + obj.screnTranId + '</a></td>';		// 화면ID
				html += '	<td>' + util.nvl(obj.screnTranNm, '') + '</td>';			// 화면명
				html += '	<td>' + biz.getCodeName($screnTranDvCd, obj.screnTranDvCd) + '</td>';		// 화면구분
				html += '	<td>' + util.nvl(obj.urlAddr, '') + '</td>';				// url주소
				html += '	<td>' + obj.menuRepstYn + '</td>';			// 메뉴표시여부
				html += '	<td>' + obj.wholAllowYn + '</td>';			// 전체표시여부
				html += '	<td>' + obj.sstvInfoYn + '</td>';			// 민감정보허용여부
				html += '	<td>' + obj.repstOrdr + '</td>';			// 표시순서
				html += '	<td>' + obj.useYn + '</td>';				// 사용여부
				html += '</tr>';
				
				idx++;
			}
		}					
	}else{
		html += '<tr><td colspan="12" class="text-center border-bottom">내용없음</td></tr>'
	}
	
	$("#tScrenTranInfo").append(html);	
}

function ASYAU10101() {
	httpAjax('ASYAU10101.json', {}, function(data) {
		var result = data.result;

		// 화면거래에 등록되지 않은 URL
		var serverOnlyList = result['serverOnlyList'];
		for (var i = 0; i < serverOnlyList.length; i++) {
			var item = serverOnlyList[i];
			
			var _html = '';
			if (i < serverOnlyList.length - 1) {
				_html += '<tr>';
			} else {
				_html += '<tr class="border-bottom">';
			}
			_html += '	<td>' + (i + 1) + '</td>';
			_html += '	<td>' + item.urlPattern + '</td>';
			_html += '	<td>' + item.controllerName + '</td>';
			_html += '	<td>' + util.nvl(item.screnTranNm,'') + '</td>';
			_html += '</tr>';
			$('#tServerOnlyList').append(_html);
		}


		// 프로그램으로 구현되지 않은 URL
		var clientOnlyList = result['clientOnlyList'];
		for (var i = 0; i < clientOnlyList.length; i++) {
			var item = clientOnlyList[i];

			var _html = '';
			if (i < clientOnlyList.length - 1) {
				_html += '<tr>';
			} else {
				_html += '<tr class="border-bottom">';
			}
			_html += '	<td>' + (i + 1) + '</td>';
			_html += '	<td>' + item.urlPattern + '</td>';
			_html += '	<td>' + item.screnTranId + '</td>';
			_html += '	<td>' + item.screnTranNm + '</td>';
			_html += '</tr>';
			$('#tClientOnlyList').append(_html);
		}

	});
}

/**

function fncScrenTranInfo(data) {
    if (scrinSeCode == "old") {
        $("#popScrenTranId").val(data.screnTranId); // 화면거래아이디
        $('#popAdmBzwkLclsCd').val(data.admBzwkLclsCd).prop('selected', 'selected'); // 어드민업무대분류코드

        $('#popScrenTranDvCd').val(data.screnTranDvCd).prop('selected', 'selected'); // 화면거래구분코드

        if (data.screnTranId.length > 9) {
            $('#popScrenSeq').val(data.screnTranId.substr(6, 2)); // 화면SEQ
            $('#popTranSeq').val(data.screnTranId.substr(8, 2)); // 거래SEQ
        }

        $("#popUpScrenTranId").val(data.upScrenTranId); // 상위화면거래아이디
        $("#popScrenTranLvel").val(data.screnTranLvel); // 화면거래레벨

        $("#popScrenTranNm").val(data.screnTranNm); // 화면거래명
        $("#popUrlAddr").val(data.urlAddr); // URL주소
        $("#popDescCtnt").val(data.descCtnt); // 설명내용

        $('#popMenuRepstYn').val(data.menuRepstYn).prop('selected', 'selected'); // 메뉴표시여부
        $('#popWholAllowYn').val(data.wholAllowYn).prop('selected', 'selected'); // 전체허용여부
        $('#popSstvInfoYn').val(data.sstvInfoYn).prop('selected', 'selected'); // 민감정보여부

        $("#popRepstOrdr").val(data.repstOrdr); // 설명내용

        $('#popUseYn').val(data.useYn).prop('selected', 'selected'); // 사용여부
    }

	// 상위화면거래 출력
	if (data.screnTranList && data.screnTranList.length > 0) {
        var optionHtml = '';
		for (var i in data.screnTranList) {
			var obj = data.screnTranList[i];

            optionHtml += '<option value="' + obj.screnTranId + '" screnTranDvCd="' + util.nvl(obj.screnTranDvCd,'') + '" admBzwkLclsCd="' + util.nvl(obj.admBzwkLclsCd,'') + '" admBzwkMclsCd="' + util.nvl(obj.admBzwkMclsCd,'') + '" screnTranLvel="' + util.nvl(obj.screnTranLvel,'') + '">';
            for (var i = 0; i < obj.screnTranLvel; i++) {
                optionHtml += '　　';
			}
            optionHtml += obj.screnTranNm + ' (' + obj.screnTranId + ')';
            optionHtml += '</option>';
		}
        $('#selectUpScrenTranId').append(optionHtml);
    }

}

		obj.screnTranNm = item.screnTranNm;

			obj.useYn = "Y";
		}else{
			obj.useYn = "N";
		}

		gArr_screnGroupList.push(obj);
	});

	var arrData = new Array();
	for(var idx in gArr_screnGroupList){
		var obj = gArr_screnGroupList[idx];

		// jsTree 체크 처리
		var state = new Object();

		if("Y" == obj.useYn) { // 그룹권한 있음

		}else{
			state.selected = false;
		}

		// jsTree 데이터 입력
		var data = new Object();
		data.id = obj.screnTranId;
		data.parent = util.nvl(obj.upScrenTranId, "#");
		data.text = obj.screnTranNm;
		data.state = state;

		arrData.push(data);
	}

	$("#jsTree").jstree({

	httpAjax('ASYAU40001.json', jsonData, function(data){
		// 사용자목록 출력
 		fncUserList(data);
 		fncPagination(data.result);	// 페이징처리
  	});
}

/*********************************************************************
 * 기능함수 
 * function fnc********(data){ }
*********************************************************************/
/**
 * 사용자목록 출력
 */
function fncUserList(data){
	
	var result = data.result.grid;
	var html = "";
	$("#tUserInfo").empty();
	
	if(result.length > 0){
		for(var i in result){
			var obj = result[i];
			
			html += '<tr class="text-center border-bottom">';
			html += '	<td>' + obj.num + '</td>';
			html += ' 	<td><a href="javascript:popASYAU40100(`'+obj.userId+'`);">' + obj.userId + '</a></td>';				//아이디
			html += '	<td><a href="javascript:popASYAU40100(`'+obj.userId+'`);">' + util.nvl(obj.userNm, '') + '</a></td>';				//이름
			html += '	<td>' + biz.getCodeName($admUserStusCd, util.nvl(obj.admUserStusCd, '')) + '</td>';		//사용자상태
			html += '	<td>' + util.nvl(obj.cpNo, '') + '</td>';					//전화번호
			html += '	<td>' + util.nvl(obj.email, '') + '</td>';				//이메일
			html += '	<td>' + util.nvl(obj.groupNm, '') + '</td>';				//그룹
			html += '	<td>' + (obj.lginTmsp ? format.formatTimeStamp(new Date(obj.lginTmsp)) : '') + '</td>';		//최종로그인 일시
			html += '	<td><a href="#none" class="btn btn-outline-primary" onclick="popASYAU40200(`'+obj.userId+'`, `' + util.nvl(obj.userNm, '') + '`);return false;">권한설정</a></td>';		//권한설정
			html += '</tr>';
		}
	}else{	
		html += '<tr><td colspan="10" class="text-center border-bottom">내용없음</td></tr>'
	}
	$("#tUserInfo").append(html);	
}
